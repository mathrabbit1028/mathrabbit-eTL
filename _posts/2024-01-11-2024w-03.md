## 1. 유전체 정렬 문제 (0:16:35)

두 문자열 각각에서 $i$번째 문자, $j$번째 문자까지 사용했을 때 가능한 최대 가중치 합을 $dp[i][j]$에 계산하자. 그러면 두 글자 모두 사용, 첫 번째 문자열에 공백 삽입, 두 번째 문자열에 공백 삽입의 세 가지 경우로 나누어 점화식을 세울 수 있고, 이 값을 바탕으로 역추적도 할 수 있다. 시간복잡도는 $O(NM)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
 
int n, m;
string x, y;
int dp[1010][1010];
string a, b;
 
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
 
    cin >> n >> m;
    cin >> x >> y;
 
    for (int i = 0; i <= n; i++) dp[i][0] = -2*i;
    for (int j = 0; j <= m; j++) dp[0][j] = -2*j;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            dp[i][j] = -1e9;
        }
    }
 
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            dp[i][j] = max(dp[i][j], dp[i][j-1]-2);
            dp[i][j] = max(dp[i][j], dp[i-1][j]-2);
            if (x[i-1] == y[j-1]) dp[i][j] = max(dp[i][j], dp[i-1][j-1]+2);
            else dp[i][j] = max(dp[i][j], dp[i-1][j-1]-1);
        }
    }
 
    cout << dp[n][m] << "\n";
 
    int i = n, j = m;
    while (i + j > 0) {
        if (j > 0 && dp[i][j] == dp[i][j-1]-2) {
            a.push_back('_');
            b.push_back(y[j-1]);
            j--;
        }
        else if (i > 0 && dp[i][j] == dp[i-1][j]-2) {
            a.push_back(x[i-1]);
            b.push_back('_');
            i--;
        }
        else {
            a.push_back(x[i-1]);
            b.push_back(y[j-1]);
            i--; j--;
        }
    }
 
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    cout << a << "\n" << b << "\n";
 
    return 0;
}
```

## 2. 트리 공예 (0:59:50)

후자의 경우에는 모든 트리가 나올 가능성이 같지만, 전자의 경우에는 트리의 모양에 따라 가능한 경우의 수가 다르므로 확률이 달라진다. 구체적으로 후자의 경우에는 전체 경우의 수가 $N^{N-2}$이고, 전자의 경우에는 가능한 경우의 수가 $N! \times (N-1)!$이다.

$N=64$에 대해 이 값의 차이를 계산하면 트리의 모양이 주어질 때 전자의 방법으로 만들 수 있는 경우의 수(뽑힌 정점의 나열)이 얼마 이상이 되어야 전자일 확률이 더 높은지 계산할 수 있다. 즉, 우리는 트리의 모양이 주어질 때 전자의 방법으로 만들 수 있는 경우의 수만 세면 된다.

일단 첫 번째로 뽑혔을 시작 정점을 하나 정하자. 그러면 그 정점을 루트로 하여 DFS를 돌리며 트리 dp를 짜자. 그러면 다음 관계를 얻을 수 있다.

- 노드 $v$의 모든 자식 노드 $u$에 대해
    - $u$의 서브트리에 속하는 간선의 연결 순서로 가능한 것은 $dp[u]$가지이다.
    - 서로 다른 서브트리에 속하는 간선의 연결 순서는 아무 순서가 되어도 상관 없다.
    - 즉, 우리는 $sz[u]$개씩 있는 같은 물체를 나열하는 경우의 수를 추가로 곱하면 전체 경우의 수를 알 수 있다.
    
    $$
    dp[v] = \prod_u {dp[u]} \times \frac{\left(\sum_u {sz[u]}\right)!}{\prod_u {(sz[u])!}}
    $$
    

경우의 수가 long long 의 범위보다 커지는데 나머지가 아닌 실제 값이 중요하므로 $\log_2$를 기준으로 생각하자. 그러면 곱셈이 덧셈으로 바뀌게 된다. 그러면 $O(N^2)$에 각각을 루트로 할 때 경우의 수의 $\log_2$를 구해낼 수 있다.

그러면 경우의 수의 덧셈을 처리해야 하므로 $\log_2 x$와 $\log_2 y$로 $\log_2 (x+y)$를 구해야 한다. 이는 일반성을 잃지 않고 $x>y$로 가정했을 떄 $\log_2 (x+y) = \log_2 x + \log_2 (1+y/x)$이고 $y/x < 1$이므로 그냥 $y/x$를 $2^{\log_2 y - \log_2 x}$로 계산하여 대입하거나 $\log(1+x)$의 테일러 전개를 사용하면 된다.

따라서 시간복잡도는 $O(TN^2)$이고 아래 코드는 99.14%의 데이터에 대해 올바른 답을 낸다.

```cpp
#include <bits/stdc++.h>
#include "craft.h"
using namespace std;
const int N = 64;
double faclg[65];
double standard = 0;
 
void init() {
    faclg[0] = 0;
    for (int i = 1; i <= N; i++) {
        faclg[i] = faclg[i-1] + log2(i);
    }
    standard -= log2(N) * (N-2);
    standard += faclg[N] + faclg[N-1];
 
    //cout << standard << "\n";
}
 
vector<int> adj[64];
int sz[64];
 
int cnt = 0;
double DFS(int v, int par = -1) {
    double ret = 0;
    sz[v] = 1;
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        if (u == par) continue;
        ret += DFS(u, v);
        ret -= faclg[sz[u]];
        sz[v] += sz[u];
    }
    ret += faclg[sz[v]-1];
    return ret;
}
 
int classify(vector<int> parent) {
    for (int i = 0; i < N; i++) adj[i].clear();
    for (int i = 1; i < N; i++) {
        adj[parent[i]].push_back(i);
        adj[i].push_back(parent[i]);
    }
    
    double ans = -1e9;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++) sz[j] = 0;
        double res = DFS(i);
        if (ans < res) swap(ans, res);
        ans = ans + pow(2, res-ans);
    }
 
    if (ans > standard) return 1;
    else return 2;
}
```

## 3. 분단의 슬픔 (3:24:59, 99.9919점)

위쪽에 있다고 알려진 정점과 아래쪽에 있다고 알려진 정점에 대해 컨벡스 헐을 그리자. 그러면 만들어진 컨벡스 헐에서 두 변을 골랐을 때, 두 변의 연장선의 교점의 양쪽에 두 변이 있다면 그 두 변에 의해 생긴 경계선만이 답의 후보가 될 수 있다. 이는 점이 랜덤으로 찍혔기 때문에 가능한 풀이이다.

컨벡스 헐의 크기가 $O(N)$이 되어 $O(N^2)$의 시간복잡도가 아니냐고 물어볼 수 있지만, 랜덤으로 찍힌 점이기 때문에 컨벡스 헐의 정점 수가 $20$개 내외이다. 따라서 충분히 시간 내에 통과할 수 있다.

(100점은 왜 안되는지 모르겠는데 자잘한 이슈인듯 합니다)

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<double, double> pdd;
 
struct pt {
    double x, y;
    bool operator<(const pt& other) const {
        return y < other.y;
    }
};
 
bool ccw(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return true;
    else return false;
}
 
pt O;
bool compare(pt A, pt B) {
    return !ccw(A, O, B);
}
 
pdd get_line(pt A, pt B) {
    double m = (A.y - B.y) / (A.x - B.x);
    return {m, A.y - m * A.x};
}
 
pt get_intersect(pdd l, pdd m) {
    double x = (m.second - l.second) / (l.first - m.first);
    return {x, l.first * x + l.second};
}
 
vector<int> x1_, y1_, x2_, y2_;
int get_score(vector<double> info) {
    int cnt = 0;
    for (int i = 0; i < x1_.size(); i++) {
        if (x1_[i] >= info[0]) {
            if (y1_[i] >= info[3] * (x1_[i] - info[0]) + info[1]) cnt++;
        }
        if (x1_[i] < info[0]) {
            if (y1_[i] >= info[2] * (x1_[i] - info[0]) + info[1]) cnt++;
        }
    }
    for (int i = 0; i < x2_.size(); i++) {
        if (x2_[i] >= info[0]) {
            if (y2_[i] < info[3] * (x2_[i] - info[0]) + info[1]) cnt++;
        }
        if (x2_[i] < info[0]) {
            if (y2_[i] < info[2] * (x2_[i] - info[0]) + info[1]) cnt++;
        }
    }
    return cnt;
}
 
vector<pt> pts;
stack<int> stk;
vector<pt> convex;
vector<double> convex_hull(vector<int> x, vector<int> y) {
    pts.clear();
    while (stk.size() > 0) stk.pop();
    convex.clear();
 
    for (int i = 0; i < x.size(); i++) pts.push_back((pt){x[i], y[i]});
 
    sort(pts.begin(), pts.end());
    O = pts[0];
    sort(pts.begin() + 1, pts.end(), compare);
 
    stk.push(0);
    stk.push(1);
    for (int i = 2; i < pts.size(); i++) {
        int top = stk.top(); stk.pop();
        while ( stk.size() > 0 && !ccw(pts[stk.top()], pts[top], pts[i]) ) {
            top = stk.top();
            stk.pop();
        }
        stk.push(top);
        stk.push(i);
    }
 
    while (stk.size() > 0) {
        convex.push_back(pts[stk.top()]);
        stk.pop();
    }
    reverse(convex.begin(), convex.end());
 
    vector<double> ret;
    for (int i = 0; i < convex.size(); i++) {
        pt A = convex[i];
        pt B = convex[(i+1)%convex.size()];
        if (A.x == B.x) continue;
        if (A.x > B.x) swap(A, B);
 
        //cout << get_line(A, B).first << "\n";
        //cout << A.x << " " << A.y << "\n";
 
        for (int j = 0; j < convex.size(); j++) {
            if (i == j) continue;
            pt C = convex[j];
            pt D = convex[(j+1)%convex.size()];
            if (C.x == D.x) continue;
            if (C.x > D.x) swap(C, D);
 
            pdd l1 = get_line(A, B);
            pdd l2 = get_line(C, D);
            if (l1.first == l2.first) continue;
            pt X = get_intersect(l1, l2);
            if (X.x > B.x && X.x > D.x) continue;
            if (X.x < A.x && X.x < C.x) continue;
 
            //if (A.x > C.x) assert(D.x < A.x);
            //if (A.x < C.x) assert(B.x < C.x);
 
            vector<double> temp;
            temp.push_back(X.x);
            temp.push_back(X.y);
            temp.push_back(l1.first);
            temp.push_back(l2.first);
            if (A.x > C.x) swap(temp[2], temp[3]);
            //for (int i = 0; i < 4; i++) cout << temp[i] << " "; cout << "\n";
 
            if (ret.size() == 0) ret = temp;
            else if (get_score(ret) < get_score(temp)) ret = temp;
        }
    }
    //cout << "\n";
    
    return ret;
}
 
 
vector<double> classify(vector<int> x1, vector<int> y1, vector<int> x2, vector<int> y2) {
 
    x1_ = x1;
    y1_ = y1;
    x2_ = x2;
    y2_ = y2;
 
    vector<double> ret1 = convex_hull(x1, y1);
    vector<double> ret2 = convex_hull(x2, y2);
 
    if (get_score(ret1) > get_score(ret2)) return ret1;
    else return ret2;
 
}
```

## 4. 이환이의 데이터 뚫기 (92점)

첫 번째 방법의 경우에는 $i$번째 반복을 하면 $i$번쨰 이전의 수가 정해지므로 매번 수를 하나씩 뽑는 순열과 같다. 즉, 각 순열이 나올 확률이 동일하다.

두 번쨰 방법의 경우에는 $N$이 작을 떄에 대해 모든 경우를 $O(N^N)$에 시도해보면 $i$번째 수가 $i$보다 큰 인덱스로 갈 확률이 높음을 알 수 있다.

- 각 $N$에 대해 $i$번째 자리에 있던 수가 $1 \sim N$번째로 가는 각 경우의 수

이를 바탕으로 다음 92점 풀이를 구현할 수 있다. 513은 경험적으로 찾은 값이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int t;
int n, arr[1010], rev[1010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> arr[i];
        for (int i = 1; i <= n; i++) rev[arr[i]] = i;
        int cnt = 0;
        for (int i = 1; i <= n; i++) {
            cnt += (arr[i] > i);
        }
        if (cnt >= 513) cout << "2\n";
        else cout << "1\n";
    }
 
    return 0;
}
```
