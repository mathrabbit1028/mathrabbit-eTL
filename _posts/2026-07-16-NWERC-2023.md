## 문제 K

grid graph를 만들면 $\mathcal{O}(N^2)$로 풀 수 있다. $(a, b)$와 $(c, d)$ 이동 규칙에 따라 이동할 수 있는 칸을 간선으로 연결하자. 좌상단을 포함하는 컴포넌트의 크기가 답이다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, ch[550][550][2];
int a, b, c, d;
int dx[4] = {-1, -1, 1, 1}, dy[4] = {-1, 1, 1, -1};

void DFS(int x, int y, int f) {
    if (x < 1 || x > n || y < 1 || y > n) return;
    if (ch[x][y][f]) return;
    ch[x][y][f] = 1;
    int u, v;
    if (f) u = a, v = b;
    else u = c, v = d;
    for (int t = 0; t < 4; t++) {
        int x2 = x + u * dx[t], y2 = y + v * dy[t];
        DFS(x2, y2, 1-f);
    }
    for (int t = 0; t < 4; t++) {
        int x2 = x + v * dx[t], y2 = y + u * dy[t];
        DFS(x2, y2, 1-f);
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> n;
    cin >> a >> b >> c >> d;
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            ch[i][j][0] = ch[i][j][1] = 0;
        }
    }
    DFS(1, 1, true);
    DFS(1, 1, false);

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (ch[i][j][0] || ch[i][j][1]) ans++;
        }
    }

    cout << ans << "\n";

    return 0;
}
```


## 문제 L
$x$와 $y$를 5로 나눈 나머지가 같은 $(x, y)$들을 모두 살펴보면 모든 배가 걸림을 보장할 수 있다. 배가 걸린 이후에는 배를 기준으로 상하좌우 네 칸씩을 추가로 살피면 충분하다. 곧, 필요한 푓수는 $n^2/5 + 16k$이고 $2\,500$ 이하이므로 충분히 가능하다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, k;
int grid[110][110];

void f(int x, int y) {
    for (int t = 1; t <= 4; t++) {
        if (y+t >= n) break;
        cout << x+1 << " " << t+y+1 << endl;
        string s; cin >> s;
    }
    for (int t = 1; t <= 4; t++) {
        if (y-t < 0) break;
        cout << x+1 << " " << y-t+1 << endl;
        string s; cin >> s;
    }
    for (int t = 1; t <= 4; t++) {
        if (x+t >= n) break;
        cout << x+t+1 << " " << y+1 << endl;
        string s; cin >> s;
    }
    for (int t = 1; t <= 4; t++) {
        if (x-t < 0) break;
        cout << x-t+1 << " " << y+1 << endl;
        string s; cin >> s;
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> n >> k;

    for (int i = 0; i < n; i++) {
        for (int j = i % 5; j < n; j += 5) {
            cout << i+1 << " " << j+1 << endl;
            string s;
            cin >> s;
            if (s != "miss") grid[i][j] = 1;
        }
    }

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == 1) {
                f(i, j);
            }
        }
    }

    return 0;
}
```

## 문제 F

$a/b$에서 $b$의 일부 자리수를 제거하는 모든 경우의 수에 대해 다음을 차례대로 한다:
- $b$의 일부 자리수를 제거하여 $b'$를 만든다.
- $b'$가 $d$로 나누어 떨어지지 않으면 버린다.
- $b' \cdot (c/d)$를 계산하여 $a'$를 만든다.
- $a'$가 $a$의 일부 자리수 제거로 만들 수 있는 수인지 확인한다.

digit의 relative order가 보존되어야 하기 때문에 이를 고려하여 $a$의 일부 자리수를 제거하여 $a'$를 만들 수 있는지 확인해야 한다.

```cpp
#include <bits/stdc++.h>
using namespace std;

long long a, b, c, d;

long long gcd(long long n, long long m) {
    if (n < m) swap(n, m);
    if (m == 0) return n;
    n = n % m;
    return gcd(n, m);
}

struct maskingReturn {
    long long b_ = 0;
    int count[10] = {0, };
};

vector<int> to_vector(long long n) {
    vector<int> res;
    while (n > 0) {
        res.push_back(n % 10);
        n = n / 10;
    }
    return res;
}

maskingReturn masking(long long n, int mask) {
    vector<int> vec = to_vector(n);
    vector<int> rem;
    maskingReturn res;
    for (int i = 0; i < 18; i++) {
        if (i >= vec.size()) break;
        if (mask & (1 << i)) {
            rem.push_back(vec[i]);
        } else {
            res.count[vec[i]]++;
        }
    }
    reverse(rem.begin(), rem.end());
    if (rem.size() > 0 && rem.front() != 0) {
        for (int digit : rem) {
            res.b_ *= 10;
            res.b_ += digit;
        }
    }
    return res;
}

bool is_subvec(vector<int>& large, vector<int>& small, int p, int q) {
    if (q == small.size()) return true;
    if (p == large.size()) return false;
    if (large[p] == small[q]) {
        if (is_subvec(large, small, p+1, q+1)) return true;
    }
    else {
        if (is_subvec(large, small, p+1, q)) return true;
    }
    return false;
}

bool check_possible(long long a, long long a_, maskingReturn &ret) {
    vector<int> init = to_vector(a);
    vector<int> final = to_vector(a_);
    
    for (int i = 0; i < 10; i++) {
        int cnt = 0;
        for (int digit : init) if (digit == i) cnt++;
        for (int digit : final) if (digit == i) cnt--;
        if (ret.count[i] != cnt) return false;
    }

    if (is_subvec(init, final, 0, 0)) return true;
    else return false;
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> a >> b >> c >> d;
    long long g = gcd(c, d);
    c = c/g;
    d = d/g;

    bool f = false;
    if (c > d) {
        f = true;
        swap(c, d);
        swap(a, b);
    } // c < d

    for (int mask = 0; mask < (1<<18); mask++) {
        maskingReturn ret = masking(b, mask);
        if (ret.b_ == 0 || ret.b_ % d != 0) continue;
        long long a_ = ret.b_ / d * c;
        if (check_possible(a, a_, ret)) {
            cout << "possible\n";
            if (f) cout << ret.b_ << " " << a_ << "\n";
            else cout << a_ << " " << ret.b_ << "\n";
            return 0;
        }
    }

    cout << "impossible\n";

    return 0;
}
```
## 문제 E

로그를 한 번 씌운다고 생각하면,
$$\log (x_1)^{(x_2)} = x_2 \log x_1$$
가 된다. 이것은, $x_1$에 다른 $x_2$ 들을 여러 번 지수로 올리는 것이 로그를 씌운 이후에는 곱하는 것과 같다는 것이다. 즉, 중간에 나오는 모든 수는
$$x_p = \prod_{i=1}^{k} x_{t_i} \cdot \log 2023$$
와 같은 꼴이 되도록 하는 고유의 집합 $C(p) = \{t_1, t_2, \ldots, t_k\}$가 존재한다. 이 관점 하에서 $x_i \leftarrow x_i^{x_j}$는 $C$에 $x_j$를 추가하는 것과 같다.

$C(p)$와 $C(q)$를 어떻게 비교할까? $N < 2023$을 생각해보자. $| C(p) | \leq N < 2023$이므로 
$$\max_{t_i \in C(p)} x_{t_i} > \max_{t_i \in C(q)} x_{t_i}$$
라면 $x_p$가 $x_q$보다 클 수밖에 없다. 왜냐하면 $2023$의 거듭제곱들로만 만들어지는 수의 집합을 정렬했을 때, 직전 수를 적어도 $2023$번 곱해야 다음 수가 나오기 때문이다. 따라서, $C(p)$와 $C(q)$를 비교할 때는 $\max_{t_i \in C(p)} x_{t_i}$와 $\max_{t_i \in C(q)} x_{t_i}$를 비교하면 된다. 만약 같다면 그 다음으로 큰 수를 비교하면 된다. 다른 말로, 문자열의 사전순 비교와 비슷하다. 

하지만 이를 나이브하게 구현하면 두 수의 비교에 최대 $\mathcal{O}(N^2)$이 소요될 수 있기 때문에 최적화를 해야 한다. 이를 위해 $p$라는 이름의 노드가 $t_1, t_2, \cdots, t_k$ 번호의 자식 노드들을 가리키고 있다고 하자. $t$들 중에 리프(즉, $2023$ 혼자)가 아닌 것이 있다면 자식 노드 중 일부도 트리일 것이다. 이제 우리는 새로운 쿼리에 의해 숫자가 업데이트될 때, 새로운 노드를 만들고 현재까지 나왔던 모든 노드의 집합을 정렬된 상태로 유지할 것이다:
- 먼저, $x_i$의 원래 자식 노드들에 $x_j$를 삽입해야 하므로, 순회하면서 알맞는 위치를 찾는다. 현재까지 나왔던 노드들은 정렬되어 있으므로 $\mathcal{O}(N)$에 가능하다.
- 이제 새로운 노드를 삽입한다:
    - 이분 탐색으로 가장 큰 수만 봤을 때 새로운 노드와 같은 노드들의 범위를 찾는다.
    - 그 범위 안에서 다음으로 큰 수를 비교한다.
    - 이것을 새로운 노드의 모든 자식을 순회할 때까지 반복한다.
    - 그러면 새로운 노드가 삽입될 위치를 $\mathcal{O}(N \log N)$에 찾을 수 있다.

구현해서는 편리를 위해 $\mathcal{O}(N^2)$의 비교 테이블을 만들어서, 두 노드가 어떤 관계인지 저장해두고, 새로운 노드가 들어올 때마다 갱신한다. 정렬을 하고 나면 테이블을 채우는 것은 새 노드마다 $\mathcal{O}(N)$이므로 문제되지 않는다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, m;

class Node {
public:
    static int count;

    int id;
    vector<Node*> children;
    Node() {
        id = ++count;
    }
};
int Node::count = 0;

Node* x[1010];
vector<Node*> sorted_nodes;
int compare_table[2020][2020];

int compare(Node* a, Node* b) {
    return compare_table[a->id][b->id];
}

void inserting(Node* new_node, int L, int lt, int rt) {
    if (L == new_node->children.size() + 1) {
        for (int i = 0; i < lt; i++) {
            compare_table[sorted_nodes[i]->id][new_node->id] = 1;
            compare_table[new_node->id][sorted_nodes[i]->id] = -1;
        }
        for (int i = lt; i < rt; i++) {
            compare_table[sorted_nodes[i]->id][new_node->id] = 0;
            compare_table[new_node->id][sorted_nodes[i]->id] = 0;
        }
        for (int i = rt; i < sorted_nodes.size(); i++) {
            compare_table[new_node->id][sorted_nodes[i]->id] = 1;
            compare_table[sorted_nodes[i]->id][new_node->id] = -1;
        }
        compare_table[new_node->id][new_node->id] = 0;
        sorted_nodes.insert(sorted_nodes.begin() + lt, new_node);
        return;
    }

    int st = lt, ed = rt; // greater than or equal to new_node
    while (st < ed) {
        int mid = (st + ed) / 2;
        int cmp;
        if (L >= new_node->children.size() && L >= sorted_nodes[mid]->children.size()) {
            cmp = 0;
        } else if (L >= new_node->children.size()) {
            cmp = 1;
        } else if (L >= sorted_nodes[mid]->children.size()) {
            cmp = -1;
        } else {
            cmp = compare(new_node->children[L], sorted_nodes[mid]->children[L]);
        }
        if (cmp < 0) {
            st = mid + 1;
        } else {
            ed = mid;
        }
    }
    int new_lt = st;
    st = lt, ed = rt;
    while (st < ed) { // greater than new_node
        int mid = (st + ed) / 2;
        int cmp;
        if (L >= new_node->children.size() && L >= sorted_nodes[mid]->children.size()) {
            cmp = 0;
        } else if (L >= new_node->children.size()) {
            cmp = 1;
        } else if (L >= sorted_nodes[mid]->children.size()) {
            cmp = -1;
        } else {
            cmp = compare(new_node->children[L], sorted_nodes[mid]->children[L]);
        }
        if (cmp <= 0) {
            st = mid + 1;
        } else {
            ed = mid;
        }
    }
    int new_rt = st;

    inserting(new_node, L + 1, new_lt, new_rt);
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    cin >> n >> m;

    for (int i = 1; i <= n; i++) {
        x[i] = new Node();
        inserting(x[i], 0, 0, sorted_nodes.size());
    }

    while (m--) {
        char c;
        int a, b;
        cin >> c;
        if (c == '!') {
            cin >> a >> b;
            Node* next_node = new Node();
            for (Node* child : x[a]->children) {
                next_node->children.push_back(child);
            }
            for (auto it = next_node->children.begin(); it != next_node->children.end(); ++it) {
                if (compare(*it, x[b]) > 0) {
                    next_node->children.insert(it, x[b]);
                    break;
                }
            }
            if (next_node->children.size() == x[a]->children.size()) {
                next_node->children.push_back(x[b]);
            }
            inserting(next_node, 0, 0, sorted_nodes.size());
            x[a] = next_node;
        }
        else {
            cin >> a >> b;
            int result = compare(x[a], x[b]);
            if (result == 0) {
                cout << "=\n";
            } else if (result > 0) {
                cout << "<\n";
            } else {
                cout << ">\n";
            }
        }
    }

    return 0;
}
```
