## 2023f-06-01

증명은 생략하겠지만, 어떤 세 점도 한 직선 위에 있지 않기 때문에 $(x, y)$로 정렬 후 앞에부터 3개씩 묶으면 가능한 하나의 해답이 된다.

```cpp
#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int t, n;
pair<pii, int> pts[3030];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < 3*n; i++) {
            pts[i].second = i + 1;
            cin >> pts[i].first.x >> pts[i].first.y;
        }
        sort(pts, pts + 3*n);
        for (int i = 0; i < 3*n; i += 3) {
            cout << pts[i].second << " " << pts[i + 1].second << " " << pts[i + 2].second << "\n";
        }
    }
 
    return 0;
}
```

## 2023f-06-02

구하는 것은 숫자의 존재 여부를 표시한 배열에서 연속하게 붙어 있는 구간의 길이 $a_1, a_2, \cdots, a_k$에 대해 $\sum (a_i+1)/2$와 같다. 즉, 연속하게 붙어 있는 구간의 시점 및 종점 쌍 $s, e)$만 관리하면 되며, `std::set`을 사용하면 $O(\log N)$의 시간에 관리할 수 있다.

구체적인 구현은 다음 네 가지 경우로 나누어진다.

- $i$번째 수 $v$가 한 구간 $[s, e)$에 포함되는 경우: 답 증가 없음
- $i$번째 수 $v$에 대해 구간 $[v + 1, e)$와 $[s, v)$ 중 하나만 존재하는 경우: 인접한 구간의 길이가 1 증가하므로, 그 구간의 길이가 짝수였던 경우에만 답이 $1$ 증가
- $i$번째 수 $v$에 대해 구간 $[v + 1, e)$와 $[s, v)$가 모두 존재하는 경우: 두 구간이 합쳐지므로, 두 구간의 길이가 모두 짝수였던 경우에만 답이 $1$ 증가
- $i$번째 수 $v$에 대해 구간 $[v + 1, e)$와 $[s, v)$ 모두 존재하지 않는 경우: 길이 $1$의 새로운 구간이 추가되므로 답이 $1$ 중가

시간복잡도는 $O(N \log N)$이다.

```cpp
#include <bits/stdc++.h>
#define x first
#define y second
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, ans = 0;
set<pii> s; //(ed, st) : [st, ed)
 
vector<int> vec;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 1; i <= n; i++) {
        int a; cin >> a;
        bool ladj = false, radj = false;
        auto rt = s.upper_bound({a, a});
        if (rt != s.end() && rt->second <= a) {
            cout << ans << " ";
            continue;
        }
        if (rt != s.end() && rt->second == a + 1) radj = true;
        auto lt = s.upper_bound({a, a});
        if (lt != s.begin() && (--lt)->first == a) ladj = true;
 
        if (!ladj && !radj) {
            s.insert({a + 1, a});
            ans++;
        }
        if (!ladj && radj) {
            int t = rt->first;
            if ((rt->first - rt->second) % 2 == 0) ans++;
            s.erase(rt);
            s.insert({t, a});
        }
        if (ladj && !radj) {
            int t = lt->second;
            if ((lt->first - lt->second) % 2 == 0) ans++;
            s.erase(lt);
            s.insert({a + 1, t});
        }
        if (ladj && radj) {
            pii t = {rt->first, lt->second};
            if ((rt->first - rt->second) % 2 == 0 && (lt->first - lt->second) % 2 == 0) ans++;
            s.erase(rt);
            s.erase(lt);
            s.insert(t);
        }
 
        cout << ans << " ";
    }
 
    return 0;
}
```

## 2023f-06-03

왼쪽에 있는 열린 괄호와 오른쪽에 있는 닫힌 괄호를 매칭하되 가중치의 합이 최대가 되도록 매칭하는 문제이다.

동적계획법을 활용하면 $O(N^2)$에 풀 수 있지만 $O(N \log N)$으로 풀려면 그리디를 사용해야 한다.

다음과 같은 그리디 알고리즘이 사용 가능함을 보일 수 있다.

- 가능한 여는 괄호들의 후보 집합 $S$를 정의한다. 즉, 여는 괄호를 만나면 $S$에 추가한다.
- 닫는 괄호가 나오면 $S$들 중 가장 가중치가 큰 여는 괄호와 매칭시킨다. 물론 가중치의 합이 양수일 때만 매칭한다.
- 이미 매칭된 여는 괄호와 다시 재매칭하는 것을 고려하기 위해 매칭이 발생하면 닫는 괄호의 가중치의 음의 값을 갖는 괄호를 $S$에 추가한다.

이 방식이 동작하는 이유는 $i+1$번째 괄호를 사용해야 최적이라면 $i+1$번째 괄호와 새로운 괄호가 매칭되거나, $i+1$번쨰 괄호가 기존의 괄호와 매칭되기 때문이다.

간단하게 우선순위 큐를 활용하면 $O(N \log N)$에 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int n, type[303030];
ll arr[303030];
ll ans = 0;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n;
    for (int i = 1; i <= n; i++) {
        char c; cin >> c;
        if (c == '(') type[i] = 0;
        else type[i] = 1;
    }

    for (int i = 1; i <= n; i++) cin >> arr[i];

    priority_queue<ll> pq;
    for (int i = 1; i <= n; i++) {
        if (type[i] == 0) pq.push(arr[i]);
        else {
            if (pq.size() > 0 && pq.top() + arr[i] > 0) {
                ans += pq.top() + arr[i];
                pq.pop();
                pq.push(-arr[i]);
            }
        }
    }

    cout << ans << "\n";

    return 0;
}
```

## 2023f-06-04 (upsolving)

https://www.acmicpc.net/problem/21095와 같은 문제이자 https://www.acmicpc.net/problem/18448에 쿼리 추가된 문제이다. 일단 쿼리가 없는 버전의 문제를 해결하자.

먼저 매개변수 탐색을 고려할 수 있는데 인접한 수의 합이 $m$ 이하가 되도록 수를 최대 몇 개 고를 수 있는지 구하면 된다. 

이제 인접한 수의 합이 $m$ 이하가 되도록 고르는 문제는 다음 과정으로 풀 수 있다.

- 제일 작은 수가 사용됨은 자명하고 cyclic shift가 있어도 답이 변하지 않으므로 최솟값을 1번으로 옮기는 cyclic shift를 하자. 이러한 변환의 장점은 $w_{i_1} + w_{i_k}$를 고려할 필요가 없다는 것이다.
- $dp[i]$를 $i$번을 고를 때 $i$번까지 고를 수 있는 숫자의 개수로 정의하자. 그러면

$$
dp[i]=\max_{j<i, w_i+w_j \leq m} dp[j]+1
$$

- 위 점화식은 좌표압축을 하고 segtree를 쓰면 $O(\log N)$에 각각의 값을 구할 수 있다. 즉, 시간복잡도는 $O(N \log N)$이다.

곧, 쿼리가 없는 문제의 시간복잡도는 $O(N \log N \log X)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, k, arr[404040];

vector<int> vec;
int N;

struct maxseg {
    int tree[808080];
    void init(int v, int st, int ed) {
        if (st == ed) {
            tree[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }

    void update(int v, int st, int ed, int idx, int val) {
        if (st == idx && ed == idx) {
            tree[v] = max(tree[v], val);
            return;
        }
        if (st > idx || ed < idx) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, idx, val);
        update(2 * v + 1, mid + 1, ed, idx, val);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }

    int get(int v, int st, int ed, int lt, int rt) {
        if (st > rt || ed < lt) return 0;
        if (lt <= st && ed <= rt) return tree[v];
        int mid = (st + ed) / 2;
        return max(get(2 * v, st, mid, lt, rt), get(2 * v + 1, mid + 1, ed, lt, rt));
    }
} seg;

int dp[202020];
int solve(int m) {
    int ret = 1;
    seg.init(1, 0, N - 1);
    dp[1] = 1;
    seg.update(1, 0, N - 1, arr[1], 1);
    for (int i = 2; i <= n; i++) {
        int rt = upper_bound(vec.begin(), vec.end(), m - vec[arr[i]]) - vec.begin();
        dp[i] = seg.get(1, 0, N - 1, 0, rt - 1) + 1;
        seg.update(1, 0, N - 1, arr[i], dp[i]);
        ret = max(ret, dp[i]);
    }
    return ret;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    for (int i = 1; i <= n; i++) vec.push_back(arr[i]);
    sort(vec.begin(), vec.end());
    vec.erase(unique(vec.begin(), vec.end()), vec.end());
    N = vec.size();

    int s;
    for (int i = 1; i <= n; i++) {
        arr[i] = lower_bound(vec.begin(), vec.end(), arr[i]) - vec.begin();
        if (arr[i] == 0) s = i;
    }
    for (int i = 1; i < s; i++) arr[n + i] = arr[i];
    for (int i = 1; i <= n; i++) arr[i] = arr[i + s - 1];

    long long st = 1, ed = 2e9;
    while (st < ed) {
        long long mid = (st + ed) / 2;
        if (solve(mid) >= k) ed = mid;
        else st = mid + 1;
    }

    cout << st << "\n";

    return 0;
}
```

이제 본 문제인 쿼리가 있는 문제를 해결하자. 매개변수 탐색은 쿼리가 있으므로 병렬 이분 탐색을 이용하여 해결해야 한다. 즉, 고정된 $m$에 대하여 인접한 수의 합이 $m$ 이하가 되도록 각 구간에서 고를 수 있는 수의 개수의 최댓값을 빠르게 찾으면 된다.

만약 dp + segtree의 기존 풀이를 사용한다면 pbs를 쓸 수 없을 것이다. 즉, 그리디한 풀이를 찾아야 한다. 이제 그리디하게 인접한 수의 합이 $m$ 이하가 되도록 최대 몇 개 넣을 수 있을지 구하자.

- 일단, $2w_i ≤ m$인 $w_i$는 전부 포함시키는 것이 최적이다.
- 이미 선택된 각 $w_i$들 사이에는 최대 하나만 넣을 수 있다. 둘 이상을 넣으면 같은 구간에 있는 두 수의 합이 $m$보다 커질 것이기 때문이다
- 이제 각 $w_i$들 사이의 구간에서 가장 작은 수를 골라 넣을 수 있는지 확인한다.

$O(N)$에 그리디하게 최대 넣을 수 있는 개수를 구할 수 있으므로 $O(N \log X)$에 해결 가능하다.

이제 이 알고리즘에 pbs를 적용하자. pbs로 문제를 해결하려면 $m$이 증가할 때 segtree에 update를 해 배열을 변경할 수 있고, query를 날려 최대 개수를 구할 수 있으면 된다. 

답에 포함될 수 있는지 여부를 저장하는 세그먼트 트리를 만들고, 각 구간의 경계와 구간별 최솟값의 인덱스를 `std::set`으로 관리하자. 각각에서 query와 update는 다음과 같이 처리하면 된다.

- update: $m → M$
    - 정렬된 배열을 이용해 $m < 2w_i < M$에 해당하는 인덱스를 찾는다.
    - 세그에서 그 인덱스의 값을 1로 바꾸고, 셋에서 양옆 값을 찾고 그 사이 구간에서 최솟값의 인덱스를 찾는다(min seg). 그리고 필요한 $m$의 값과 인덱스를 묶어 힙에 넣는다.
    - 힙에서 필요한 값이 $M$ 이하이면 세그에서 그 인덱스의 값도 1로 바꾼다.
- query: $[l, r]$
    - $[l, r]$에 포함되는 가장 넓은 자명한 원소들로 이루어진 구간을 찾는다.
    - 그 구간에서 seg의 합을 구한다.
    - 바깥 구간은 min seg로 최솟값을 얻어와 가능한 값이 있는지 판단한다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
 
int n, q, arr[101010], l[101010], r[101010], k[101010];
vector<pii> vec;
ll st[101010], ed[101010];
vector<pii> mid;
 
struct minseg {
    pii tree[404040];
    pii merge(pii lt, pii rt) {
        if (lt.first > rt.first) return rt;
        else if (lt.first < rt.first) return lt;
        else {
            return {lt.first, min(lt.second, rt.second)};
        }
    }
    void init(int v, int st, int ed) {
        if (st == ed) {
            tree[v] = {arr[st], st};
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        tree[v] = merge(tree[2 * v], tree[2 * v + 1]);
    }
    pii get(int v, int st, int ed, int lt, int rt) {
        if (st > rt || lt > ed) return {2e9, 0};
        if (lt <= st && ed <= rt) return tree[v];
        int mid = (st + ed) / 2;
        return merge(get(2 * v, st, mid, lt, rt), get(2 * v + 1, mid + 1, ed, lt, rt)); 
    }
} table;
 
struct sumseg {
    int tree[404040];
    void init(int v, int st, int ed) {
        if (st == ed) {
            tree[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        tree[v] = tree[2 * v] + tree[2 * v + 1];
    }
    void update(int v, int st, int ed, int idx, int val) {
        if (st > idx || ed < idx) return;
        if (st == idx && ed == idx) {
            tree[v] = val;
            return;
        }
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, idx, val);
        update(2 * v + 1, mid + 1, ed, idx, val);
        tree[v] = tree[2 * v] + tree[2 * v + 1];
    }
    int get(int v, int st, int ed, int lt, int rt) {
        if (st > rt || lt > ed) return 0;
        if (lt <= st && ed <= rt) return tree[v];
        int mid = (st + ed) / 2;
        return get(2 * v, st, mid, lt, rt) + get(2 * v + 1, mid + 1, ed, lt, rt); 
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> q;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    for (int i = 0; i <= n; i++) vec.push_back({arr[i], i});
    sort(vec.begin() + 1, vec.end());
    table.init(1, 1, n);
    for (int i = 1; i <= q; i++) cin >> l[i] >> r[i] >> k[i];
 
    for (int i = 1; i <= q; i++) st[i] = 0, ed[i] = 2147483647;
    while (st[1] < ed[1]) {
        mid.clear();
        mid.push_back({0, 0});
        for (int i = 1; i <= q; i++) mid.push_back({(st[i] + ed[i]) / 2, i});
        sort(mid.begin() + 1, mid.end());
        seg.init(1, 1, n);
        set<int> s;
        priority_queue<pii> pq;
 
        int p = 1;
        for (int i = 1; i <= q; i++) {
            while (p <= n && vec[p].first * 2 <= mid[i].first) {
                while (!pq.empty()) {
                    if (pq.top().first + mid[i].first >= 0) {
                        seg.update(1, 1, n, pq.top().second, 1);
                        pq.pop();
                    }
                    else break;
                }

                auto it = s.insert(vec[p].second).first;
                auto up = it, dn = it; up++;
                seg.update(1, 1, n, *it, 1);
                if (up != s.end() && *up > *it+1) {
                    int idx = table.get(1, 1, n, *it+1, *up-1).second;
                    pq.push({-max(arr[*it]+arr[idx], arr[*up]+arr[idx]), idx});
                }
                if (it != s.begin() && *it > *(--dn)+1) {
                    int idx = table.get(1, 1, n, *dn+1, *it-1).second;
                    pq.push({-max(arr[*it]+arr[idx], arr[*dn]+arr[idx]), idx});
                }

                while (!pq.empty()) {
                    if (pq.top().first + mid[i].first >= 0) {
                        seg.update(1, 1, n, pq.top().second, 1);
                        pq.pop();
                    }
                    else break;
                }
                
                p++;
            }

            while (!pq.empty()) {
                if (pq.top().first + mid[i].first >= 0) {
                    seg.update(1, 1, n, pq.top().second, 1);
                    pq.pop();
                }
                else break;
            }
 
            int L = l[mid[i].second], R = r[mid[i].second], K = k[mid[i].second];
            int cnt = 0;
            if (s.lower_bound(L) == s.end() || *(s.lower_bound(L)) > R) {
                if (table.get(1, 1, n, L, R).first * 2 <= mid[i].first) cnt = 1;
            }
            else {
                int lt = *(s.lower_bound(L)), rt = *--(s.upper_bound(R));
                cnt = seg.get(1, 1, n, lt, rt);
                int val = min(table.get(1, 1, n, L, lt-1).first, table.get(1, 1, n, rt+1, R).first);
                if (val <= mid[i].first - max(arr[lt], arr[rt])) cnt++;
            }
 
            if (cnt >= K) ed[mid[i].second] = mid[i].first;
            else st[mid[i].second] = mid[i].first + 1;
        }
    }
 
    for (int i = 1; i <= q; i++) cout << st[i] << "\n";
 
    return 0;
}
```

## 2023f-06-05

https://www.acmicpc.net/problem/19270와 같은 문제이다.
