## 1. 경찰서 (0:33:12)

SCC마다 하나 이상의 경찰서가 있어야 한다. SCC를 구하고 각 SCC에서 가장 가중치가 작은 정점에 경찰서를 두면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, m;
vector<int> adj1[101010], adj2[101010];
int w[101010];
 
int ch[101010];
stack<int> stk;
vector< vector<int> > scc;
void dfs1(int v) {
    ch[v] = 1;
    for (int i = 0; i < adj1[v].size(); i++) {
        int u = adj1[v][i];
        if (ch[u]) continue;
        dfs1(u);
    }
    stk.push(v);
}
 
void dfs2(int v) {
    ch[v] = 1;
    scc.back().push_back(v);
    for (int i = 0; i < adj2[v].size(); i++) {
        int u = adj2[v][i];
        if (ch[u]) continue;
        dfs2(u);
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> w[i];
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj1[a].push_back(b);
        adj2[b].push_back(a);
    }
 
    for (int i = 1; i <= n; i++) {
        if (ch[i]) continue;
        dfs1(i);
    }
 
    for (int i = 1; i <= n; i++) ch[i] = 0;
    while (stk.size() > 0) {
        int v = stk.top();
        stk.pop();
        if (ch[v]) continue;
        scc.push_back(vector<int>());
        dfs2(v);
    }
 
    ll ans = 0;
    for (int i = 0; i < scc.size(); i++) {
        // for (int j = 0; j < scc[i].size(); j++) cout << scc[i][j] << " ";
        // cout << "-1\n";
        int mn = 1e9;
        for (int j = 0; j < scc[i].size(); j++) mn = min(mn, w[scc[i][j]]);
        ans += mn;
    }
 
    cout << ans << "\n";
    
    return 0;
}
```

## 2. 프로듀스 IOI (1:15:16)

전형적인 2-SAT 문제이다. 추가로 해야하는 작업 중 하나는 $x_1$이 TRUE인 해의 존재성을 판단하는 것인데, 이는 $x_1$이 포함된 SCC에서 도달할 수 있는 모든 정점 중에 $x_i$와 $!x_i$가 모두 존재하지 않으면 된다. $T \rightarrow T$여야 하기 때문이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int n, m;
vector<int> adj1[202020], adj2[202020];
 
int ch[202020];
stack<int> stk;
vector< vector<int> > scc;
void dfs1(int v) {
    ch[v] = 1;
    for (int i = 0; i < adj1[v].size(); i++) {
        int u = adj1[v][i];
        if (ch[u]) continue;
        dfs1(u);
    }
    stk.push(v);
}
 
void dfs2(int v) {
    ch[v] = 1;
    scc.back().push_back(v);
    for (int i = 0; i < adj2[v].size(); i++) {
        int u = adj2[v][i];
        if (ch[u]) continue;
        dfs2(u);
    }
}
 
bool res;
set<int> S;
void check(int v) {
    ch[v] = 1;
    if (S.find(v%n) == S.end()) S.insert(v%n);
    else res = false;
    for (int i = 0; i < adj1[v].size(); i++) {
        int u = adj1[v][i];
        if (ch[u]) continue;
        check(u);
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    while (scanf("%d %d", &n, &m) == 2) {
        bool ans = true;
        scc.clear();
        while (stk.size() > 0) stk.pop();
 
        for (int i = 1; i <= 2*n; i++) adj1[i].clear();
        for (int i = 1; i <= 2*n; i++) adj2[i].clear();
        
        for (int i = 0; i < m; i++) {
            int a, b, na, nb;
            scanf("%d %d", &a, &b);
            na = -a; nb = -b;
            if (a < 0) a = -a+n;
            if (b < 0) b = -b+n;
            if (na < 0) na = -na+n;
            if (nb < 0) nb = -nb+n;
            adj1[na].push_back(b);
            adj1[nb].push_back(a);
            adj2[b].push_back(na);
            adj2[a].push_back(nb);
        }
 
        for (int i = 1; i <= 2*n; i++) ch[i] = 0;
        for (int i = 1; i <= 2*n; i++) {
            if (ch[i]) continue;
            dfs1(i);
        }
 
        for (int i = 1; i <= 2*n; i++) ch[i] = 0;
        while (stk.size() > 0) {
            int v = stk.top();
            stk.pop();
            if (ch[v]) continue;
            scc.push_back(vector<int>());
            dfs2(v);
        }
 
        for (int i = 0; i < scc.size(); i++) {
            set<int> s;
            for (int j = 0; j < scc[i].size(); j++) {
                if (s.find(scc[i][j]%n) == s.end()) s.insert(scc[i][j]%n);
                else ans = false;
            }
        }
 
        for (int i = 1; i <= 2*n; i++) ch[i] = 0;
        res = true;
        S.clear();
        check(1);
        if (!res) ans = false;
 
        cout << (ans ? "yes\n" : "no\n");
 
    }
    
    return 0;
}
```

## 3. 레드 블랙 트리 (1:27:28)

세 명제 $P, Q, R$ 중 두 개 이상이 참인 것과 $(P \lor Q) \wedge (Q \lor R) \wedge (R \lor P)$ 이 동치이다. 즉, 2-SAT 문제로 환원 가능하다. 따라서 2-SAT 문제를 풀면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int k, n;
vector<int> adj1[10101], adj2[10101];
 
int ch[10101];
stack<int> stk;
vector< vector<int> > scc;
 
int ans[5050];
 
void add(int a, int b) {
    int na = -a, nb = -b;
    if (a < 0) a = -a+k;
    if (b < 0) b = -b+k;
    if (na < 0) na = -na+k;
    if (nb < 0) nb = -nb+k;
    adj1[na].push_back(b);
    adj1[nb].push_back(a);
    adj2[a].push_back(nb);
    adj2[b].push_back(na);
}
 
void dfs1(int v) {
    ch[v] = 1;
    for (int i = 0; i < adj1[v].size(); i++) {
        int u = adj1[v][i];
        if (ch[u]) continue;
        dfs1(u);
    }
    stk.push(v);
}
 
void dfs2(int v) {
    ch[v] = 1;
    scc.back().push_back(v);
    for (int i = 0; i < adj2[v].size(); i++) {
        int u = adj2[v][i];
        if (ch[u]) continue;
        dfs2(u);
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> k >> n;
    for (int i = 0; i < n; i++) {
        int v, u, w;
        char a, b, c;
        cin >> v >> a >> u >> b >> w >> c;
        if (a == 'B') v = -v;
        if (b == 'B') u = -u;
        if (c == 'B') w = -w;
        add(v, u);
        add(u, w);
        add(w, v);
    }
 
    for (int i = 1; i <= 2*k; i++) ch[i] = 0;
    for (int i = 1; i <= 2*k; i++) {
        if (ch[i]) continue;
        dfs1(i);
    }
 
    for (int i = 1; i <= 2*k; i++) ch[i] = 0;
    while (stk.size() > 0) {
        int v = stk.top();
        stk.pop();
        if (ch[v]) continue;
        scc.push_back(vector<int>());
        dfs2(v);
    }
 
    for (int i = 0; i < scc.size(); i++) {
        set<int> s;
        for (int j = 0; j < scc[i].size(); j++) {
            if (s.find(scc[i][j]%k) == s.end()) s.insert(scc[i][j]%k);
            else {
                cout << "-1\n";
                exit(0);
            }
        }
    }
    
    for (int i = 1; i <= k; i++) ans[i] = -1;
    for (int i = 0; i < scc.size(); i++) {
        for (int j = 0; j < scc[i].size(); j++) {
            int v = scc[i][j];
            if (v > k) v -= k;
            if (ans[v] == -1) {
                if (scc[i][j] <= k) ans[v] = 0;
                else ans[v] = 1;
            }
        }
    }
 
    for (int i = 1; i <= k; i++) cout << (ans[i] == 1 ? 'R' : 'B');
    
    return 0;
}
```

## 5. 절도 (2:42:09)

https://www.acmicpc.net/problem/4013

SCC로 주어진 그래프를 DAG로 압축할 수 있고, DAG를 만들면 위상정렬 순서대로 동적계획법을 활용해 각 SCC에 도달했을 때 얻을 수 있는 최대 돈의 양을 구할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
int k, n, m;
vector<int> adj1[505050], adj2[505050], adj[505050];
 
int ch[505050], w[505050], sum[505050], st, ans[505050];
stack<int> stk;
vector< vector<int> > scc;
int num[505050];
int p;
vector<int> ed;
void dfs1(int v) {
    ch[v] = 1;
    for (int i = 0; i < adj1[v].size(); i++) {
        int u = adj1[v][i];
        if (ch[u]) continue;
        dfs1(u);
    }
    stk.push(v);
}
 
void dfs2(int v) {
    ch[v] = 1;
    scc.back().push_back(v);
    for (int i = 0; i < adj2[v].size(); i++) {
        int u = adj2[v][i];
        if (ch[u]) continue;
        dfs2(u);
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj1[a].push_back(b);
        adj2[b].push_back(a);
    }
    for (int i = 1; i <= n; i++) cin >> w[i];
    cin >> st >> p;
    ed.resize(p);
    for (int i = 0; i < p; i++) cin >> ed[i];
 
    for (int i = 1; i <= n; i++) ch[i] = 0;
    for (int i = 1; i <= n; i++) {
        if (ch[i]) continue;
        dfs1(i);
    }
 
    for (int i = 1; i <= n; i++) ch[i] = 0;
    while (stk.size() > 0) {
        int v = stk.top();
        stk.pop();
        if (ch[v]) continue;
        scc.push_back(vector<int>());
        dfs2(v);
    }
 
    for (int i = 0; i < scc.size(); i++) {
        for (int j = 0; j < scc[i].size(); j++) {
            num[scc[i][j]] = i+1;
            sum[i+1] += w[scc[i][j]];
        }
    }
    st = num[st];
 
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < adj1[i].size(); j++) {
            if (num[i] == num[adj1[i][j]]) continue;
            adj[num[i]].push_back(num[adj1[i][j]]);
        }
    }
 
    for (int i = 1; i <= scc.size(); i++) {
        sort(adj[i].begin(), adj[i].end());
        adj[i].erase(unique(adj[i].begin(), adj[i].end()), adj[i].end());
    }
 
    for (int i = 1; i <= scc.size(); i++) ans[i] = -1;
    ans[st] = sum[st];
    for (int i = 1; i <= scc.size(); i++) {
        if (ans[i] == -1) continue;
        for (int j = 0; j < adj[i].size(); j++) {
            ans[adj[i][j]] = max(ans[adj[i][j]], ans[i] + sum[adj[i][j]]);
        }
    }
 
    int res = 0;
    for (int i = 0; i < p; i++) {
        res = max(res, ans[num[ed[i]]]);
    }
 
    cout << res << "\n";
    
    return 0;
}
```

## 4. 파괴 (2:53:04)

https://www.acmicpc.net/problem/1867

이분 그래프의 최소 버텍스 커버는 최대 매칭과 같다는 정리를 활용하자. $A_x$와 $B_y$ 정점을 잇는 간선이 존재하는 $N$개씩의 정점이 양쪽에 있는 이분 그래프에서 답은 최소 버텍스 커버이므로 최대 매칭을 구하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n, k, ans = 0;
vector<int> adj[550];
int lt[550], rt[550], ch[550];
 
bool solve(int v) {
    ch[v] = 1;
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        if (rt[u] == 0 || (ch[rt[u]] == 0 && solve(rt[u]))) {
            lt[v] = u;
            rt[u] = v;
            return true;
        }
    }
    return false;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> k;
    for (int i = 0; i < k; i++) {
        int x, y;
        cin >> x >> y;
        adj[x].push_back(y);
    }
 
    for (int i = 1; i <= n; i++) {
        if (lt[i] == 0) {
            for (int i = 1; i <= n; i++) ch[i] = 0;
            if (solve(i)) ans++;
        }
    }
 
    cout << ans << "\n";
    
    return 0;
}
```

## 7. 똑떨

https://www.acmicpc.net/problem/16583

주어진 그래프가 트리라면 리프부터 간선을 매칭하여 그리디하게 간선을 분리할 수 있다. 이때 간선의 개수가 $M$개이면 $\lfloor \frac{M}{2} \rfloor$개의 매칭을 만들 수 있다.

이제 주어진 그래프의 DFS 스패닝 트리를 그리자. 그러면 스패닝 트리에 속하지 않는 추가로 있는 간선 $(u, v)$에서 루트와 더 가까운 정점 $u$ 대신에 더미 정점과 연결되어 있다고 생각하자. 이 방법이 정당한 이유는 트리를 만들면 최대 개수의 매칭이 가능하고, 정점 번호보다 간선 자체가 매칭되는 것이기 때문이다.

## 9. 도로 경주

https://www.acmicpc.net/problem/7907

$u$번 교차로와 $v$번 교차로를 동시에 방문하는 경로가 존재한다는 것은 임의의 위상정렬에 대해 $u$와 $v$의 선후관계가 변경되지 않는다는 뜻이다. 만약 위상 관계가 명확하지 않다면 차수가 $0$인 정점들을 큐에 추가하는 순서에 따라 선후관계가 바뀔 것이다.

그런데 이 문제의 경우 그래프의 구조가 축에 평행한 직사각형 모양임을 생각하자. 즉, 차수가 $0$인 정점들은 $x$축 방향으로 이동하는 간선에 의한 것과 $y$축 방향으로 이동하는 간선에 의한 것뿐이다. 그러면 직관적으로 생각해보았을 때 임의의 정점이 가장 앞에 올 수 있는 위상정렬과 가장 뒤에 올수있는 위상정렬은 매번 $x$축 방향으로 이동하는 간선을 우선하는 것과 매번 $y$축 방향으로 이동하는 간선을 우선하는 것이 된다.
