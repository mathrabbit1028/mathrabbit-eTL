## 2. 복붙 (0:39:42)

$f(A_i)$가 $i$번째와 접두사가 겹치는 실패함수 위치(KMP의 그 값) 한 칸 전이 $j$이면 $f(A_j) + (i-j)$로 계산되므로 동적계획법과 같이 쉽게 $O(N)$에 처리할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
string s;
int n, overlap[1010101];
int res[1010101];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    cin >> s;
    
    int idx = 0;
    for (int i = 1; i < s.length(); i++) {
        while (idx > 0 && s[i] != s[idx]) idx = overlap[idx - 1];
        if (s[i] == s[idx]) {
            idx++;
            overlap[i] = idx;
        }
    }
 
    for (int i = 0; i < s.length(); i++) {
        if (overlap[i] == 0) res[i] = 0;
        else res[i] = res[overlap[i] - 1] + i - overlap[i] + 1;
    }
 
    long long ans = 0;
    for (int i = 0; i < s.length(); i++) ans += res[i];
    cout << ans << "\n";
 
    return 0;
}
```

## 4. 비밀번호 (1:05:25)

매내처 활용하는 전형적인 문제

```cpp
#include <bits/stdc++.h>
using namespace std;
string s;
int n, dp[1010101];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> s;
    n = s.length();
    for (int i = 0; i <= n; i++) s.push_back(0);
    for (int i = n-1; i >= 0; i--) s[2*i+1] = s[i];
    for (int i = 0; i <= n; i++) s[2*i] = ':';
    n = s.length();
 
    int last = 1, idx = 0;
    dp[0] = 0;
    for (int i = 1; i < n; i++) {
        int j = -1;
        if (last >= i) {
            if (dp[2*idx-i] >= last-i) j = last-i;
            else dp[i] = dp[2*idx-i];
        }
        else j = 0;
 
        if (j >= 0) {
            while (true) {
                if (i-j < 0 || i+j >= n) break;
                if (s[i-j] != s[i+j]) break;
                j++;
            }
            dp[i] = --j;
        }
 
        if (last < i+dp[i]) {
            last = i+dp[i];
            idx = i;
        }
    }
 
    int ans = 0;
    for (int i = 1; i < n; i++) {
        if (dp[ans] < dp[i]) ans = i;
    }
 
    for (int i = ans-dp[ans]; i <= ans+dp[ans]; i++) if (s[i] != ':') cout << s[i];
 
    return 0;
}
```

## 5. 테트리스 (4:16:02)

https://www.acmicpc.net/problem/1605

두 번 나오는 부분 문자열은 **접미사 배열을 만들었을 때 인접한 접미사의 최대 공통 접두사들 중 하나**이므로 suffix array와 lcp array를 만들면 문제를 해결할 수 있다.

suffix array는 $O(N \log^2N)$에 다음의 관찰을 기반으로 구할 수 있다.

- 모든 접미사의 $k$번째 문자열까지를 기준으로 정렬되어 있다고 가정하자.
- 그러면 ($i$번쨰 접미사의 결정된 순서, $i+k$번째 접미사의 결정된 순서)를 쌍으로 해 정렬하면 $i+k$번째 접미사의 결정된 순서에 추가 $k$자리의 순서가 나와 있으므로 전체 $2k$자리를 정렬된 상태로 둘 수 있다.

즉 $k$를 1부터 증가시키면서 $n$보다 커질때까지 하면 된다.

* 길이에 대한 이분탐색 + 해싱으로 해결할 수 있음

```cpp
#include <bits/stdc++.h>
using namespace std;
const int r = 26;

string s;
int n;
int sfx[202020], gup[404040], rev[202020], lcp[202020];
vector<int> res, num;
vector<int> cnt[202020];

int step;
bool compare(int i, int j) {
    if (gup[i] == gup[j]) {
        return gup[i+step] < gup[j+step];
    }
    return gup[i] < gup[j];
}

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);

    cin >> n;
    cin >> s;

    for (int i = 0; i < n; i++) sfx[i] = i;
    
    int p = 0;
    vector<int> first[26];
    for (int i = 0; i < n; i++) first[s[sfx[i]]-'a'].push_back(i);
    for (int k = 0; k < 26; k++) {
        if (first[k].size() > 0) p++;
        for (int j = 0; j < (int)first[k].size(); j++) {
            res.push_back(sfx[first[k][j]]);
            num.push_back(p);
        }
    }

    for (int i = 0; i < n; i++) sfx[i] = res[i];
    for (int i = 0; i < n; i++) gup[sfx[i]] = num[i];
    for (int i = n; i < 2*n; i++) gup[i] = -1;

    step = 1;
    while (step <= n) {
        p = 1;
        num.clear();
        sort(sfx, sfx + n, compare);

        for (int i = 0; i < n; i++) {
            if (i > 0 && compare(sfx[i-1], sfx[i])) p++;
            num.push_back(p);
        }
        for (int i = 0; i < n; i++) gup[sfx[i]] = num[i];
        step *= 2;
    }

    // for (int i = 0; i < n; i++) {
    //     for (int j = sfx[i]; j < n; j++) {
    //         cout << s[j];
    //     }
    //     cout << "\n";
    // }

    for (int i = 0; i < n; i++) rev[sfx[i]] = i;
    int k = 0;
    for (int i = 0; i < n; i++) {
        if (rev[i] == 0) continue;
        int j = sfx[rev[i]-1];
        while (i+k < n && j+k < n) {
            if (s[i+k] != s[j+k]) break;
            k++;
        }
        lcp[rev[i]] = k;
        if (k > 0) k--;
    }

    int ans = 0;
    for (int i = 1; i < n; i++) ans = max(ans, lcp[i]);
    cout << ans << "\n";

	return 0;
}
```

## 7. 선발고사

주어진 문자들로 트라이를 만들어서 트라이를 모두 탐색하되 가장 깊이가 깊은 정점을 가장 마지막에 탐색하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, cnt = 0;
string s[40404];
vector<char> ans;

struct Node {
    int dep;
    int cnt;
    Node* child[26];
    Node() {
        for (int i = 0; i < 26; i++) child[i] = NULL;
        dep = 0;
        cnt = 0;
    }
};
Node* head = new Node();

void find_dep(Node* node) {
    for (int i = 0; i < 26; i++) {
        if (node->child[i] != NULL) {
            find_dep(node->child[i]);
            node->dep = max(node->dep, node->child[i]->dep);
        }
    }
    node->dep += 1;
}

void DFS(Node* node) {
    vector< pair<int, int> > vec;
    for (int i = 0; i < 26; i++) {
        if (node->child[i] != NULL) vec.push_back({node->child[i]->dep, i});
    }
    sort(vec.begin(), vec.end());

    cnt += node->cnt;
    for (int i = 0; i < node->cnt; i++) ans.push_back('P');
    if (cnt == n) {
        cout << ans.size() << "\n";
        for (int i = 0; i < ans.size(); i++) cout << ans[i] << "\n";
        exit(0);
    }

    if (vec.size() == 0) {
        return;
    }

    for (int j = 0; j < vec.size(); j++) {
        int i = vec[j].second;
        ans.push_back('a'+i);
        DFS(node->child[i]);
        ans.push_back('-');
    }
}

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);

    cin >> n;
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < n; i++) {
        int p = 0;
        Node* now = head;
        while (p < s[i].length()) {
            if (now->child[s[i][p]-'a'] == NULL) {
                now->child[s[i][p]-'a'] = new Node();
            }
            now = now->child[s[i][p]-'a'];
            p++;
        }
        now->cnt++;
    }

    find_dep(head);

    DFS(head);

    assert(false);

	return 0;
}
```

## 6. 부계

문자열의 접두사이면서 접미사인 부분문자열을 모두 $O(N)$에 구해야 한다. KMP의 실패함수를 쓰거나 Z 알고리즘을 사용하면 된다.

이제 $a_i$를 $i$번을 마지막으로 하는 부계 체인의 최대 길이라고 하자. 그러면 $a_i$는 $i$의 접두사이면서 접미사인 문자열을 모두 찾아서 그 문자열의 dp값을 가져오면 된다. 이 과정에서 map을 사용하면 쉽게 dp 값을 저장하고 가져올 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n;
string s[2020202];

vector<int> overlap;
void get_overlap(string str) {
    overlap = vector<int>(str.length());

    int idx = 0;
    for (int i = 1; i < str.length(); i++) {
        while (idx > 0 && str[i] != str[idx]) idx = overlap[idx - 1];
        if (str[i] == str[idx]) {
            idx++;
            overlap[i] = idx;
        }
    }
}

unordered_map<string, int> dp;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);

    cin >> n;
    for (int i = 0; i < n; i++) cin >> s[i];

    for (int i = 0; i < n; i++) {
        get_overlap(s[i]);
        string t = s[i];
        int p = s[i].length(), val = 0;
        while (p > 0) {
            while (t.length() > p) t.pop_back();
            if (dp.find(t) != dp.end()) val = max(val, dp[t]);
            p = overlap[p - 1];
        }
        dp[s[i]] = val + 1;
    }

    int ans = 0;
    for (auto iter = dp.begin(); iter != dp.end(); iter++) {
        ans = max(ans, iter->second);
    }
    cout << ans << "\n";

	return 0;
}
```

## 3. 응징

https://www.acmicpc.net/problem/9250

아호코라식을 사용하면 된다.

```cpp
#include <bits/stdc++.h>

    Node* child[26];
    Node* par;
    Node* fail;
    bool output;

        for (int i = 0; i < 26; i++) child[i] = NULL;
        par = NULL;
        fail = NULL;

        cnt = 0;
    }

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(0);
 
    head = new Node();
    head->fail = head;

    cin >> n;
    for (int i = 0; i < n; i++) cin >> s[i];
    for (int i = 0; i < n; i++) {
        int p = 0;
        Node* now = head;
        while (p < s[i].length()) {
            if (now->child[s[i][p]-'a'] == NULL) {
                now->child[s[i][p]-'a'] = new Node();
                now->child[s[i][p]-'a']->par = now;
            }
            now = now->child[s[i][p]-'a'];
            p++;
        }
        now->cnt++;
        now->output = true;
    }

    queue<Node*> q;
    q.push(head);

    while (!q.empty()) {
        Node* now = q.front();
        q.pop();
        for (int i = 0; i < 26; i++) {
            if (now->child[i] == NULL) continue;
            Node* next = now->child[i];
            Node* node = now;
            q.push(next);
            while (node->par != NULL) {
                if (node->fail->child[i] != NULL) {
                    next->fail = node->fail->child[i];
                    break;
                }
                node = node->fail;
            }
            if (node->par == NULL) {
                next->fail = head;
            }
            if (next->fail->output) next->output = true;
        }
    }
    
    cin >> m;

    while (m--) {
        string t;
        bool ans = false;
        Node* now = head;
        cin >> t;
        for (int i = 0; i < t.length(); i++) {
            while (now->par != NULL && now->child[t[i]-'a'] == NULL) now = now->fail;
            if (now->child[t[i]-'a'] != NULL) {
                now = now->child[t[i]-'a'];
            }
            if (now->output) {
                ans = true;
                break;
            }
        }
        if (ans) cout << "YES\n";
        else cout << "NO\n";
    }
 
	return 0;
}
```

## 8. RNA

https://www.acmicpc.net/problem/16780

- 서브테스크 1: $O(NM|S_i|)$ naive 풀이로 통과할 수 있다.
- 서브테스크 2
    - $N$개의 문자열을 trie로 관리하자. 그러면 각 쿼리에서 $O(|P|)$의 시간에 접두사 중 $P$가 있는 서브트리를 골라낼 수 있다.
    - $N$개의 문자열을 뒤집은 문자열을 trie로 관리하자. 그러면 각 쿼리에서 $O(|Q|)$의 시간에 접두사 중 $Q$를 뒤집은 문자열이 있는 서브트리를 골라낼 수 있다.
    - 오일러 투어 테크닉을 사용하면 접두사 중 $P$가 있는 서브트리를 구간으로 나타낼 수 있다. 같은 방법으로 접미사 중 $Q$가 있는 서브트리를 구간으로 나타낼 수 있다.
    - 이제 각 ett의 인덱스마다 $N$개의 문자열의 번호 목록을 저장하자. 그러면 접두사 중 $P$가 있는 $O(N)$개의 문자열의 번호와 접미사 중 $Q$가 있는 $O(N)$개의 문자열의 번호를 $O(N)$에 알아낼 수 있다.
    - 이렇게 알아낸 두 집합에 모두 속하는 점의 개수를 구하면 된다. 시간복잡도는 $O(\sum |S_i| + \sum |P_i| + \sum |Q_i| + MN)$이다.
- 서브테스크 4
    - 오일러 투어에서 인덱스를 이용하면 각 문자열을 $(x, y)$의 점 하나로 나타낼 수 있다.
    - 이제 각 쿼리는 영역 $[a, b] \times [c, d]$에 있는 점의 개수를 구하는 문제로 환원된다.
    - pst나 머지 소트 트리를 사용하면 해결할 수 있다.

```cpp
        p = dict[i].length()-1;

            if (now->child[dict[i][p]-'A'] == NULL) {
                now->child[dict[i][p]-'A'] = new Node();
                now->child[dict[i][p]-'A']->par = now;
            }
            now = now->child[dict[i][p]-'A'];
            p--;
        }
        now->cnt++;
        now->num.push_back(i);
    }

    r = -1;
    get_ett(root);
    r = -1;
    get_ett(rev_root, true);

    sort(pts, pts+n);

    for (int i = 0; i < m; i++) qrys[i].idx = i;
    for (int i = 0; i < m; i++) {
        Node* pfx_ptr = find_node(root, pfx[i]);
        Node* sfx_ptr = find_node(rev_root, sfx[i]);
        if (pfx_ptr == NULL || sfx_ptr == NULL) qrys[i].a = -1;
        else {
            qrys[i].a = pfx_ptr->in;
            qrys[i].b = pfx_ptr->out;
            qrys[i].c = sfx_ptr->in;
            qrys[i].d = sfx_ptr->out;
        }
    }

    pstRoot[0] = new pstNode();
    seg.init(pstRoot[0], 0, r);
    for (int i = 0; i < n; i++) root_ptr[i] = NULL;
    for (int i = 0; i < n; i++) {
        pstRoot[i+1] = new pstNode();
        seg.update(pstRoot[i], pstRoot[i+1], 0, r, pts[i].y, 1);
        root_ptr[pts[i].x] = pstRoot[i+1];
    }
    for (int i = 0; i < 2000000; i++) {
        if (root_ptr[i] == NULL) {
            if (i > 0) root_ptr[i] = root_ptr[i-1];
            else root_ptr[i] = pstRoot[0];
        }
    }

    for (int i = 0; i < m; i++) {
        int ans = 0;
        if (qrys[i].a >= 0)
            ans = seg.get(root_ptr[qrys[i].b], 0, r, qrys[i].c, qrys[i].d)
                - (qrys[i].a > 0 ? seg.get(root_ptr[qrys[i].a-1], 0, r, qrys[i].c, qrys[i].d) : 0);
        cout << ans << "\n";
    }

    return 0;
}
```
