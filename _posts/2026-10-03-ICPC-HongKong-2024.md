## 문제 K

주어진 문자열이 `R`로 시작하는데, 쿼리 문자열이 `L`로 시작하는 경우 불가능하다. 유사하게, 주어진 문자열이 `L`로 끝나는데, 쿼리 문자열이 `R`로 끝나는 경우도 불가능하다. 따라서, 주어진 문자열의 시작과 끝을 확인하고, 쿼리 문자열의 시작과 끝을 확인하여 불가능한 경우를 걸러낼 수 있다.

두 경우 모두 아니라면, 쿼리 문자열이 주어진 문자열의 subsequence인 경우 가능하다. 이것은 주어진 문자열의 `L`과 `R`의 위치를 증가하는 순으로 기록하면 $O(T \log S)$ 시간에 확인할 수 있다. 여기서 $T$는 쿼리 문자열의 길이, $S$는 주어진 문자열의 길이이다.

## 문제 G

트라이를 만들고, 각 정점 $v$를 지나가는 문자열의 수 $C(v)$를 저장하자. 그러면 $f(i, j) - f(i-1, j)$는, $i$번째 문자열을 추가하고 나서 $i$번째 문자열이 지나가는 경로 상의 정점 $v$들에 대해 $j \mid C(v)$인 정점의 수이다. 왜냐하면, $j \mid C(v)$라면 원래 하나의 그룹에서 어떤 문자열을 새로 추가된 문자열로 바꾸어 그룹의 the longest common prefix의 길이를 1 늘릴 수 있기 때문이다.

따라서, 각 정점마다 업데이트 된 $C(v)$를 계산하고, $C(v)$들의 약수 $d$에 대해 $f(i, d)$를 1 증가시키면서, 문제에서 구하라고 한 값을 업데이트한다. 약수는 $O(N \log N)$에 전처리할 수 있고, $d$ 하나당 $f(i, d)$와 답을 업데이트하는 데 걸리는 시간은 $O(1)$이기 때문에 시간 안에 가능하다.

## 문제 L

전부 `W`로 만드는 것이 가능한지 판단하겠다. 중요한 것은 좌하단 꼭짓점이 `B`라면 무조건 통과해야 하는 경로가 하나 정해진다는 것이다. 그리고 그 다음부터는 좌하단 꼭짓점을 통과할 필요가 없다. 유사하게, 그 다음 대각선을 생각하자. 대각선의 각 점들 중에서 뒤집어야 하는 점들만 통과하는 경로는 항상 존재하고, 그리고 그 경로는 대각선을 기준으로 좌하단 영역을 절대 지나지 않는다. 즉, 이것을 반복적으로 업데이트하는 방식으로 해결할 수 있다.

참고로, 대각선이 제일 윗 가로줄에서 시작하더라도 그 부분은 빼고 생각해야 한다. 어차피 지나가야 하는 부분이기 때문이다. 최종적으로 가능한지 판정할 때는, 우상단 꼭짓점 하나만 남았을 때이다. 우상단 꼭짓점이 `B`라면 우상단 꼭짓점을 뒤집고 나서(아니라면 아무 것도 하지 않고) 결과가 전부 `W`가 되었는지 확인하면 된다.

경로의 개수는 최대 $N+M$개이므로 문제가 해결되었다.

```cpp
#include <bits/stdc++.h>

using namespace std;

int T;
int n, m;
int grid[220][220], _grid[220][220];

void _goto(int &r, int &c, int i, int j, string &s) {
    assert(r <= i && c <= j);
    while (c < j) {
        s += 'R';
        c++;
        _grid[r][c] = 1 - _grid[r][c];
    }
    while (r < i) {
        s += 'D';
        r++;
        _grid[r][c] = 1 - _grid[r][c];
    }
}

bool solve(void) {
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            _grid[i][j] = grid[i][j];
        }
    }
    
    // 1-n <= j-i <= m-1
    vector<string> ans;
    for (int k = 1-n; k <= m-1; k++) {
        string s = "";
        int r = 1, c = 1;
        _grid[r][c] = 1 - _grid[r][c];
        for (int i = 2; i <= n; i++) {
            int j = i + k;
            if (j < 1 || j > m) continue;
            if (_grid[i][j] == 1) {
                _goto(r, c, i, j, s);
            } else {
                _goto(r, c, i-1, j, s);
            }
        }
        _goto(r, c, n, m, s);
        ans.push_back(s);
        // cout << s << endl;
    }
    if (_grid[1][m] == 1) {
        string s = "";
        int r = 1, c = 1;
        _grid[r][c] = 1 - _grid[r][c];
        _goto(r, c, 1, m, s);
        _goto(r, c, n, m, s);
        ans.push_back(s);
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (_grid[i][j] == 1) return false;
        }
    }

    cout << "YES" << endl;
    cout << ans.size() << endl;
    for (string str : ans) cout << str << endl;
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> T;

    while (T--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                char c;
                cin >> c;
                if (c == 'W') grid[i][j] = 0;
                else grid[i][j] = 1;
            }
        }

        // target 0, i.e. 'W'
        if (solve()) continue;

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                grid[i][j] = 1 - grid[i][j];
            }
        }
        // target 1, i.e. 'B'
        if (solve()) continue;

        cout << "NO" << endl;
    }

    return 0;

}
```

## 문제 B

dp 상태를 정의하자. 생각해보면 방패가 부서진 전사들의 목록만 안다면, 전사들의 체력 중 최댓값만이 중요하다. 곧, dp 상태를 $dp[atk][hth]$로 정의하자:
- `atk` : 지금까지 누적된 공격력
- `hth` : 방패가 부서진 전사들의 체력 중 최댓값

그러면 전이식은:

$$dp[atk + deal][hth'] \leftarrow dp[atk][hth] + c[deal]$$

여기서 $hth'$는 $hth - deal$과 `atk + deal` 공격력으로 방패가 부서진 전사들의 체력 중 최댓값이다.

dp 상태의 개수는 많아야 $O(m^2)$개 정도인데, 실제로는 그렇게 많지 않아서 unordered_map을 이용하여 sparse하게 dp를 구현하면 된다. 또한, dp 상태를 계산할 때마다, 그 상태가 최적해를 이루는지 확인하고, 최적해를 이루는 경우의 수를 세어주면 된다. 

```cpp
#include <bits/stdc++.h>

using namespace std;

int T, n, m, k;
unordered_map<int, long long> dp[20202]; // acculated attack, highest health of unprotected wariers.

const long long MOD = 998'244'353;
unordered_map<int, long long> cnt[20202];

int a[505050], b[505050], _end;
vector<int> warriors[10101];
long long c[110];

long long solve(int atk, int hth) {
    if (atk >= _end && hth <= 0) return 0;
    if (dp[atk].count(hth)) return dp[atk][hth];

    long long res = 1e18;
    int max_hth = 0;
    for (int deal = 1; deal <= k; deal++) {
        if (atk + deal <= _end)
            max_hth = max(max_hth, warriors[atk + deal].empty() ? 0 : warriors[atk + deal][0]);
        res = min(res, 
            c[deal] + solve(atk + deal, max(max_hth, hth - deal))
        );
    }
    return dp[atk][hth] = res;
}

long long count(int atk, int hth) {
    if (atk >= _end && hth <= 0) return 1;
    if (cnt[atk].count(hth)) return cnt[atk][hth];

    long long res = 0;
    int max_hth = 0;
    for (int deal = 1; deal <= k; deal++) {
        if (atk + deal <= _end)
            max_hth = max(max_hth, warriors[atk + deal].empty() ? 0 : warriors[atk + deal][0]);
        if (dp[atk][hth] == c[deal] + solve(atk + deal, max(max_hth, hth - deal))) {
            res += count(atk + deal, max(max_hth, hth - deal));
            res %= MOD;
        }
    }
    return cnt[atk][hth] = res;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> T;
    while (T--) {
        cin >> n >> m;
        for (int i = 0; i <= 2*(m+k); i++) dp[i].clear();
        for (int i = 0; i <= 2*(m+k); i++) cnt[i].clear();
        for (int i = 1; i <= m; i++) warriors[i].clear();

        for (int i = 1; i <= n; i++) cin >> a[i];
        for (int i = 1; i <= n; i++) cin >> b[i];
        for (int i = 1; i <= n; i++) warriors[b[i]].push_back(a[i]);
        for (int i = 1; i <= m; i++) {
            sort(warriors[i].begin(), warriors[i].end(), greater<int>());
            if (!warriors[i].empty()) _end = i;
        }
        
        cin >> k;
        for (int i = 1; i <= k; i++) cin >> c[i];

        cout << solve(0, 0) << " " << count(0, 0) << "\n";
    }

}
```

## 문제 F

어떤 위치 $x+y \rightarrow x+y-1 \rightarrow \cdots \rightarrow x$로 돈을 전달한다고 하자. 그 경우에 $x$가 갖는 돈은:

$$L(x, y) = \left\lfloor \sum_{i=0}^y \frac{a_{x+i}}{2^i} \right\rfloor$$

로 주어진다. 중요한 것은 2진법 덧셈처럼 생각하면 집합 $\{L(x, x), L(x, x+1), \ldots \}$의 크기가 $O(\log a)$라는 것이다. 따라서, 이 집합을 $L(x, y-1) < L(x, y)$인 $y$들과 $L(x, y)$의 값을 묶어 $O(\log a)$개의 상태 $S(x)$로 표현할 수 있다. 그러면, $S(x)$ 사이에 다음과 같은 전이식을 정의할 수 있다:

$$S(x) = \text{reduce}\left(\left\{\left(y, a_x + \left\lfloor{L(x+1, y+1) \over 2 }\right\rfloor\right) \mid (y, L(x+1, y)) \in S(x+1)\right\} \cup \{(x, a_{x})\}\right)$$

여기서 $\text{reduce}$는 $L(x, y)$가 증가하는 순서대로 정렬하고, $L(x, y)$가 같은 경우에는 $y$가 작은 것을 남기는 연산이다. 참고로, 원형이므로 $x = 2n$부터 시작해야 한다.

거꾸로, 

$$R(x, y) = \left\lfloor \sum_{i=0}^y \frac{a_{x-i}}{2^i} \right\rfloor$$

도 구할 수 있으므로, 이 다음에는 투 포인터로 인덱스가 감소하는 방향으로 돈을 보내는 사람과 인덱스가 증가하는 방향으로 돈을 보내는 사람의 경계를 찾으면 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>

using namespace std;

int T;
int n;
int previous_n = 0;
vector<long long> arr;
using State = pair<int, int>;
vector<State> L[1010101], R[1010101];

void sort_and_unique(vector<State>& states) {
    stable_sort(states.begin(), states.end(), [](const State& lhs, const State& rhs) {
        return lhs.first < rhs.first;
    });

    int size = 0;
    for (const State& state : states) {
        if (size > 0 && states[size - 1].first == state.first) {
            states[size - 1].second = state.second;
        } else {
            states[size++] = state;
        }
    }
    states.resize(size);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> T;

    while (T--) {
        cin >> n;
        arr.resize(n);
        int limit = 2 * max(n, previous_n);
        for (int i = 0; i <= limit; i++) {
            vector<State>().swap(L[i]);
            vector<State>().swap(R[i]);
        }
        previous_n = n;
        for (int i = 0; i < n; i++) cin >> arr[i];

        for (int i = 2*n-1; i >= 0; i--) {
            int j = i % n;
            for (auto it = L[i+1].rbegin(); it != L[i+1].rend(); it++) {
                int val = it->first, idx = it->second;
                int new_val = val/2 + arr[j];
                if (idx < i + n) L[i].emplace_back(new_val, idx);
            }
            L[i].emplace_back(arr[j], i);
            sort_and_unique(L[i]);
            if (i + 1 >= n) vector<State>().swap(L[i + 1]);
        }

        for (int i = 1; i <= 2*n; i++) {
            int j = i % n;
            for (auto it = R[i-1].rbegin(); it != R[i-1].rend(); it++) {
                int val = it->first, idx = it->second;
                int new_val = val/2 + arr[j];
                if (idx > i - n) R[i].emplace_back(new_val, idx);
            }
            R[i].emplace_back(arr[j], i);
            sort_and_unique(R[i]);
            if (i <= n) vector<State>().swap(R[i - 1]);
        }

        for (int i = 0; i < n; i++) {
            vector<State>& left = L[i];
            vector<State>& right = R[i+n];

            long long ans = 0;
            auto it = right.begin();
            for (auto it2 = left.rbegin(); it2 != left.rend(); it2++) {
                int val = it2->first;
                int idx = it2->second;
                // idx -> idx-1 -> ... -> i
                // idx%n+1 -> idx%n+2 -> ... -> i+n
                int lower = idx%n + 1;
                if (lower <= i) lower += n;
                while (it != right.end() && it->second >= lower) {
                    ans = max(ans, (long long)it->first + val);
                    it++;
                }
            }

            cout << ans - arr[i] << " ";
        }
        cout << endl;
    }

    return 0;

}
```

## 문제 A

$[l, r]$이 $k$-symmatric인지 미리 전처리하자. Bitset을 활용해 계산할 수 있다면, $n^2 / 64$가 충분히 작기 때문에 전처리가 가능하다. 길이 $n$의 이진수 $S_l$의 $r$번째 비트가 $[l, r]$이 $k$-symmatric인지 여부를 나타내도록 정의하자. 편의를 위해 $l > r$에서는 항상 비트값이 1이다. 그러면,

$$S_n = 11\cdots1_{(2)}, \quad S_l[r] \leftarrow S_{l+1}[r-1] \land [|A_l - A_r| \leq k]$$

이고, bitset으로 표현하면

$$S_l = (S_{l+1} \cdot 2 + 1) \land \left(\sum_{r=0}^{n-1} [|A_l - A_r| \leq k] \cdot 2^r\right)$$

이다. 여기서 중요한 점은 $A_l$의 값이 $k$ 이하로 작다는 것이다. 따라서, $\sum_{r=0}^{n-1} [|x - A_r| \leq k] \cdot 2^r$를 모든 $x = 1, 2, \ldots, k$에 대해 전처리할 수 있다.

두 번째 문제는 메모리 제한이다. 모든 값을 저장할 수는 없기 때문에, $B$개마다 $S_l$을 저장하고, 임의의 $l$에 대해 $[l, r]$이 $k$-symmatric인지 여부를 확인할 때는 $l$ 이상의 가장 작은 저장된 문자열 $S_{l'}$를 찾아서, $S_{l'}[r - (l' - l)]$를 확인하고, $l' - l$만큼 추가로 비교해야 할 쌍을 비교하면 된다.

전처리가 끝나면, 이분탐색으로 모든 중심에 대해 최대 길이를 찾자. 시간복잡도는 $O(nk + n^2 / 64 + B \cdot n \log n)$이고, 공간복잡도는 $O(n^2 / 64B)$이다.

```cpp
#include <bits/stdc++.h>

using namespace std;

const int B = 32;

int n, k;

vector<int> arr;
bitset<202020> is_diff[1010];
vector<bitset<202020>> is_sym;

bool get_sym(int _l, int _r) {
    int idx = (n - _l) / B;
    int l = n - B * idx, r = _r + _l - l;
    // (_l, _r) <- (_l - 1, _r + 1) <- ... <- (l, _r + _l - l)
    
    if (l <= r && !is_sym[idx][r]) return false;
    while (l > _l) {
        l--; r++;
        if (l <= r && abs(arr[l] - arr[r]) > k) return false;
    }
    return true;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    
    cin >> n >> k;
    arr.resize(n);
    for (int i = 0; i < n; i++) cin >> arr[i];

    for (int x = 1; x <= 1'000; x++) {
        for (int r = 0; r < n; r++) is_diff[x][r] = (abs(arr[r] - x) <= k);
    }

    bitset<202020> now;
    for (int i = 0; i < n; i++) now[i] = 1;
    is_sym.push_back(now);
    for (int l = n - 1; l >= 0; l--) {
        now = (now << 1);
        now[n] = 0;
        now[l+1] = now[l] = 1;
        now &= is_diff[arr[l]];

        if ((n - l) % B == 0) {
            is_sym.push_back(now);
        }
    }

    for (int c = 0; c < n; c++) {
        int mn = 0, mx = n - 1;
        while (mn < mx) {
            int mid = (mn + mx + 1) / 2;
            int l = c - mid, r = c + mid;
            if (l < 0 || r >= n) {
                mx = mid-1;
                continue;
            }
            if (get_sym(l, r)) mn = mid;
            else mx = mid-1;
        }
        cout << 2*mn+1 << " ";
    }
    cout << endl;

    for (int c = 1; c < n; c++) {
        int mn = 0, mx = n - 1;
        while (mn < mx) {
            int mid = (mn + mx + 1) / 2;
            int l = c - mid, r = c - 1 + mid;
            if (l < 0 || r >= n) {
                mx = mid-1;
                continue;
            }
            if (get_sym(l, r)) mn = mid;
            else mx = mid-1;
        }
        cout << 2*mn << " ";
    }
    cout << endl;

    return 0;
}
```
