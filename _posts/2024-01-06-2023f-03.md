## 2023f-03-01

(https://www.acmicpc.net/problem/16598)

$0$부터 $k$까지 모두 채우기 위해 필요한 버튼 입력의 수를 센다. 그리고 투 포인터로 버튼 입력이 $P$ 이하인 가장 긴 구간의 길이를 찾는다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

int n, P, arr[202020], sum[2020202], ans;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    for (int i = 0; i <= 2000000; i++) sum[i] = 1;

    cin >> n >> P;
    for (int i = 1; i <= n; i++) {
        int t; cin >> t;
        sum[t] = 0;
    }
    
    for (int i = 1; i <= 2000000; i++) {
        sum[i] = sum[i - 1] + sum[i];
    }

    int p = 0, q = 0;
    while (q <= 2000000) {
        if (sum[q] - (p == 0 ? 0 : sum[p-1]) <= P) {
            ans = max(ans, q-p+1);
            q++;
        }
        else p++;
    }

    cout << ans << "\n";

    return 0;
}
```

## 2023f-03-02

문제 조건을 해석하면 임의로 두 부분구간 $[a, b)$와 $[b, c)$를 잡았을 때 $[a, b)$의 구간합과 $[b, c)$의 구간합의 차이가 최대가 되도록 하는 것임을 알 수 있다. 즉, $b$를 고정하고 $[a, b)$의 구간합의 최댓값과 최솟값을 찾고, $[b, c)$의 구간합의 최댓값과 최솟값을 찾으면 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

int n;
ll arr[101010];
ll lt[101010][2], rt[101010][2], ans = 0; //(min, max)

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    lt[1][0] = lt[1][1] = arr[1];
    for (int i = 2; i <= n; i++) {
        lt[i][0] = min(arr[i], arr[i] + lt[i-1][0]);
        lt[i][1] = max(arr[i], arr[i] + lt[i-1][1]);
    }
    rt[n][0] = rt[n][1] = arr[n];
    for (int i = n-1; i >= 1; i--) {
        rt[i][0] = min(arr[i], arr[i] + rt[i+1][0]);
        rt[i][1] = max(arr[i], arr[i] + rt[i+1][1]);
    }

    for (int i = 2; i <= n; i++) {
        ans = max(ans, abs(lt[i-1][0] - rt[i][1]));
        ans = max(ans, abs(lt[i-1][1] - rt[i][0]));
    }

    cout << ans << "\n";

    return 0;
}
```

## 2023f-03-03

(https://www.acmicpc.net/problem/23152)

학생들을 실력 순으로 나열하자. 이때 같은 실력이고 리더쉽이 뛰어난 학생을 붙여놓자. 그러면 팀을 만드는 것은 한 학생 양쪽에서 리더쉽이 뛰어나지 않은 학생 둘을 매칭하는 것과 같다. 또한 각 리더쉽이 뛰어난 학생은 오른쪽에 있는 학생들 중 한 명과 매칭될 수 있다.

이러한 관찰을 기반으로 리더쉽이 뛰어난 학생 $A_i$명의 나열을 기준으로 그룹으로 나누자. 그러면 각 그룹에 있는 리더쉽이 뛰어나지 않은 학생의 수 $S$는 $\sum B_i - S$ + 왼쪽에 있는 리더쉽이 뛰어난 학생의 수 이하여야 한다.

엄밀하게 $A_l > 0, A_r > 0, A_{l+1} = A_{l+2} = \cdots = A_{r-1}=0$이라고 하자. 그리고 $A_l$ 주변에서 $k$ 명의 리더쉽이 뛰어나지 않은 학생을 가져갔다고 하자. 그러면 $[l, r]$에서 하나의 그룹이 만들어지고 이 그룹에서 $S$는

$$
\sum_{i=l+1}^{r-1} {B_i} + (B_l - A_l - k) \leq S \leq \sum_{i=l+1}^{r-1} {B_i} + (B_l - A_l - k)  + (B_r - A_r)
$$

을 만족하고

$$
2S \leq \sum {(B_i - A_i)} + \sum_{i=1}^l A_i
$$

여야 팀 구성이 존재할 수 있다. 모든 구간에서 $S$가 충분히 작아야 한다는 사실을 고려했을 때, $S$를 조건을 만족하면서 최대한 크게 해야 $k$의 값이 커지고 바로 오른쪽 구간에서의 $S$의 하한이 작아진다.

이 과정을 $O(N)$에 구현하면 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

int t, n;
ll arr[101010][2], sum[101010][2];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> t;
    while (t--) {
        bool ans = true;
        cin >> n;
        for (int i = 1; i <= n; i++) cin >> arr[i][0] >> arr[i][1];
        for (int i = 1; i <= n; i++) {
            sum[i][0] = sum[i - 1][0] + arr[i][0];
            sum[i][1] = sum[i - 1][1] + arr[i][1];
        }
        int l = 0, r; ll k = 0;
        for (int i = 1; i <= n+1; i++) {
            if (i <= n && arr[i][0] == 0) continue;
            r = i;
            
            ll upp = (sum[n][1] - sum[n][0] + sum[l][0]) / 2;
            if (upp < sum[r-1][1] - sum[l][1] + (arr[l][1] - arr[l][0] - k)) {
                ans = false;
                break;
            }
            k = min(upp - (sum[r-1][1] - sum[l][1] + (arr[l][1] - arr[l][0] - k)), arr[r][1] - arr[r][0]);
            
            l = i;
        }

        if (ans) cout << "YES\n";
        else cout << "NO\n";
    }

    return 0;
}
```

## 2023f-03-04

(https://www.acmicpc.net/problem/23493)

**그래프에서 최단거리를 찾는 문제이므로 다익스트라를 사용할 확률이 높다. 따라서 풀이의 방향을 다익스트라를 사용하는 쪽으로 전개해보자. 기본적으로 다익스트라는 동적계획법 기반이므로 dp 점화식을 만들어야 하고 이를 위해서는 최적해의 형태를 특정해야 한다.**

최적해는 다음과 같은 형태로 특정된다.

- $s$에서 $t$로 가는 한 경로가 존재하고, 그 **경로 상의 점들** $a_i$와 $b_i$를 생각하자. 단, $a_i$에서 $b_i$로 가는 경로가 존재하고, **$b_i$가 $a_{i+1}$보다 $s$에 가까워야 한다**. 그러면 아래와 같이 $s \rightarrow t \rightarrow s$의 경로를 만들 수 있다.

$$
s \rightarrow t = a_1 \rightarrow b_1 \rightarrow a_2 \rightarrow b_2 \rightarrow \cdots \rightarrow a_k \rightarrow b_k = s
$$

이 형태를 바탕으로 dp 상태 전이를 만들어보자. $s$에서 정방향 간선을 통해 $t$로 가는 직접 경로와 $s$에서 역방향 간선을 통해 $t$로 가는 교차 경로를 나타내기 위해 $(u, v)$의 쌍으로 경로를 표현할 것이다.

- 정방향 간선 사용
    - $u$와 정방향 간선으로 연결된 모든 정점 $w$에 대해 $(u, v) \rightarrow (w, v)$
    - 가중치는 $w \neq v$일 때 $p_w$이다. $w=v$이면 $0$이다.
- 역방향 간선 사용
    - $v$와 역방향 간선으로 연결된 모든 정점 $w$에 대해 $(u, v) \rightarrow (u, w)$
    - 가중치는 $w \neq u$일 때 $p_w$이다. $w = u$이면 $0$이다.
- 교차전이
    - 교차전이는 $s \rightarrow t$ 경로에서 겹치는 부분을 처리하기 위한 작업이다.
        - 정방향으로 이동하는 경로가 $b_i$에 도달하고 역방향으로 이동하는 경로가 $a_{i+1}$에 도달했다고 생각하자. 만약 계속 진행하면 $b_i$가 $a_i$에서 지나갔던 정점으로 지나가더라도 상태에는 지나간 경로가 기록되지 않으므로 중복하여 가중치를 더하게 된다. 즉, 서로의 위치를 바꿈으로써 중간 구간을 겹쳐 사용하고 정방향 이동이 $a_{i+1}$에 도달하고 역방향 이동이 $b_i$에 도달하게 하는 것이다.
    - $u$와 $v$를 연결하는 경로가 존재한다면 $(u, v) \rightarrow (v, u)$
    - 가중치는 $u$와 $v$ 사이의 단순 최단거리에서 $p_v$를 뺸 값이다.

시작 상태는 $(s, s)$, 도착 상태는 $(t, t)$이다. $O(N)$개의 전이, $O(N^2)$개의 상태가 존재하므로 $O(N^3 \log N)$에 다익스트라 알고리즘을 사용하면 해결할 수 있다. 참고로 교차전이에서 $u$와 $v$ 사이의 단순 최단거리를 $O(1)$에 구하기 위해 $O(N^3)$에 플로이드-와샬 알고리즘으로 전체-쌍 최단경로 문제에 대한 table을 만들어 놓아야 한다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pii;

int T, n, m, s, t, p[220];
vector<int> adj[220];
vector<int> rev[220];
int dis[220][220];
int ans[220][220];

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> T;
    while (T--) {
        cin >> n >> m >> s >> t;
        for (int i = 1; i <= n; i++) cin >> p[i];

        for (int i = 1; i <= n; i++) {
            adj[i].clear();
            rev[i].clear();
            for (int j = 1; j <= n; j++) {
                if (i == j) dis[i][j] = p[i];
                else dis[i][j] = 1e9;
                ans[i][j] = 1e9;
            }
        }

        for (int i = 1; i <= m; i++) {
            int u, v;
            cin >> u >> v;
            adj[u].push_back(v);
            rev[v].push_back(u);
            dis[u][v] = p[v];
        }

        for (int k = 1; k <= n; k++) {
            for (int i = 1; i <= n; i++) {
                for (int j = 1; j <= n; j++) {
                    dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j]);
                }
            }
        }

        priority_queue<pair<int, pii>> pq;
        pq.push({-p[s], (pii){s, s}});
        ans[s][s] = p[s];

        while (!pq.empty()) {
            int u = pq.top().second.first, v = pq.top().second.second, val = -pq.top().first;
            pq.pop();
            if (val > ans[u][v]) continue;

            for (int i = 0; i < adj[u].size(); i++) {
                int w = adj[u][i];
                if (w == v) {
                    if (ans[w][v] > val) {
                        ans[w][v] = val;
                        pq.push({-ans[w][v], (pii){w, v}});
                    }
                }
                else {
                    if (ans[w][v] > val + p[w]) {
                        ans[w][v] = val + p[w];
                        pq.push({-ans[w][v], (pii){w, v}});
                    }
                }
            }

            for (int i = 0; i < rev[v].size(); i++) {
                int w = rev[v][i];
                if (w == u) {
                    if (ans[u][w] > val) {
                        ans[u][w] = val;
                        pq.push({-ans[u][w], (pii){u, w}});
                    }
                }
                else {
                    if (ans[u][w] > val + p[w]) {
                        ans[u][w] = val + p[w];
                        pq.push({-ans[u][w], (pii){u, w}});
                    }
                }
            }

            if (ans[v][u] > val + dis[u][v] - p[v]) {
                ans[v][u] = val + dis[u][v] - p[v];
                pq.push({-ans[v][u], (pii){v, u}});
            }
        }

        if (ans[t][t] >= 1e9) cout << "-1\n";
        else cout << ans[t][t] << "\n";

    }

    return 0;
}
```

## 2023f-03-05

(https://www.acmicpc.net/problem/23472)

세그먼트 트리를 사용해 구간에 속하는 한 점을 고르는 문제는 쉽게 해결할 수 있다. 유사하게 이 문제도 한 점을 고르는 문제로 환원해보자.

가장 간단한 방법은 고를 수 있는 $O(N^2)$개의 점 쌍을 생각하는 것이다. 각 점 쌍에 대해 $(x_a, x_b, w_a + w_b)$만 있어도 각 쿼리의 조건을 만족하는지 쉽게 파악할 수 있으므로, 이차원 좌표평면에서 $(x_a, x_b)$에 $w_a + w_b$가 작혀있다고 생각하자. 그리고 $L_i, R_i$가 주어질 때 $x=L_i, y=R_i$를 기준으로 영역을 나누면 2, 4 사분면이 해당하는 영역이므로 이차원 좌표평면에서 오프라인 쿼리를 사용한 스위핑과 세그먼트 트리로 $O(N^2 \log N + Q \log N)$의 시간에 문제를 해결할 수 있다.

즉, 문제를 like-linear한 시간에 해결하려면 고를 수 있는 점 쌍의 개수를 like-linear하게 줄여야 한다. 지금 like-linear하게 하지 못하는 가장 큰 이유가 $y$좌표 조건 때문이므로 $y$좌표 조건을 무시할 수 있도록 $a_y < m \leq b_y$인 경우만 고려해보자. 그러면 다음 관찰이 성립한다.

- 구역 $[-\infty, 0) \times (0, m)$에서 가장 가중치가 큰 점 또는 구역 $(0, \infty) \times [m, \infty)$에서 가장 가중치가 큰 점 중 적어도 하나는 포함해야 한다.
    - 만약 두 점이 $x=L_i, x=R_i$를 기준으로 같은 쪽에 있다면 두 점을 모두 고르는 것이 최적이다
    - 만약 두 점이 $x=L_i, x=R_i$를 기준으로 다른 쪽에 있다면 안쪽에서 두 점을 고를 때는 안쪽에 있는 가장 가중치가 큰 점 하나를 포함하는게 더 합이 크고, 바깥쪽에서 두 점을 고를 때도 바깥쪽에 있는 가장 가중치가 큰 점 하나를 포함하는게 더 합이 크기 때문이다.

이제 우리는 각 $m$에 대해 $O(N)$개의 점만 고려해도 충분하다. 하지만 naive한 방식으로 한다면 제한하였음에도 불구하고 유의미한 구분이 생기는 $m$의 종류의 수가 $O(N)$이므로 여젼히 $O(N^2)$의 계산 시간이 든다.

최적화의 포인트는 $a_y < m \leq b_y$를 제외한 나머지 경우는 $m \leq a_y < b_y$와 $0 < a_y < b_y < m$의 경우라는 사실이다. 그리고 두 경우는 전체 $y$좌표 구간이었던 $(0, \infty)$를 $[m, \infty)$와 $(0, m)$으로 분할한 것이다. 즉, 재귀적으로 절반씩 분할해나가는 방법으로 $O(N \log N)$의 시간에 $O(N \log N)$개의 최적해일 수 있는 점의 쌍을 찾으면 된다.

(사실 분할 정복을 처음에 생각한다면 $a_y < m \leq b_y$인 경우만 생각해보자는 아이디어가 역으로 나올 수도 있다. 라고 썼지만 대체 이런 아이디어는 누가 생각해낸걸까)

즉, $O(N \log N)$개의 후보 점을 찾으면 스위핑과 세그먼트 트리를 사용하여 $O(N \log^2 N + Q \log N)$에 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, q;
struct pts {
    int x;
    int y;
    int w;
    bool operator<(const pts& other) const {
        if (y == other.y) return w < other.w;
        return y < other.y;
    }
} lt[101010], rt[101010];
int w_max_lt[101010][2], w_max_rt[101010][2];
vector<pts> appropriate_pts;

struct qry {
    int l, r, idx;
    bool operator<(const qry& other) const {
        return r < other.r;
    }
} qrys[505050];
vector<int> xlist, ylist;
int ans[505050];

void init() {
    for (int i = 1; i <= n; i++) {
        if (w_max_lt[i-1][0] > lt[i].w) {
            w_max_lt[i][0] = w_max_lt[i-1][0];
            w_max_lt[i][1] = w_max_lt[i-1][1];
        }
        else if (w_max_lt[i-1][0] < lt[i].w) {
            w_max_lt[i][0] = lt[i].w;
            w_max_lt[i][1] = i;
        }
        else {
            w_max_lt[i][0] = lt[i].w;
            w_max_lt[i][1] = min(w_max_lt[i-1][1], i);
        }

        ylist.push_back(lt[i].y);
        xlist.push_back(lt[i].x);
    }

    for (int i = n; i >= 1; i--) {
        if (w_max_rt[i+1][0] > rt[i].w) {
            w_max_rt[i][0] = w_max_rt[i+1][0];
            w_max_rt[i][1] = w_max_rt[i+1][1];
        }
        else if (w_max_rt[i+1][0] < rt[i].w) {
            w_max_rt[i][0] = rt[i].w;
            w_max_rt[i][1] = i;
        }
        else {
            w_max_rt[i][0] = rt[i].w;
            w_max_rt[i][1] = min(w_max_rt[i+1][1], i);
        }

        ylist.push_back(rt[i].y);
    }

    sort(xlist.begin(), xlist.end());
    sort(ylist.begin(), ylist.end());
}

pair<int, int> find_appropriate_points(int bottom, int top) {
    int mid = (bottom + top) / 2;
    int lt_st = lower_bound(lt + 1, lt + n + 1, (pts){0, ylist[bottom], 0}) - lt;
    int lt_ed = lower_bound(lt + 1, lt + n + 1, (pts){0, ylist[mid]+1, 0}) - lt - 1;
    int rt_st = lower_bound(rt + 1, rt + n + 1, (pts){0, ylist[mid], 0}) - rt;
    int rt_ed = lower_bound(rt + 1, rt + n + 1, (pts){0, ylist[top]+1, 0}) - rt - 1;

    if (bottom == top) {
        return {(lt_st > lt_ed ? -1 : lt_st), (rt_st > rt_ed ? -1 : rt_st)};
    }

    pair<int, int> dn_prev = find_appropriate_points(bottom, mid);
    pair<int, int> up_prev = find_appropriate_points(mid+1, top);

    for (int i = rt_st; i <= rt_ed; i++) {
        int w_max_lt_idx = dn_prev.first;
        if (w_max_lt_idx == -1) continue;
        appropriate_pts.push_back((pts){lt[w_max_lt_idx].x, rt[i].x, lt[w_max_lt_idx].w + rt[i].w});
    }

    for (int i = lt_st; i <= lt_ed; i++) {
        int w_max_rt_idx = up_prev.second;
        if (w_max_rt_idx == -1) continue;
        appropriate_pts.push_back((pts){lt[i].x, rt[w_max_rt_idx].x, lt[i].w + rt[w_max_rt_idx].w});
    }

    int dn_ret, up_ret;
    if (lt[dn_prev.first].w > lt[up_prev.first].w) dn_ret = dn_prev.first;
    else if (lt[dn_prev.first].w < lt[up_prev.first].w) dn_ret = up_prev.first;
    else dn_ret = min(dn_prev.first, up_prev.first);
    if (rt[dn_prev.second].w > rt[up_prev.second].w) up_ret = dn_prev.second;
    else if (rt[dn_prev.second].w < rt[up_prev.second].w) up_ret = up_prev.second;
    else up_ret = max(dn_prev.second, up_prev.second);
    return {dn_ret, up_ret};
}

struct maxseg {
    int tree[404040];
    void init(int v, int st, int ed) {
        if (st == ed) {
            tree[v] = -1;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }
    void update(int v, int st, int ed, int idx, int val) {
        if (st == idx && ed == idx) {
            tree[v] = max(tree[v], val);
            return;
        }
        if (st > idx || ed < idx) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, idx, val);
        update(2 * v + 1, mid + 1, ed, idx, val);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }
    int get(int v, int st, int ed, int lt, int rt) {
        if (lt <= st && ed <= rt) return tree[v];
        if (lt > ed || rt < st) return -1;
        int mid = (st + ed) / 2;
        return max(
            get(2 * v, st, mid, lt, rt),
            get(2 * v + 1, mid + 1, ed, lt, rt)
        );
    }
} seg;

int main() {

    cin >> n;
    for (int i = 1; i <= n; i++) cin >> lt[i].x >> lt[i].y >> lt[i].w;
    for (int i = 1; i <= n; i++) cin >> rt[i].x >> rt[i].y >> rt[i].w;
    sort(lt + 1, lt + n + 1);
    sort(rt + 1, rt + n + 1);
    
    init();

    find_appropriate_points(0, 2*n-1);
    sort(appropriate_pts.begin(), appropriate_pts.end());

    cin >> q;
    for (int i = 1; i <= q; i++) cin >> qrys[i].l >> qrys[i].r;
    for (int i = 1; i <= q; i++) qrys[i].idx = i;
    for (int i = 1; i <= q; i++) ans[i] = -1;
    sort(qrys + 1, qrys + q + 1);

    seg.init(1, 0, n-1);
    int p = 0;
    for (int i = 1; i <= q; i++) {
        while (p < appropriate_pts.size() && appropriate_pts[p].y < qrys[i].r) {
            int idx = lower_bound(xlist.begin(), xlist.end(), appropriate_pts[p].x) - xlist.begin();
            seg.update(1, 0, n-1, idx, appropriate_pts[p].w);
            p++;
        }

        int qidx = lower_bound(xlist.begin(), xlist.end(), qrys[i].l) - xlist.begin();
        ans[qrys[i].idx] = max(
            ans[qrys[i].idx], 
            seg.get(1, 0, n-1, qidx, n-1)
        );
    }

    seg.init(1, 0, n-1);
    p = appropriate_pts.size()-1;
    for (int i = q; i >= 1; i--) {
        while (p >= 0 && appropriate_pts[p].y > qrys[i].r) {
            int idx = lower_bound(xlist.begin(), xlist.end(), appropriate_pts[p].x) - xlist.begin();
            seg.update(1, 0, n-1, idx, appropriate_pts[p].w);
            p--;
        }

        int qidx = lower_bound(xlist.begin(), xlist.end(), qrys[i].l) - xlist.begin();
        ans[qrys[i].idx] = max(
            ans[qrys[i].idx], 
            seg.get(1, 0, n-1, 0, qidx-1)
        );
    }

    for (int i = 1; i <= q; i++) cout << ans[i] << "\n";

    return 0;
}
```
