## 1. 도로 포장 (0:13:00)

가능한 모든 $O(\frac{2N!}{N!N!})$가지 경우를 모두 시도하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n;
long long w[20][20];
long long now = 0, ans = 0;
 
int ch[20];
 
void solve(int L) {
    if (L == n/2) ans = max(ans, now);
    int a = 0;
    for (int i = 1; i <= n; i++) {
        if (ch[i] == 0) {
            a = i;
            break;
        }
    }
    ch[a] = 1;
    for (int i = 1; i <= n; i++) {
        if (ch[i] == 0) {
            ch[i] = 1;
            now += w[a][i];
            solve(L+1);
            now -= w[a][i];
            ch[i] = 0;
        }
    }
    ch[a] = 0;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 1; i <= n*(n-1)/2; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        w[a][b] = w[b][a] = c;
    }
 
    solve(0);
 
    cout << ans << "\n";
    
    return 0;
}
```

## 2. 순찰자 1 (0:59:31)

$O(N^2 \cdot 2^N)$ TSP를 짜면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n;
long long w[20][20];
long long dp[20][1<<20];
long long ans = 1e18;
 
int cnt = 0;
long long solve(int v, int mask) {
    if (mask == (1<<n)-1) {
        if (w[v][0] == 0) return 1e18;
        else return w[v][0];
    }
    if (dp[v][mask] >= 0) return dp[v][mask];
 
    long long val = 1e18;
    for (int i = 0; i < n; i++) {
        if (mask & (1<<i)) continue;
        if (w[v][i] == 0) continue;
        val = min(val, solve(i, mask | (1<<i)) + w[v][i]);
    }
    return dp[v][mask] = val;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> w[i][j];
        }
    }
 
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < (1<<n); j++) dp[i][j] = -1;
    }
 
    cout << solve(0, 1) << "\n";
    
    return 0;
}
```

## 3. 순찰자 2 (2:14:49)

$O(N!)$은 TLE가 나지만 $O((N/2)!^2)$은 TLE가 나지 않음을 이용하여 meet in the middle을 사용하자. $N/2$번째에 방문하는 지점을 고정해 놓고 풀면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n, m, k;
long long l, w[14][14];
 
long long get_dis(vector<int> vec) {
    long long ret = 0;
    for (int i = 1; i < vec.size(); i++) {
        ret += w[vec[i-1]][vec[i]];
    }
    return ret;
}
 
int ch[14];
vector<int> lt, rt;
vector<long long> lt_dis, rt_dis;
void solve(int L, int bef, int exc) {
    if (L == m) {
        lt.clear(); lt.push_back(0);
        rt.clear(); rt.push_back(exc);
        for (int i = 1; i < n; i++) {
            if (i == exc) continue;
            if (ch[i] == 1) lt.push_back(i);
            else rt.push_back(i);
        }
        lt.push_back(exc);
        rt.push_back(0);
 
        lt_dis.clear();
        rt_dis.clear();
        while (true) {
            lt_dis.push_back(get_dis(lt));
            if (!next_permutation(lt.begin()+1, lt.end()-1)) break;
        }
        while (true) {
            rt_dis.push_back(get_dis(rt));
            if (!next_permutation(rt.begin()+1, rt.end()-1)) break;
        }
 
        sort(rt_dis.begin(), rt_dis.end());
        for (int i = 0; i < lt_dis.size(); i++) {
            auto iter = lower_bound(rt_dis.begin(), rt_dis.end(), l-lt_dis[i]);
            if (iter == rt_dis.end()) continue;
            if (l == *iter + lt_dis[i]) {
                cout << "possible\n";
                exit(0);
            }
        }
 
        return;
    }
    for (int i = bef; i < n; i++) {
        if (i == exc) continue;
        ch[i] = 1;
        solve(L+1, i+1, exc);
        ch[i] = 0;
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> l;
    m = (n-1)/2;
    k = n-m-1;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> w[i][j];
        }
    }
 
    for (int a = 1; a < n; a++) {
        solve(0, 1, a);
    }
 
    cout << "impossible\n";
 
    return 0;
}
```

## 4. 순찰자 3

일단 문제에서 조건으로 준 트리를 구해야 한다. 자명하게 문제에서 언급한 트리는 주어진 정점으로 만들 수 있는 완전 그래프의 MST이다.

이제 트리에서 어떤 순서로 정점을 방문해야 거리가 최대가 되는지 생각하자. 어떤 간선을 기준으로 양쪽에 각각 $a$개와 $b$개의 정점이 있다면 그 간선이 통과될 수 있는 횟수의 최댓값은 $2\min(a, b)$이다. 이제 적절한 배치로 모든 간선에 대해 $2\min(a, b)$번 통과되는 방법이 있음을 보이자.

이제 트리를 구했으면 트리의 센트로이드를 생각하면, 트리의 센트로이드에 연결된 모든 부분 서브트리의 크기는 $N/2$ 이하이므로, 각각의 서브트리에 속하는 간선이 통과될 수 있는 횟수는 그 정점의 자식 노드의 개수가 된다(바깥쪽이 안쪽보다 정점 개수가 많기 때문). 따라서 센트로이드에 직접 연결된 서브트리를 하나의 그룹으로 볼 때 매 이동에서 다른 그룹에 있는 정점으로 이동하도록 하면 $2\min(a, b)$번 통과하는 것이 가능하다.

이제 $O(N^2 \log N)$에 MST를 구하고, $O(N)$에 센트로이드를 구하자. 시작 정점은 센트로이드여야 하므로 센트로이드를 시작 정점으로 하고 각 그룹에 속하는 정점의 목록을 저장하자. 그리디하게 가장 많은 개수의 정점이 남은 집합의 아무 정점을 골라 사용하면 항상 가능하므로 이 경로를 하나 출력하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;

int n, w[110][110];
vector< pair<ll, pii> > edges;
int root[110];
vector<int> adj[110];

int _find(int v) {
    if (root[v] == v) return v;
    else return root[v] = _find(root[v]);
}

bool _union(int v, int u) {
    v = _find(v);
    u = _find(u);
    if (v == u) return false;
    root[v] = u;
    return true;
}

int sz[110];
void get_sz(int v, int par = -1) {
    sz[v] = 1;
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        if (u == par) continue;
        get_sz(u, v);
        sz[v] += sz[u];
    }
}

int get_cent(int v, int par = -1) {
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        if (u == par) continue;
        if (sz[u]*2 > n) return get_cent(u, v);
    }
    return v;
}

void add(int v, int par, vector<int>* vec) {
    (*vec).push_back(v);
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        if (u == par) continue;
        add(u, v, vec);
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> w[i][j];
            if (i < j) edges.push_back({w[i][j], (pii){i, j}});
        }
    }

    sort(edges.begin(), edges.end());
    for (int i = 0; i < n; i++) root[i] = i;
    int cnt = 0;
    for (int i = 0; i < edges.size(); i++) {
        int v = edges[i].second.first, u = edges[i].second.second;
        if (_union(v, u)) {
            cnt++;
            adj[v].push_back(u);
            adj[u].push_back(v);
        }
        if (cnt == n-1) break;
    }

    get_sz(1);
    int c = get_cent(1);

    ll ans = 0;
    vector<int> res;
    vector<int> gup[110];
    for (int i = 0; i < adj[c].size(); i++) {
        add(adj[c][i], c, &gup[i]);
    }

    int v = c, g = -1;
    res.push_back(c);
    for (int i = 1; i < n; i++) {
        int sz = 0, idx;
        for (int j = 0; j < adj[c].size(); j++) {
            if (j == g) continue;
            if (sz < gup[j].size()) {
                sz = gup[j].size();
                idx = j;
            }
        }
        res.push_back(gup[idx].back());
        ans += w[v][gup[idx].back()];
        v = gup[idx].back();
        gup[idx].pop_back();
        g = idx;
    }
    ans += w[v][c];
    res.push_back(c);

    cout << ans << "\n";
    for (int i = 0; i < res.size(); i++) cout << res[i]+1 << " ";
    cout << "\n";

    return 0;
}
```
