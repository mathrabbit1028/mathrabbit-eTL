## 2023f-02-01

불가능한 경우는 한 종류의 알파벳이 절반 초과로 차지하는 경우 뿐이다. 즉, 이 경우를 제외한 모든 경우에는 답이 존재한다.

또한 답이 존재한다면 주어진 문자열을 사전순으로 나열했을 때, 앞 절반은 사전순으로 가장 앞선 것을 택한 세구다세구 문자열이 존재한다. 즉, 경계에 해당하는 문자열이 “e”라면 “e”와 대칭되는 부분까지 “e”를 제외한 문자를 사전 순으로 배치하고, 그 이후에 “e”의 개수만큼 e를 출력한 뒤, 그 뒷 부분을 사전 순으로 출력하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    string s;
    cin >> s;
    int n = s.length();
    sort(s.begin(), s.end());

    int m = n/2; // 0 ~ m-1, n-m ~ n-1

    int alpha[26];
    for (int i = 0; i < 26; i++) alpha[i] = 0;
    for (int i = 0; i < n; i++) alpha[s[i]-'a']++;
    for (int i = 0; i < 26; i++) if (alpha[i] > n-m) {
        cout << "NIE\n";
        return 0;
    }

    cout << "TAK\n";

    int cnt = 0, st = n-m;
    for (int i = m-1; i >= 0; i--) {
        if (s[i] == s[m-1]) cnt++;
        else break;
    }
    for (int i = n-m; i < n; i++) {
        if (s[i] == s[m-1]) st = i+1;
        else break;
    }

    for (int i = 0; i < n-m; i++) cout << s[i];
    for (int i = 0; i < cnt; i++) cout << s[st+i];
    for (int i = n-m; i < st; i++) cout << s[i];
    for (int i = st+cnt; i < n; i++) cout << s[i];

    cout << "\n";
    
    return 0;
}
```

## 2023f-02-02

$[i_s, i_e]$를 고정하면 bitwise-or이 $K$여야 하므로 bitwise-or의 특성에 의해 투 포인터를 사용할 수 있다. 이를 단순 구현하면 $O(N^3 \log X)$에 구현할 수 있다.

단순 구현으로는 상수가 커 TLE를 받을 수 있으므로 다음 두 가지 최적화를 하자.

- $[p, q]$ 구간에서 포함되면 안되는 비트가 포함되는 경우 구간을 $[q+1, q]$로 바꾸고 bitcnt를 0으로 한다
- 현재 보고 있는 영역의 넓이가 답 이상이면 이후를 보지 않고 $p$를 1 늘린다.

```cpp
#include <bits/stdc++.h>
using namespace std;

int n, k, arr[550][550][32];

int bitcnt[32];

void add(int s, int e, int y) {
    for (int i = 0; i < 32; i++) {
        bitcnt[i] += arr[e][y][i] - arr[s-1][y][i];
    }   
}

void rem(int s, int e, int y) {
    for (int i = 0; i < 32; i++) {
        bitcnt[i] -= arr[e][y][i] - arr[s-1][y][i];
    }   
}

void clear() {
    for (int i = 0; i < 32; i++) bitcnt[i] = 0;
}

int comp() {
    int ret = -1;
    for (int i = 0; i < 32; i++) {
        bool x = !!bitcnt[i];
        bool y = !!(k & (1<<i));
        if (x && !y) ret = max(ret, 1);
        if (!x && y) ret = max(ret, 0);
    }
    return ret;
}
// -1 : same
//  0 : add more bits
//  1 : remove some bits

int ans = 1e9;
int res[4];

void solve(int s, int e) {
    for (int i = 0; i < 32; i++) bitcnt[i] = 0;

    int p = 1, q = 0;
    while (true) {
        if ((e-s+1)*(q-p+1) >= ans) {
            if (p == n) break;
            rem(s, e, p);
            p++;
        }
        int ret = comp();
        if (ret == -1) {
            if (ans > (e-s+1)*(q-p+1)) {
                ans = (e-s+1)*(q-p+1);
                res[0] = s;
                res[1] = p;
                res[2] = e;
                res[3] = q;
            }
            if (p == n) break;
            rem(s, e, p);
            p++;
        }
        if (ret == 0) {
            if (q == n) break;
            q++;
            add(s, e, q);
        }
        if (ret == 1) {
            if (p == n) break;
            clear();
            p = q+1;
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            int num; cin >> num;
            for (int k = 0; k < 32; k++) {
                if (num & (1<<k)) arr[i][j][k] = 1;
                else arr[i][j][k] = 0;
            }
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            for (int k = 0; k < 32; k++) {
                arr[i][j][k] += arr[i - 1][j][k];
            }
        }
    }

    for (int s = 1; s <= n; s++) {
        for (int e = s; e <= n; e++) {
            solve(s, e);
        }
    }

    cout << res[0] << " " << res[1] << " " << res[2] << " " << res[3] << "\n";

    return 0;
}
```

## 2023f-02-03

시작점 $x$ 기준이 아니라 중앙점 $m$을 기준으로 생각해보자. 구체적으로 $[m-a, m)$ 구간의 합과 $[m, m+a)$ 구간의 합이 모두 $k$ 이하가 되는 $a$의 최댓값 $f(m)$을 $m$마다 찾는 것이다. 이는 이분탐색을 이용하면 각 $m$에 대해 $O(\log N)$에 해결할 수 있다.

이제 이 정보로 주어진 문제를 해결해보자. $[x, x+l)$이 주어진 조건을 만족할 필요충분조건은 다음을 만족하는 것이다.

$$
f\left( x+ \frac{l}{2} \right) \geq  \frac{l}{2}
$$

즉, $m - f(m) \leq x \leq m$이면 $l = 2(m-x)$가 답이 될 수 있다. 즉, $[m-f(m), m]$의 구간을 $m$으로 range update를 하고 $x$에 대한 점 쿼리를 날리면 $l$의 최댓값을 계산할 수 있다.

따라서 전체 시간복잡도는 $O(N \log N)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

int n;
ll k, arr[101010], sum[101010];

struct maxlaxyseg {
    int tree[404040], lazy[404040];
    void prop(int v, int st, int ed) {
        tree[v] = max(tree[v], lazy[v]);
        if (st != ed) {
            lazy[2 * v] = max(lazy[2 * v], lazy[v]);
            lazy[2 * v + 1] = max(lazy[2 * v + 1], lazy[v]);
        }
        lazy[v] = 0;
    }
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            tree[v] = lazy[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }
    
    void update(int v, int st, int ed, int lt, int rt, int val) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) {
            lazy[v] = val;
            prop(v, st, ed);
            return;
        }
        if (lt > ed || rt < st) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, lt, rt, val);
        update(2 * v + 1, mid + 1, ed, lt, rt, val);
        tree[v] = max(tree[2 * v], tree[2 * v + 1]);
    }

    int get(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) return tree[v];
        if (lt > ed || rt < st) return 0;
        int mid = (st + ed) / 2;
        return max(
            get(2 * v, st, mid, lt, rt), 
            get(2 * v + 1, mid + 1, ed, lt, rt)
        );
    }
} seg;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> arr[i];
    for (int i = 1; i <= n; i++) sum[i] = sum[i - 1] + arr[i];

    seg.init(1, 1, n);

    for (int i = 1; i <= n; i++) {
        int st = 0, ed = min(i-1, n-i+1);
        while (st < ed) {
            int a = (st + ed + 1) / 2;
            if (sum[i-1] - sum[i-a-1] <= k && sum[i+a-1] - sum[i-1] <= k) st = a;
            else ed = a - 1;
        }
        seg.update(1, 1, n, i-st, i, i);
    }

    for (int i = 1; i <= n; i++) cout << 2 * (seg.get(1, 1, n, i, i) - i) << " ";
    cout << "\n";

    return 0;
}
```
