## 1. 아름다운 다각형 (0:17:14)

신발끈 공식을 쓰면 된다. 

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pt;
 
#define x first
#define y second
 
bool ccw(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return true;
    else return false;
}
 
int n;
pt pts[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> pts[i].x >> pts[i].y;
    }
 
    __int128_t A = 0;
    for (int i = 0; i < n; i++) {
        A += pts[i].x * pts[(i+1)%n].y;
        A -= pts[(i+1)%n].x * pts[i].y;
    }
 
    ll ans = abs((ll)A);
    if (ans%2 == 0) cout << ans/2 << ".0\n";
    else cout << ans/2 << ".5\n";
```

## 2. 다각형 망가트리기 (0:39:24)

$x$축에 평행한 반직선을 그어 경계와 만나는 횟수가 홀수개이면 안이고, 짝수개이면 밖이다. 기울기가 같은 직선은 세면 안되고, 꼭짓점을 지나는 경우에는 선분이 반직선보다 $y$좌표가 큰 쪽에 있는 경우만 세야 함에 유의하자.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pt;
 
#define x first
#define y second
 
int CCW(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return 1;
    if (res < 0) return -1;
    return 0;
}
 
bool intersect(pt A, pt B, pt C, pt D) {
    if (A.y > B.y) swap(A, B);
    if (((CCW(A, B, C) <= 0 && CCW(A, B, D) >= 0) || (CCW(A, B, C) >= 0 && CCW(A, B, D) <= 0)) && 
        ((CCW(C, D, A) <= 0 && CCW(C, D, B) >= 0) || (CCW(C, D, A) >= 0 && CCW(C, D, B) <= 0)) ) {
        if (CCW(A, B, C) == 0 && CCW(A, B, D) == 0 && CCW(C, D, A) == 0 && CCW(C, D, B) == 0) return false;
        if (CCW(C, D, B) == 0) return false;
        return true;
    }
    else return false;
}
 
int n;
pt pts[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> pts[i].x >> pts[i].y;
    }
 
    int q = 2;
    while (q--) {
        pt P;
        cin >> P.x >> P.y;
 
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            cnt += (int)intersect(pts[i], pts[(i+1)%n], P, (pt){2e9, P.y});
        }
        cout << (cnt % 2 == 0 ? "out" : "in") << "\n";
    }
 
 
    return 0;
}
```

## 3. 최고의 대장장이 토르비욘 (1:01:13)

https://www.acmicpc.net/problem/3679 / https://justicehui.github.io/icpc/2019/08/23/BOJ3679/

가장 $x$좌표가 작은 원점으로 두고 CCW를 이용하여 점들을 시계방향 또는 반시계방향으로 정렬하자. 이때, 원점에서 시작하는 한 직선 위에 있는 점들은 거리가 가까울수록 앞에, 끝나는 한 직선 위에 있는 점들은 거리가 멀수록 앞에 두어야 한다. 나머지 경우에는 거리에 따른 순서가 중요하지 않다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pt;
 
#define x first
#define y second
 
ll dis(pt A, pt B) {
    return (A.x - B.x) * (A.x - B.x) + (A.y - B.y) * (A.y - B.y);
}
 
int CCW(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return 1;
    if (res < 0) return -1;
    return 0;
}
 
bool intersect(pt A, pt B, pt C, pt D) {
    if (A.y > B.y) swap(A, B);
    if (((CCW(A, B, C) <= 0 && CCW(A, B, D) >= 0) || (CCW(A, B, C) >= 0 && CCW(A, B, D) <= 0)) && 
        ((CCW(C, D, A) <= 0 && CCW(C, D, B) >= 0) || (CCW(C, D, A) >= 0 && CCW(C, D, B) <= 0)) ) {
        if (CCW(A, B, C) == 0 && CCW(A, B, D) == 0 && CCW(C, D, A) == 0 && CCW(C, D, B) == 0) return false;
        if (CCW(C, D, B) == 0) return false;
        return true;
    }
    else return false;
}
 
pt O;
bool compare(pair<pt, int> A, pair<pt, int> B) {
    if (CCW(A.first, O, B.first) == 0) return dis(O, A.first) < dis(O, B.first);
    return CCW(A.first, O, B.first) > 0;
}
 
int n;
pair<pt, int> pts[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) {
            cin >> pts[i].first.x >> pts[i].first.y;
            pts[i].second = i;
        }
 
        sort(pts, pts+n);
        O = pts[0].first;
        sort(pts+1, pts+n, compare);
        
        int p = n;
        while (CCW(pts[p%n].first, pts[p-1].first, pts[p-2].first) == 0) p--;
        reverse(pts+p-1, pts+n);
 
        for (int i = 0; i < n; i++) cout << pts[i].second << " ";
        cout << "\n";
    }
 
    return 0;
}
```

## 5. 감시 (2:56:58)

https://www.acmicpc.net/problem/10321

컨벡스 헐의 두 점을 이은 모든 직선에 대해 그 직선과 거리가 가장 먼 직선 왼쪽과 오른쪽 점을 이으면 그 두 점을 대각선으로 하는 가장 큰 넓이의 사각형을 얻을 수 있다.

이제 하나의 점 $P$만 고정하고 다른 점 $Q$를 시계방향으로 회전시키자. 그러면 직선 $PQ$보다 위에 있고 직선과 가장 먼 점은 컨벡스 헐 상에서 똑같이 시계방향으로 회전한다. 유사하게 직선 $PQ$보다 아래에 있고 직선과 가장 먼 점도 컨벡스 헐 상에서 반시계방향으로 회전한다.

이를 이용하면 $O(N)$에 $P$를 포함하는 모든 사각형을 고려할 수 있으므로 $O(N^2)$에 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pt;

#define x first
#define y second

ll dis(pt A, pt B) {
    return (A.x - B.x) * (A.x - B.x) + (A.y - B.y) * (A.y - B.y);
}

ll area(pt A, pt B, pt C) {
    ll ret = 0;
    ret += A.x * B.y - B.x * A.y;
    ret += B.x * C.y - C.x * B.y;
    ret += C.x * A.y - A.x * C.y;
    return abs(ret);
}
 
int CCW(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return 1;
    if (res < 0) return -1;
    return 0;
}
 
bool intersect(pt A, pt B, pt C, pt D) {
    if (A.y > B.y) swap(A, B);
    if (((CCW(A, B, C) <= 0 && CCW(A, B, D) >= 0) || (CCW(A, B, C) >= 0 && CCW(A, B, D) <= 0)) && 
        ((CCW(C, D, A) <= 0 && CCW(C, D, B) >= 0) || (CCW(C, D, A) >= 0 && CCW(C, D, B) <= 0)) ) {
        if (CCW(A, B, C) == 0 && CCW(A, B, D) == 0 && CCW(C, D, A) == 0 && CCW(C, D, B) == 0) return false;
        if (CCW(C, D, B) == 0) return false;
        return true;
    }
    else return false;
}
 
pt O;
bool compare(pt A, pt B) {
    if (CCW(A, O, B) == 0) return dis(O, A) < dis(O, B);
    return CCW(A, O, B) > 0;
}
 
int n;
pt pts[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    int t;
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 0; i < n; i++) {
            cin >> pts[i].x >> pts[i].y;
        }
 
        sort(pts, pts+n);
        O = pts[0];
        sort(pts+1, pts+n, compare);
        int p = n;
        while (CCW(pts[p%n], pts[p-1], pts[p-2]) == 0) p--;
        reverse(pts+p-1, pts+n);
 
        stack<int> stk;
        stk.push(0);
        for (int i = 1; i < n; i++) {
            int a = stk.top(); stk.pop();
            while (stk.size() > 0 && CCW(pts[stk.top()], pts[a], pts[i]) >= 0) {
                a = stk.top(); stk.pop();
            }
            stk.push(a);
            stk.push(i);
        }
        int a = stk.top(); stk.pop();
        while (stk.size() > 0 && CCW(pts[stk.top()], pts[a], pts[0]) >= 0) {
            a = stk.top(); stk.pop();
        }
        stk.push(a);

        ll ans = 0;
        vector<int> hull;
        while (stk.size() > 0) { hull.push_back(stk.top()); stk.pop(); }
        reverse(hull.begin(), hull.end());
        int m = hull.size();
        for (int i = 0; i < m; i++) hull.push_back(hull[i]);
        for (int i = 0; i < m; i++) {
            int p = i, q = i+1;
            for (int j = i+1; j < i+m; j++) {
                while (p < j && abs(area(pts[hull[i]], pts[hull[j]], pts[hull[p]])) < 
                    abs(area(pts[hull[i]], pts[hull[j]], pts[hull[p+1]]))) p++;
                while (q < j) q++;
                while (q < i+m && abs(area(pts[hull[i]], pts[hull[j]], pts[hull[q]])) < 
                    abs(area(pts[hull[i]], pts[hull[j]], pts[hull[q+1]]))) q++;
                ans = max(ans, abs(area(pts[hull[i]], pts[hull[j]], pts[hull[p]])) + 
                    abs(area(pts[hull[i]], pts[hull[j]], pts[hull[q]])));
            }
        }

        if (ans%2 == 0) cout << ans/2 << "\n";
        else cout << ans/2 << ".5\n";

        hull.clear();
    }
 
    return 0;
}
```

## 8. Broken Line (3:46:15)

https://www.acmicpc.net/problem/10523

전체의 20% 이상은 상당히 빡빡한 조건임을 상기하자. 주어진 조건을 만족하는 직선이 존재한다면 임의의 두 점을 골라 얻은 직선이 주어진 조건을 만족할 확률은 $1-(1-p)^2 \geq 0.36$ 이상이다. 즉, 주어진 조건을 만족하는 직선이 존재한다면 $x$번 이 과정을 시행했을 때 주어진 조건을 만족하는 직선을 찾은 횟수를 나타내는 확률분포는 $B(x, 0.64)$를 따른다. 따라서 $100$번 정도만 시도했을 때 발견되지 않는다면 없다고 말할 수 있다: $100$번 시도하였을 때 직선이 있는데 없다고 결론지을 확률은 $0.64^{100} \simeq 4 \times 10^{-20}$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pt;

#define x first
#define y second

ll dis(pt A, pt B) {
    return (A.x - B.x) * (A.x - B.x) + (A.y - B.y) * (A.y - B.y);
}

ll area(pt A, pt B, pt C) {
    ll ret = 0;
    ret += A.x * B.y - B.x * A.y;
    ret += B.x * C.y - C.x * B.y;
    ret += C.x * A.y - A.x * C.y;
    return abs(ret);
}
 
int CCW(pt A, pt B, pt C) {
    ll res = A.x * B.y + B.x * C.y + C.x * A.y - B.x * A.y - C.x * B.y - A.x * C.y;
    if (res > 0) return 1;
    if (res < 0) return -1;
    return 0;
}
 
bool intersect(pt A, pt B, pt C, pt D) {
    if (A.y > B.y) swap(A, B);
    if (((CCW(A, B, C) <= 0 && CCW(A, B, D) >= 0) || (CCW(A, B, C) >= 0 && CCW(A, B, D) <= 0)) && 
        ((CCW(C, D, A) <= 0 && CCW(C, D, B) >= 0) || (CCW(C, D, A) >= 0 && CCW(C, D, B) <= 0)) ) {
        if (CCW(A, B, C) == 0 && CCW(A, B, D) == 0 && CCW(C, D, A) == 0 && CCW(C, D, B) == 0) return false;
        if (CCW(C, D, B) == 0) return false;
        return true;
    }
    else return false;
}
 
pt O;
bool compare(pt A, pt B) {
    if (CCW(A, O, B) == 0) return dis(O, A) < dis(O, B);
    return CCW(A, O, B) > 0;
}
 
int n, p;
pt pts[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n;
    cin >> p;
    for (int i = 0; i < n; i++) {
        cin >> pts[i].x >> pts[i].y;
    }

    if (n == 1) {
        cout << "possible\n";
        return 0;
    }

    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<int> uintd(0, n-1);
    for (int i = 0; i < 100; i++) {
        int a = uintd(gen), b = uintd(gen);
        while (a == b) b = uintd(gen);
        int cnt = 0;
        for (int j = 0; j < n; j++) {
            if (CCW(pts[a], pts[b], pts[j]) == 0) cnt++;
        }
        if (cnt >= ceil(0.01 * p * n)) {
            cout << "possible\n";
            return 0;
        }
    }
    
    cout << "impossible\n";
    return 0;
}
```

## 6. 이중 감시

https://www.acmicpc.net/problem/9875

문제를 해석하면 $A_1, A_2, \cdots, A_N$으로 만든 컨벡스 헐의 경계 및 내부에 속하는 모든 $B_i$의 개수를 세는 문제가 된다. 컨벡스 헐의 볼록성을 이용하면 쉽게 셀 수 있다.

볼록다각형 내부 점 판별은 각 볼록다각형을 특정 한 점 $P_0$을 모두 포함하는 삼각형들로 분할하고 각 삼각형에 속하는지 보면 된다. 생각해보면 직선 $P_0 P_i$로 나뉜 영역들 중 어느 영역에 속하는지에 따라 속할 수 있는 삼각형이 정확히 하나로 정해지고, $P_i$는 $P_0$를 원점으로 두었을 때 단조성을 가지므로 각도에 대한 이분탐색으로 영역을 찾을 수 있다.

그 영역이 삼각형 $P_0 P_i P_{i+1}$이면 $P_i P_{i+1}$과 $P_0 Q$의 교점이 존재하는지 여부로 쉽게 내부를 판별할 수 있고, 시간복잡도는 $O(\log N)$이다. 따라서 문제를 $O(N \log N)$에 해결할 수 있다.

문제는 적어도 하나는 지나게 할 때 봐야하는 점이 선분이라 무한히 많다는 것이다. 하지만, 위의 각도 접근을 고려해보면 양 끝점만 생각하여도 충분하므로 $O(N) \times O(N \log N) = O(N^2 \log N)$에 문제를 해결할 수 있다.
