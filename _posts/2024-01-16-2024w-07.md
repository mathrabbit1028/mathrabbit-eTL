## 2. 쪽지 시험 (0:29:08)

가능한 답안은 $3^K$가지이므로 각 가능한 답안을 제출한 학생의 수를 세자. 이제 각 정답 $3^K$가지에 대해 $2^K$가지의 답을 쓴 사람만 모두 틀린 사람이므로 $O(3^K \cdot 2^K)$의 시간에 모든 경우를 시도해 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n, k, m, mx = 0, mn = 1e9;
int cnt[20202];
int now = 0;
int solve(int L, int answer) {
    if (L == k) return cnt[now];
    int a = answer%3;
    int ret = 0;
    if (a != 0) {
        now += pow(3, L) * 0;
        ret += solve(L+1, answer/3);
        now -= pow(3, L) * 0;
    }
    if (a != 1) {
        now += pow(3, L) * 1;
        ret += solve(L+1, answer/3);
        now -= pow(3, L) * 1;
    }
    if (a != 2) {
        now += pow(3, L) * 2;
        ret += solve(L+1, answer/3);
        now -= pow(3, L) * 2;
    }
    return ret;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n >> k;
    for (int i = 0; i < n; i++) {
        int sum = 0;
        for (int j = 0; j < k; j++) {
            int t;
            cin >> t;
            sum += (t-1) * pow(3, j);
        }
        cnt[sum]++;
    }
 
    for (int i = 0; i < pow(3, k); i++) {
        int val = solve(0, i);
        assert(now == 0);
        mx = max(mx, val);
        mn = min(mn, val);
    }
 
    cout << mn << " " << mx << "\n";
 
    return 0;
}
```

## 3. 양팔 저울 (1:04:27)

mitm을 사용하자. 한 종류의 추를 올려놓지 않거나, 물건과 같은 쪽에 올려놓거나, 물건과 다른 쪽에 올려놓을 수 있으므로 각 절반 부분에서 $3^{N/2}$가지 경우가 나온다. 이 경우를 모두 저장하고 이분탐색으로 합이 추의 무게가 되는 경우의 수가 존재하는지 판단하면 $O(N \cdot 3^{N/2})$의 시간에 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
 
int n, q;
vector<ll> arr;
vector<ll> lt, rt;
 
ll now = 0;
void solve(int st, int ed, vector<ll>* vec) {
    if (st == ed) {
        (*vec).push_back(now);
        return;
    }
    now += arr[st];
    solve(st+1, ed, vec);
    now -= arr[st];
    solve(st+1, ed, vec);
    now -= arr[st];
    solve(st+1, ed, vec);
    now += arr[st];
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n;
    arr.resize(n);
    for (int i = 0; i < n; i++) cin >> arr[i];
    
    solve(0, n/2, &lt);
    solve(n/2, n, &rt);
 
    sort(rt.begin(), rt.end());
 
    cin >> q;
    while (q--) {
        ll w;
        bool ans = false;
        cin >> w;
        for (int i = 0; i < lt.size(); i++) {
            if (*lower_bound(rt.begin(), rt.end(), w - lt[i]) + lt[i] == w) ans = true;
        }
        cout << (ans ? "Y " : "N ");
    }
 
    return 0;
}
```

## 4. 격자판 채우기 (2:30:16)

앞 $3 \times 2$를 채우면 주어진 규칙으로 끝까지 수가 결정된다. 그리고 앞 $3 \times 2$를 채우는 경우의 수는 $O(S^4)$이다. 고로 앞 $3 \times 2$를 결정하고 매 경우에 주어진 숫자 조건과 $0$ 이상 $S$ 이하 조건을 위배하지 않는지 확인하면 된다. 이렇게 되면 최악의 경우 $O(N \cdot S^4)$의 시간이 소요되어 TLE를 받을 것 같지만, 생각보다 많은 경우 중간에서 불가능해지기 때문에 중간에 불가능해지는 순간 다음 경우로 넘어가게 되면 주어진 시간 내에 통과할 수 있다.

참고로, 하나의 숫자까지는 범위로 저장해도 된다. 즉, 한 숫자가 어느 범위에 있을 때 다음 숫자의 범위를 결정할 수 있고, 이를 역순으로 반복하면 처음 시작 숫자의 범위를 결정할 수 있다. 이 아이디어를 사용하면 중간 가지치기 없이 완전한 $O(N \cdot S^3)$풀이를 얻는다.

**$N=1$의 코너케이스가 존재함에 유의하자.**

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
 
int n, s, arr[110][3];
ll ans = 0;
const ll MOD = 993'244'853;
 
int solve(int L, int a, int b, int c, int d, int e, int f) {
    assert(a + b + c == s);
    assert(d + e + f == s);
    if (a < 0 || a > s) return 0;
    if (b < 0 || b > s) return 0;
    if (c < 0 || c > s) return 0;
    if (d < 0 || d > s) return 0;
    if (e < 0 || e > s) return 0;
    if (f < 0 || f > s) return 0;
    if (arr[L-2][0] >= 0 && a != arr[L-2][0]) return 0;
    if (arr[L-2][1] >= 0 && b != arr[L-2][1]) return 0;
    if (arr[L-2][2] >= 0 && c != arr[L-2][2]) return 0;
    if (arr[L-1][0] >= 0 && d != arr[L-1][0]) return 0;
    if (arr[L-1][1] >= 0 && e != arr[L-1][1]) return 0;
    if (arr[L-1][2] >= 0 && f != arr[L-1][2]) return 0;
 
    if (L == n+1) return 1;
 
    return solve(L+1, d, e, f, s-c-e, s-(s-c-e)-(s-a-e), s-a-e);
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n >> s;
    for (int i = 1; i <= n; i++) cin >> arr[i][0];
    for (int i = 1; i <= n; i++) cin >> arr[i][1];
    for (int i = 1; i <= n; i++) cin >> arr[i][2];
 
    if (n == 1) {
        for (int a = 0; a <= s; a++) {
            for (int b = 0; a+b <= s; b++) {
                int c = s-a-b;
                if (arr[1][0] >= 0 && a != arr[1][0]) continue;
                if (arr[1][1] >= 0 && b != arr[1][1]) continue;
                if (arr[1][2] >= 0 && c != arr[1][2]) continue;
                ans += 1;
                ans %= MOD;
            }
        }
    }
    
    for (int a = 0; a <= s; a++) {
        for (int b = 0; a+b <= s; b++) {
            int c = s-a-b;
            for (int d = 0; d <= s; d++) {
                for (int e = 0; d+e <= s; e++) {
                    int f = s-d-e;
                    ans += solve(3, a, b, c, d, e, f);
                    ans %= MOD;
                }
            }
        }
    }
 
    cout << ans << "\n";
 
    return 0;
}
```

## 5. 소수 전구 (3:23:44)

$1000$ 이하의 소수는 $168$개이지만, $\sqrt N$보다 큰 소수 $p, q$의 최소공배수는 $pq > N$이기 때문에 이후에는 모든 전구가 두 번 이상 상태가 바뀌지 않는다. 아직 조작 상태를 결정하지 않은 소수 스위치들의 값이 전부 $\sqrt N$보다 크면 켜져 있는 전구의 개수와 꺼져 있는 전구의 개수를 비교하여 단순한 그리디로 문제를 해결할 수 있다.

중요한 사실은 $N \leq 1000$에서 $\sqrt N$ 이하의 소수는 $31$까지 최대 $11$개라는 것이다. 그리고 $\sum p^{-1} < \sum s^{-1}$은 $O(\log N)$ 이하이기 때문에 실제 재귀 동작 횟수의 상한은 $2^{11} \times N \log N$이 되어 시간 안에 동작한다.

정확히 시간복잡도 식을 쓰면 소수계량함수 $\pi(N)$에 대해 $O(2^{\pi(\sqrt N)} \cdot N \log N)$로 표현된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
 
int n, k, ans = 0;
vector<int> prime;
 
int on[1010], cnt = 0;
 
void solve(int L) {
    if (L == k) {
        ans = max(ans, cnt);
        return;
    }
 
    if (prime[L] > 50) {
        int temp = cnt;
        while (L < k) {
            int t = 0;
            for (int i = prime[L]; i <= n; i += prime[L]) {
                if (on[i] == 0) t++;
                else t--;
            }
            if (t > 0) cnt += t;
            L++;
        }
        ans = max(ans, cnt);
        cnt = temp;
        return;
    }
 
    for (int i = prime[L]; i <= n; i += prime[L]) {
        on[i] = 1 - on[i];
        if (on[i] == 0) cnt -= 1;
        else cnt += 1;
    }
    solve(L+1);
    for (int i = prime[L]; i <= n; i += prime[L]) {
        on[i] = 1 - on[i];
        if (on[i] == 0) cnt -= 1;
        else cnt += 1;
    }
    solve(L+1);
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n >> k;
    prime.resize(k);
    for (int i = 0; i < k; i++) cin >> prime[i];
    sort(prime.begin(), prime.end());
 
    solve(0);
 
    cout << ans << "\n";
 
    return 0;
}
```

## 6. 세 조건

동적계획법을 활용하자. 아래에서 $X=10$이다.

- $sum[i][j]$ : $i$번째 자리(0-based)까지 볼 때, 자릿수의 합이 $j$인 수의 개수
    - 모든 $0 \leq k \leq 9$에 대해 $dp[i+1][j+k] \gets dp[i][j]$
    - 공간 복잡도 및 시간 복잡도는 $O(NX)$와 $O(NX^2)$
- $prod[i][j][{factor}]$ : $i$번째 자리(0-based)까지 볼 때, 자리수의 합이 $j$이며 곱에 있는 소인수 $2, 3, 5, 7$의 지수가 튜플 $factor$인 수의 개수
    - 모든 $0 \leq k \leq 9$에 대해 $dp[i+1][j+k][factor+(k+1\textrm{의 지수})] \gets dp[i][j][factor]$
    - $2, 3, 5, 7$의 지수는 각각 최대 $3N, 2N, N, N$이므로 공간 복잡도 및 시간 복잡도는 $O(6N^5X)$와 $O(6N^6X)$
    - 실제 상태의 개수는 훨씬 작기 때문에 `unordered_map` 등으로 처리하면 훨씬 적은 개수의 상태로 문제를 풀 수 있다.

이 테이블을 모두 채우면 주어진 숫자 $s$의 $\textrm{SUM}(s)$, $\textrm{PROD}(s)$를 이용하여 다음 방식으로 마지막 조건을 제외했을 때 더 앞에 오는 길이 $N$의 숫자의 개수를 구할 수 있다. $R(factor)$은 $2^a \cdot 3^b \cdot 5^c \cdot 7^d$로 계산한 값이다.

$$
\sum_{1 \leq j < \textrm{SUM}(s)} sum[n-1][j] + \sum_{\forall R(factor) < \textrm{PROD}(s)} prod[n-1][\textrm{SUM}(s)][factor]
$$

이제 두 값 모두 같지만 더 앞에 오는 경우의 수를 세어야 하는데, 모든 자리수 $i$에 대해 $i-1$번째 자리까지는 $s$와 같지만 $i$번째 자리가 $s$보다 작은 $\textrm{SUM}$과 $\textrm{PROD}$의 값이 같은 개수를 전부 더함으로써 계산할 수 있고, 이미 앞에서 테이블을 만들어 놓았으므로 반복문으로 쉽게 해결할 수 있다.

**놀랍게도 이 문제는 다항 시간인 $O(N^6)$에 해결된다.** (정확히는 다항 시간이 아니긴 함: 진법을 $X$라 두면 소수계량함수 $\pi(X)$에 대해 $O(N^{2+\pi(X)} \cdot X^2)$이므로)

* 참고: $0 \leq d \leq 9$에 대해 $a_d$를 자리수 중 $d$의 개수라고 하자. 그러면 $a_0 + a_1 + \cdots + a_9 = N$인데, $N \leq 14$에서 가능한 $(a_0, a_1, \cdots, a_9)$의 개수가 $10^6$개 미만임을 이용하는 방법도 있다고 한다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
 
ll n, arr[20];
ll ans = 0;
ll sum[15][150];
unordered_map<ll, ll> prod[15][150];
 
ll SUM = 0;
struct factor {
    ll two;
    ll three;
    ll five;
    ll seven;
    factor operator+(const factor& other) const {
        factor ret;
        ret.two = two + other.two;
        ret.three = three + other.three;
        ret.five = five + other.five;
        ret.seven = seven + other.seven;
        return ret;
    }
    factor operator-(const factor& other) const {
        factor ret;
        ret.two = two - other.two;
        ret.three = three - other.three;
        ret.five = five - other.five;
        ret.seven = seven - other.seven;
        return ret;
    }
    factor& operator+=(const factor& other) {
        two += other.two;
        three += other.three;
        five += other.five;
        seven += other.seven;
        return *this;
    }
    factor& operator-=(const factor& other) {
        two -= other.two;
        three -= other.three;
        five -= other.five;
        seven -= other.seven;
        return *this;
    }
} PROD;
factor const CONST[10] = {
    {0, 0, 0, 0},  //  1
    {1, 0, 0, 0},  //  2
    {0, 1, 0, 0},  //  3
    {2, 0, 0, 0},  //  4
    {0, 0, 1, 0},  //  5
    {1, 1, 0, 0},  //  6
    {0, 0, 0, 1},  //  7
    {3, 0, 0, 0},  //  8
    {0, 2, 0, 0},  //  9
    {1, 0, 1, 0},  // 10
};
 
ll incoding(factor f) {
    return f.two + f.three*100 + f.five*10000 + f.seven*1000000;
}
 
factor decoding(ll code) {
    factor ret;
    ret.two = code%100; code /= 100;
    ret.three = code%100; code /= 100;
    ret.five = code%100; code /= 100;
    ret.seven = code%100;
    return ret;
}
 
ll real_val(factor f) {
    return (ll)pow(2, f.two) * (ll)pow(3, f.three) * (ll)pow(5, f.five) * (ll)pow(7, f.seven);
}
 
void trans(ll i, ll j, ll d, factor add) {
    //if (j + d > SUM) return;
    //if (j + d + 9*(n-i-2) < SUM) return;
    for (auto it = prod[i][j].begin(); it != prod[i][j].end(); it++) {
        prod[i+1][j+d][it->first + incoding(add)] += it->second;
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    string s;
    cin >> s;
    n = s.size();
    for (ll i = 0; i < n; i++) arr[i] = s[i] - '0';
 
    for (ll k = 0; k <= 9; k++) sum[0][k] = 1;
    for (ll i = 0; i < n-1; i++) {
        for (ll j = 0; j <= 9*i+9; j++) {
            for (ll k = 0; k <= 9; k++) {
                sum[i+1][j+k] += sum[i][j];
            }
        }
    }
    
    SUM = 0;
    for (ll i = 0; i < n; i++) SUM += arr[i];
    for (ll j = 0; j < SUM; j++) ans += sum[n-1][j];
 
    for (ll k = 0; k <= 9; k++) prod[0][k][incoding(CONST[k])] = 1;
    for (ll i = 0; i < n-1; i++) {
        for (ll j = 0; j <= 9*i+9; j++) {
            for (ll k = 0; k <= 9; k++) {
                trans(i, j, k, CONST[k]);
            }
        }
    }
 
    PROD = {0, 0, 0, 0};
    for (ll i = 0; i < n; i++) {
        PROD += CONST[arr[i]];
    }
 
    for (auto it = prod[n-1][SUM].begin(); it != prod[n-1][SUM].end(); it++) {
        if (real_val(decoding(it->first)) < real_val(PROD)) ans += it->second;
    }
 
    for (ll i = 0; i < n-1; i++) {
        for (ll k = 0; k < arr[i]; k++) {
            if (PROD.two < CONST[k].two || PROD.three < CONST[k].three ||
                PROD.five < CONST[k].five || PROD.seven < CONST[k].seven) continue;
            for (auto it = prod[n-2-i][SUM-k].begin(); it != prod[n-2-i][SUM-k].end(); it++) {
                if (real_val(decoding(it->first)) == real_val(PROD - CONST[k])) ans += it->second;
            }
        }
        PROD -= CONST[arr[i]];
        SUM -= arr[i];
    }
 
    cout << ans << "\n";
 
    return 0;
}
```

## 7. 최단경로

가중치가 정점 기준으로 계산되기 때문에 최적해의 형태는 다음을 만족해야만 한다.

- 최적해가 $v_1 \rightarrow v_2 \rightarrow \cdots \rightarrow v_k$일 때, 임의의 두 $i+1<j$인 두 $i, j$에 대해 $v_i$와 $v_j$를 연결하는 간선이 있으면 안된다.
    - $v_i \rightarrow v_{i+1} \rightarrow \cdots \rightarrow v_j$를 $v_i \rightarrow v_j$로 바꾸면 가중치 합이 더 작아지기 때문이다.

이를 활용해 분기 한정을 하면 시간 안에 문제가 해결된다.

이 알고리즘의 시간복잡도를 분석해보자. 이 알고리즘의 시간복잡도가 $T(n)$으로 두자. $v_1$의 차수가 $d$일 때 다음 상태 전이의 개수는 $d$이고 경로상에 있을 수 있는 정점은 $f(n-d)$개이므로 이러한 최적해의 개수의 상한은 $\max_{1 \leq d \leq n-1} df(n-d)$가 된다. 이 값은 $d = 3$에서 최대이므로 $T(n) \leq 3T(n-3)$이고 이를 풀면 $T(n) \leq 3^{N/3} \cdot T(1) = 3^{N/3} \cdot N$이다. 즉, 이 알고리즘의 최악 시간복잡도는 $O(3^{N/3} \cdot N)$이 되어 주어진 $N \leq 40$에서 시간 안에 작동한다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef pair<int, int> pii;
int n, m, w[44], ans = 1e9;
vector<int> adj[44];

vector<int> path;
int cnt[44];

void add(int v, int diff) {
    for (int i = 0; i < adj[v].size(); i++) {
        cnt[adj[v][i]] += diff;
    }
}

bool not_opt() {
    if (path.size() <= 2) return false;

    int ch[44];
    for (int i = 1; i <= n; i++) ch[i] = 0;
    for (int j = 0; j < adj[path.back()].size(); j++) {
        ch[adj[path.back()][j]] = 1;
    }
    for (int i = 0; i < path.size()-2; i++) {
        if (ch[path[i]] == 1) return true;
    }
    return false;
}

void solve(int v) {
    if (not_opt()) return;

    if (v == n) {
        int sum = 0;
        for (int i = 1; i <= n; i++) if (cnt[i] > 0) sum += w[i];
        ans = min(ans, sum);
        return;
    }

    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i];
        path.push_back(u);
        add(u, 1);
        solve(u);
        add(u, -1);
        path.pop_back();
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> w[i];
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    add(1, 1);
    path.push_back(1);
    solve(1);

    cout << ans << "\n";

    return 0;
}
```

## 8. 문자열 색칠하기

앞에 $N$개를 빨간색과 파란색으로 색칠한 문자열이 $S_R, S_B$이면 뒤에 $N$개를 색칠한 문자열은 각각 $S_B$를 뒤집은 문자열과 $S_R$을 뒤집은 문자열이어야 한다. 따라서 앞 절반과 뒤 절반의 $2^N$가지 경우의 수를 모두 저장하고 겹치는 경우의 수를 세면 된다. $S_R$과 $S_B$ 사이에 알파벳이 아닌 문자를 추가하고 `unordered_map`과 같은 구조로 관리하면 쉽게 구현할 수 있다(사이에 문자를 추가하지 않으면 “aa”와 같은 경우 왼쪽과 오른쪽 모두 빨간색으로 칠하는 경우 등이 세질 수 있다). 시간복잡도는 $O(N \cdot 2^N)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
int n;
string s;
map<string, int> lt;
long long ans = 0;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
    
    cin >> n;
    cin >> s;

    for (int i = 0; i < (1<<n); i++) {
        string t;
        for (int j = 0; j < n; j++) {
            if (i & (1<<j)) {
                t.push_back(s[j]);
            }
        }
        t.push_back('!');
        for (int j = 0; j < n; j++) {
            if (!(i & (1<<j))) {
                t.push_back(s[j]);
            }
        }
        lt[t]++;
    }

    for (int i = 0; i < (1<<n); i++) {
        string t;
        for (int j = 0; j < n; j++) {
            if (i & (1<<j)) {
                t.push_back(s[j+n]);
            }
        }
        t.push_back('!');
        for (int j = 0; j < n; j++) {
            if (!(i & (1<<j))) {
                t.push_back(s[j+n]);
            }
        }
        reverse(t.begin(), t.end());
        ans += lt[t];
    }
    cout << ans << "\n";
}
```
