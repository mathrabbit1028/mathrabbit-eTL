## 1. 피보나치 수열 (0:03:53)

$dp[i] = dp[i-1] + dp[i-2]$로 계산하면 $O(N)$에 계산할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n;
ll dp[100];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    dp[1] = dp[2] = 1;
    for (int i = 3; i <= n; i++) dp[i] = dp[i-1] + dp[i-2];
    cout << dp[n] << "\n";
 
    return 0;
}
```

## 2. 건포도 (0:14:20)

영역 $[s, e]$와 $[l, r]$의 부분 초콜릿을 자를 때 얻는 최소 민트초코 상자의 개수를 $dp[s][e][l][r]$로 정의하자. 이제 가로로 자르는 경우와 세로로 자르는 경우로 나누어 가능한 모든 자르는 경우를 시도해 각 상태별로 $O(N+M)$에 계산할 수 있다. 상태의 개수는 $O(N^2M^2)$이므로 전체 시간복잡도는 $O(N^2M^2(N+M))$이 되고, 상수가 작기 때문에 시간 내에 문제를 해결할 수 있다.

아래 코드는 구현의 편의를 위해 top-down 방식으로 구성되었음에 유의하라.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, m, arr[51][51], dp[51][51][51][51];
 
int sum(int s, int e, int l, int r) {
    return arr[e][r] - arr[s-1][r] - arr[e][l-1] + arr[s-1][l-1];
}
 
void solve(int s, int e, int l, int r) {
    if (dp[s][e][l][r] >= 0) return;
    if (s == e && l == r) {
        dp[s][e][l][r] = 0;
        return;
    }
 
    dp[s][e][l][r] = 1e9;
    for (int i = s; i < e; i++) {
        solve(s, i, l, r);
        solve(i+1, e, l, r);
        dp[s][e][l][r] = min(dp[s][e][l][r], dp[s][i][l][r] + dp[i+1][e][l][r]);
    }
    for (int i = l; i < r; i++) {
        solve(s, e, l, i);
        solve(s, e, i+1, r);
        dp[s][e][l][r] = min(dp[s][e][l][r], dp[s][e][l][i] + dp[s][e][i+1][r]);
    }
    dp[s][e][l][r] += sum(s, e, l, r);
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> m;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) cin >> arr[i][j];
        for (int j = 1; j <= m; j++) arr[i][j] = arr[i][j-1] + arr[i][j];
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) arr[i][j] = arr[i-1][j] + arr[i][j];
    }
 
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j++) {
            for (int k = 1; k <= m; k++) {
                for (int l = k; l <= m; l++) {
                    dp[i][j][k][l] = -1;
                }
            }
        }
    }
 
    solve(1, n, 1, m);
 
    cout << dp[1][n][1][m] << "\n";
 
    return 0;
}
```

## 3. 부분수열 세기 (0:22:13)

다음과 같은 관찰을 하자.

- $j<i$에서 $i$번째 문자와 $j$번째 문자가 같다면, $j-1$번 및 그 이전 문자들로만 이루어진 부분수열에 $i$번째 문자를 추가하는 것은 겹친다.

따라서 $dp[i]$를 $i$번째 문자를 포함하는 부분문자열의 개수로 정의하면, $i$번째 문자가 다시 등창하는 가장 $i$번째와 가까운 위치 $j(<i)$에 대해

$$
dp[i]= dp[j] + dp[j+1] + \cdots + dp[i-1] = S[i-1] - S[j-1]
$$

로 계산할 수 있다. 각 문자마다 위치를 배열로 저장하면 이분탐색으로 $O(\log N)$에 $j$를 구할 수 있고, $j$를 구하면 누적합으로 $O(1)$에 $dp[i]$를 계산할 수 있으므로 전체 시간복잡도는 $O(N \log N)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
const int MOD = 10'007;
int n, dp[101010], sum[101010];
string s;
vector<int> idx[26];
 
int get(int s, int e) {
    if (s == 0) return sum[e];
    else return (sum[e] + MOD - sum[s-1]) % MOD;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> s;
    n = s.length();
 
    for (int i = 0; i < n; i++) {
        idx[s[i]-'a'].push_back(i);
    }
 
    dp[0] = 1;
    sum[0] = 1;
    for (int i = 1; i < n; i++) {
        auto iter = lower_bound(idx[s[i]-'a'].begin(), idx[s[i]-'a'].end(), i);
        if (iter == idx[s[i]-'a'].begin()) dp[i] = get(0, i-1) + 1;
        else dp[i] = get(*--iter, i-1);
        sum[i] = (sum[i-1] + dp[i]) % MOD;
    }
 
    cout << sum[n-1] << "\n";
 
    return 0;
}
```

## 4. 늦잠 (0:47:57)

$(x_i, y_i)$에 있는 학생을 깨운 직후 문정후 코치의 위치는 $(x_i, y)$ 또는 $(x, y_i)$이다. 따라서 좌표의 범위를 $X$로 두면 매 학생을 깨우는 순간마다 문정후 코치가 있을 수 있는 위치는 $O(X)$개이므로 각 $i$에 대해 최대 $O(NX)$개의 상태만 고려하여도 된다.

문정후 코치가 $(x, y_i)$에 도달하여 $i$번째 학생을 깨울 때 $i$번째 학생까지 깨우기 위해 필요한 최소 이동 거리를 $dp_1[i][x]$로 두자. 유사하게 $dp_2[i][y]$를 잡자. 그러면

$$
dp_1 [i][x] \gets dp_1[i-1][x^\prime]+ |x-x^\prime|+ |y_i - y_{i-1}| \\ dp_1 [i][x] \gets dp_2[i-1][y^\prime]+ |x-x_{i-1}|+ |y_i - y^\prime|
$$

의 전이가 나오게 된다($dp_2$는 대칭적으로 쉽게 식을 유도 가능하다). 그런데 만약 이를 모든 가능한 $x^\prime$과 $y^\prime$에 대해 시행한다면 각 상태를 계산하는데 $O(X)$의 시간이 걸리므로 $O(NX^2)$으로 TLE를 받게 된다.

이떄 주어진 형태의 식에서 $x > x^\prime$에서 $dp_1[i-1][x^\prime]-x^\prime$을 빠르게 구할 수 있고, $x<x^\prime$에서 $dp_1[i-1][x^\prime]+x^\prime$를 빠르게 구할 수 있다면 전체 식을 빠르게 구할 수 있을 것이다. 이를 위해 작은 $x$부터 $dp_1[i][x]-x$의 누적 $\min$과 큰 $x$부터 $dp_1[i][x]+x$의 누적 $\min$을 전처리하자. 그러면 각 상태를 $O(1)$에 처리할 수 있고, 이는 네 가지 경우 모두 마찬가지이다.

이 아이디어로 코드를 작성하면 $O(NX)$의 시간복잡도로 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, dp[20202][2020][2], pfx[20202][2020][2], sfx[20202][2020][2];
int pos[20202][2];
 
void make_table(int i) {
    pfx[i][-1000+1000][0] = dp[i][-1000+1000][0] - (-1000);
    for (int x = -999; x <= 1000; x++) {
        pfx[i][x+1000][0] = min(pfx[i][x+1000-1][0], dp[i][x+1000][0] - x);
    }
    pfx[i][-1000+1000][1] = dp[i][-1000+1000][1] - (-1000);
    for (int y = -999; y <= 1000; y++) {
        pfx[i][y+1000][1] = min(pfx[i][y+1000-1][1], dp[i][y+1000][1] - y);
    }
    sfx[i][+1000+1000][0] = dp[i][+1000+1000][0] + (+1000);
    for (int x = +999; x >= -1000; x--) {
        sfx[i][x+1000][0] = min(sfx[i][x+1000+1][0], dp[i][x+1000][0] + x);
    }
    sfx[i][+1000+1000][1] = dp[i][+1000+1000][1] + (+1000);
    for (int y = +999; y >= -1000; y--) {
        sfx[i][y+1000][1] = min(sfx[i][y+1000+1][1], dp[i][y+1000][1] + y);
    }
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> pos[i][0] >> pos[i][1];
 
    for (int x = -1000; x <= 1000; x++) {
        dp[0][x+1000][0] = 1e9;
    }
    for (int y = -1000; y <= 1000; y++) {
        dp[0][y+1000][1] = 1e9;
    }
    dp[0][0+1000][0] = 0;
    dp[0][0+1000][1] = 0;
    make_table(0);
 
    for (int i = 1; i <= n; i++) {
        for (int x = -1000; x <= 1000; x++) {
            dp[i][x+1000][0] = 1e9;
            dp[i][x+1000][0] = min(dp[i][x+1000][0], pfx[i-1][x+1000][0] + x + abs(pos[i][1] - pos[i-1][1]));
            dp[i][x+1000][0] = min(dp[i][x+1000][0], sfx[i-1][x+1000][0] - x + abs(pos[i][1] - pos[i-1][1]));
            dp[i][x+1000][0] = min(dp[i][x+1000][0], pfx[i-1][pos[i][1]+1000][1] + pos[i][1] + abs(x - pos[i-1][0]));
            dp[i][x+1000][0] = min(dp[i][x+1000][0], sfx[i-1][pos[i][1]+1000][1] - pos[i][1] + abs(x - pos[i-1][0]));
        }
        for (int y = -1000; y <= 1000; y++) {
            dp[i][y+1000][1] = 1e9;
            dp[i][y+1000][1] = min(dp[i][y+1000][1], pfx[i-1][pos[i][0]+1000][0] + pos[i][0] + abs(y - pos[i-1][1]));
            dp[i][y+1000][1] = min(dp[i][y+1000][1], sfx[i-1][pos[i][0]+1000][0] - pos[i][0] + abs(y - pos[i-1][1]));
            dp[i][y+1000][1] = min(dp[i][y+1000][1], pfx[i-1][y+1000][1] + y + abs(pos[i][0] - pos[i-1][0]));
            dp[i][y+1000][1] = min(dp[i][y+1000][1], sfx[i-1][y+1000][1] - y + abs(pos[i][0] - pos[i-1][0]));
        }
        make_table(i);
    }
 
    int ans = 1e9;
    for (int x = -1000; x <= 1000; x++) ans = min(ans, dp[n][x+1000][0]);
    for (int y = -1000; y <= 1000; y++) ans = min(ans, dp[n][y+1000][1]);
    cout << ans << "\n";
 
    return 0;
}
```

## 5. 쇼킹핫치킨 (0:56:48)

$dp[i][j]$를 $i$개의 음식으로 $j<A$의 나트륨 양을 만들었을 때 섭취하는 최소 열량으로 정의하자. 그러면 각 $N$개의 음식에 대해 $O(NX)$개의 상태를 전이해야 하므로 $O(N^2X)$의 시간복잡도에 dp 테이블을 채울 수 있다.

답은 섭취하는 열량이 $B$ 이하인 $dp[i][j]$가 존재하는 $i$의 최댓값이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, A, B;
int arr[110][2];
int dp[110][5050];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> A >> B;
    for (int i = 1; i <= n; i++) cin >> arr[i][0] >> arr[i][1];
 
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= A; j++) dp[i][j] = 1e9;
    }
    dp[0][0] = 0;
 
    for (int t = 1; t <= n; t++) {
        for (int i = n; i >= 1; i--) {
            for (int j = A; j >= arr[t][0]; j--) {
                dp[i][j] = min(dp[i][j], dp[i-1][j-arr[t][0]] + arr[t][1]);
            }
        }
    }
 
    for (int i = n; i >= 0; i--) {
        for (int j = 0; j <= A; j++) {
            if (dp[i][j] <= B) {
                cout << i << "\n";
                return 0;
            }
        }
    }
 
    return 0;
}
```

## 6. 대탈출 (2:00:36)

다음을 관찰해야 한다.

- 앞에서부터 연속해서 두 배열을 보면서 같은 개수가 되었을 때 그룹화하면, 그룹끼리 상호작용하는 경우 연산 횟수가 최소가 아니다.
    - 예를 들어 1 2 1 / 2 1 1 배열을 생각하자.
    - 앞에서부터 보면 (1 2) (1) / (2 1) (1)이 되는데, 다른 그룹의 1끼리 합치는 것 등을 하면 연산 횟수가 최소가 아니다.
- 그러면 가능한 그룹의 형태는 다음 네 개 중 하나이다.
    - 1 / 1 또는 2 / 2
    - 1 2 2 … 2 / 2 2 … 2 1
    - 1 2 2 … 2 1 / 2 2 … 2 2
    - 2 2 … 2 2 / 1 2 2 … 2 1
- 2~4번째 그룹의 경우, 그룹의 길이 $L$에 대해 $L-1$이 필요한 연산 횟수이다.

각 그룹이 독립적이므로 모든 $L_i$에 대해 $L_i-1$개씩 있는 같은 것이 있는 순열을 배열하는 경우의 수를 곱하면 각 그룹의 순서가 결정되었을 때 전체 순서의 경우의 수를 알 수 있다. 이제 각 그룹에서 순서의 경우의 수를 세자.

- 다음 세 경우에서 처음 배열의 $2$의 개수 $n$에 대해 처음 $x+1$번째 $2$를 $1$ 두개로 나눈다면,
    - 1 2 2 … 2 / 2 2 … 2 1 : $p_n = \sum_{x=0}^{n-1} {q_x \cdot p_{n-x-1} \cdot \binom{2n-1}{2x+1}}$
    - 1 2 2 … 2 1 / 2 2 … 2 2 : $q_n = \sum_{x=0}^{n-1} {q_x \cdot q_{n-x-1} \cdot \binom{2n}{2x+1}}$
    - 2 2 … 2 2 / 1 2 2 … 2 1 : $r_n = \sum_{x=0}^{n-1} {p_x \cdot p_{n-x-1} \cdot \binom{2n-2}{2x}}$
    - 이 점화식은 $O(L^2)$에 계산할 수 있다.

따라서 전체 문제를 $O(L^2)$에 해결할 수 있다.

* (다른 풀이) 경계를 1, 연결된 칸을 0로 보자. 그러면 연속된 0이 없도록 하면서 1을 0으로 바꾸거나 0을 1로 바꾸어 배열을 같게 만들어야 하는 문제로 환원된다. 그러면 최소 횟수는 0을 모두 1로 바꾸고 1을 0으로 바꾸면 되므로 다른 인덱스의 개수가 최소 횟수가 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int L, n, m;
int bef[3030], tar[3030];
vector<int> vec;
 
const ll MOD = 1'000'000'007;
ll fac[3030], inv[3030], P[3030], Q[3030], R[3030];
 
ll mypow(ll a, int x) {
    if (x == 0) return 1LL;
    ll ret = mypow(a, x/2);
    ret = (ret * ret) % MOD;
    if (x%2 == 0) return ret;
    if (x%2 == 1) return (ret * a) % MOD;
}
 
ll C(int N, int R) {
    ll ret = fac[N];
    ret = (ret * inv[R]) % MOD;
    ret = (ret * inv[N-R]) % MOD;
    return ret;
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> L;
    fac[0] = 1;
    for (int i = 1; i <= L; i++) fac[i] = (fac[i-1] * i) % MOD;
    for (int i = 0; i <= L; i++) inv[i] = mypow(fac[i], MOD-2);
    P[0] = Q[0] = R[0] = 1;
    for (int i = 1; i <= L; i++) {
        for (int j = 0; j < i; j++) {
            P[i] += (Q[j] * P[i-j-1] % MOD) * C(2*i-1, 2*j+1) % MOD;
            P[i] %= MOD;
            Q[i] += (Q[j] * Q[i-j-1] % MOD) * C(2*i, 2*j+1) % MOD;
            Q[i] %= MOD;
            R[i] += (P[j] * P[i-j-1] % MOD) * C(2*i-2, 2*j) % MOD;
            R[i] %= MOD;
        }
    }
 
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> bef[i];
    cin >> m;
    for (int i = 1; i <= m; i++) cin >> tar[i];
    
    int p = 1, q = 1, a = 0, b = 0, t = 0, s1 = 0, s2 = 0;
    ll cnt = 1;
    while (p <= n || q <= m) {
        if (a >= b) b += tar[q++], t++, s2 += (tar[q-1]==2);
        else a += bef[p++], t++, s1 += (bef[p-1]==2);
        if (a == b) {
            if (t > 2) vec.push_back(a-1);
            if (s1 == s2) cnt = (cnt * P[s1]) % MOD;
            if (s1 < s2) cnt = (cnt * Q[s1]) % MOD;
            if (s1 > s2) cnt = (cnt * R[s1]) % MOD;
            a = 0; b = 0; t = 0; s1 = 0; s2 = 0;
        }
    }
 
    int tot = 0;
    for (int i = 0; i < vec.size(); i++) tot += vec[i];
    cnt = (cnt * fac[tot]) % MOD;
    for (int i = 0; i < vec.size(); i++) {
        cnt = (cnt * inv[vec[i]]) % MOD;
    }
 
    cout << tot << " " << cnt << "\n";
 
    return 0;
}
```

## 7. 부동산 (2:35:52)

https://www.acmicpc.net/problem/6171

$w_i \geq w_j$이며 $h_i \geq h_j$인 $j$번 부동산은 고려하지 않아도 됨을 이용하면 모든 $i < j$에 대해 $w_i < w_j$이고 $h_i > h_j$가 되도록 잘 남기고 정렬할 수 있다. 그러면 최적해 묶음은 무조건 연속된 구간 $[l, r]$로 나타나게 된다.

이제 $dp[i]$를 $i$번 부동산까지 고려했을 때 필요한 최소 비용으로 정의하자. 그러면 모든 $1 \leq j \leq i$에 대해

$$
dp[i] \gets dp[j-1] + h[j] \times w[i]
$$

이고 $h[j]$는 감소하므로 CHT를 활용하여 $O(N \log N)$에 문제를 해결할 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

#define w first
#define h second
int n;
pll arr[50505];
vector<pll> vec;
ll dp[50505];
 
bool compare(pll p, pll q) {
    if (p.w == q.w) return p.h > q.h;
    else return p.w > q.w;
}
 
struct Line { // ax + b, x >= s
    ll a, b;
    double s;
};
double intersect(Line f, Line g) {
    return (g.b - f.b) / (f.a - g.a);
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> arr[i].w >> arr[i].h;
    }
    sort(arr, arr + n, compare);
 
    ll mx = 0;
    for (int i = 0; i < n; i++) {
        if (mx < arr[i].h) vec.push_back(arr[i]);
        mx = max(mx, arr[i].h);
    }
    reverse(vec.begin(), vec.end());
 
    int n = vec.size();
    for (int i = 1; i <= n; i++) arr[i] = vec[i-1];
    for (int i = 1; i < n; i++) assert(arr[i].h > arr[i+1].h);
 
    // dp[i] <- dp[j-1] + h[j] * w[i] for all 1 <= j <= i
 
    vector<Line> lines(n);
    dp[0] = 0;
    int last = -1;
    for (int i = 1; i <= n; i++) {
        Line now = {arr[i].h, dp[i-1], 0};
        while (last >= 0) {
            now.s = intersect(lines[last], now);
            if (lines[last].s < now.s) break;
            last--;
        }
        lines[++last] = now;
 
        int st = 0, ed = last;
        while (st < ed) {
            int mid = (st + ed + 1) / 2;
            if (arr[i].w < lines[mid].s) ed = mid-1;
            else st = mid;
        }
        dp[i] = lines[st].a * arr[i].w + lines[st].b;
    }
 
    cout << dp[n] << "\n";
 
    return 0;
}
```

## 9. 딧불이의 생일선물 (3:49:03)

먼저 다음을 관찰하자.

- 어떤 서브트리를 뒤집더라도 서브트리 안의 반발력은 최소이다.

이제 주어진 트리가 포화이진트리이므로 절반씩 나누며 문제를 해결해보자. 전체 구간에 대해 절반을 나누게 되면 두 구간을 합칠 때 중요하게 작용하는 것은 경계에 있는 두 리프의 번호이다. 즉, 이 값을 추가 인자로 넣자. 즉, 다음과 같이 dp를 정의하자.

- $dp[st][ed][lt][rt]$: 구간 $[st, ed]$를 나타내는 서브트리의 모빌에서 제일 왼쪽이 $lt$이고 제일 오른쪽이 $rt$일 때 가능한 최소 반발력
    - 중요한 것은 뒤집더라도 반발력이 최소이기 때문에 $lt$는 번호가 작은 서브트리($[st, mid]$)에 속해야 하고, $rt$는 번호가 큰 서브트리($[mid+1, ed]$)에 속해야 한다.
    - $lt$와 $rt$는 다른 자식 트리에 속해야 한다. 만약 같은 자식 트리에 속한다면 유효하지 않은 것이다.
- 그러면 $[st, ed]$의 가운데 $mid$에 대해 $[st, mid]$에서 $lt$ 하나를 고르고, $[mid+1, ed]$에서 $rt$ 하나를 고르자. 이제 가능한 $O(N^2)$가지 경우를 모두 시도하면 된다.

단순히 보았을 때 상태가 $O(N^4)$개가 되어 쓸 수 없는 풀이 같지만, $(st, ed)$ 쌍 대신 트리의 깊이만 주더라도 $lt$와 $rt$를 포함해야 하므로 $(st, ed)$가 결정된다. 따라서 $O(KN^2)$개의 상태만 고려하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int k, n;
int arr[550][550];
int dp[10][550][550];
 
int solve(int L, int st, int ed, int lt, int rt) {
    if (lt != 0 && lt > (st + ed)/2) swap(lt, rt);
    if (rt != 0 && rt <= (st + ed)/2) swap(lt, rt);
    //printf("%d %d %d %d %d\n", L, st, ed, lt, rt);
 
    if (L == 1) {
        dp[L][lt][rt] = arr[st][ed];
        return arr[st][ed];
    }
 
    if (dp[L][lt][rt] >= 0) return dp[L][lt][rt];
    dp[L][lt][rt] = 1e9;
 
    int mid = (st + ed) / 2;
    for (int i = st; i <= mid; i++) {
        if (lt != 0 && !((lt <= (st + mid)/2)^(i <= (st+mid)/2))) continue;
        //if (lt != 0) cout << st << " " << mid << " " << i << " " << lt << "\n";
        for (int j = mid+1; j <= ed; j++) {
            if (rt != 0 && !((rt <= (mid+1 + ed)/2)^(j <= (mid+1 + ed)/2))) continue;            
            dp[L][lt][rt] = min(dp[L][lt][rt], solve(L-1, st, mid, lt, i) + solve(L-1, mid+1, ed, j, rt) + arr[i][j]);
        }
    }
 
    return dp[L][lt][rt];
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> k;
    n = (1<<k);
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            cin >> arr[i][j];
        }
    }
 
    for (int t = 1; t <= k; t++) {
        for (int i = 0; i <= n; i++) {
            for (int j = 0; j <= n; j++) {
                dp[t][i][j] = -1;
            }
        }
    }
 
    cout << solve(k, 1, n, 0, 0) << "\n";
 
    return 0;
}
```

## 8. 이성호

제곱 dp를 먼저 짜보자 - $O(Q^2)$

$dp[i][j]$: 한 명은 $i$번 위치에 있고, 한 명은 $j$번 위치에 있을 때 필요한 비용 $i>j$

$dp[i+1][j] \gets dp[i][j] + |x_{i+1} - x_i|$ : $i$번 행의 구간 전체에 특정 값 더하기 - lazy propagation

$dp[i+1][i] \gets \min(dp[i][j]+|x_{i+1}-x_j|)$ : 절댓값 크기 따라 나눠서 보면 $dp[i][j] \pm x_j$가 나오는데 이 값을 세그에 저장해서 최솟값 얻기

$O(Q \log (N+Q))$

## 12. 독재

https://www.acmicpc.net/problem/17018

`H`에 -1점, `G`에 1점을 할당하면 구간의 점수 합이 $0$ 이상인 선거구에서 *국민의 C*가 당선된다. 따라서 이렇게 부여한 점수로 누적합 배열 $S_i$를 만들자. 그러면

- 모든 $i-k \leq j < i$에 대해 $dp[i] \gets dp[j] + [S_i \geq S_j]$ (아이버슨 괄호)

먼저 50점 풀이, $S_i$를 인덱스로 하여 세그먼트 트리를 만들고 각 노드에 $S_i \in [s, e]$인 $i$에 대해 $dp[i]$ 목록을 저장하자. 그러면 $dp[i]$는 $S_i < S_j$인 $j$들 중 $dp[j]$의 최솟값을 찾고, $S_i \geq S_j$인 $j$들 중 $dp[j]$의 최솟값을 $O(\log N)$에 찾아 구할 수 있고, $i$번째 값 추가는 $S_i$에 $dp[i]$를 추가함으로써 할 수 있다. 마지막으로 길이 $k$ 이하 조건은 $S_{i-k}$에서 $dp[i-k]$에 해당하는 값을 제거함으로써 만족시킬 수 있다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

int n, k;
int arr[303030];
int sum[303030];
int dp[303030];

map<int, int> seg[2020202];
void add(int v, int st, int ed, int idx, int val) {
    if (st > idx || ed < idx) return;
    seg[v][val]++;
    if (st == idx && ed == idx) return;
    int mid = (st + ed) / 2;
    add(2 * v, st, mid, idx, val);
    add(2 * v + 1, mid + 1, ed, idx, val);
}
void rem(int v, int st, int ed, int idx, int val) {
    if (st > idx || ed < idx) return;
    seg[v][val]--;
    if (seg[v][val] == 0) seg[v].erase(val);
    if (st == idx && ed == idx) return;
    int mid = (st + ed) / 2;
    rem(2 * v, st, mid, idx, val);
    rem(2 * v + 1, mid + 1, ed, idx, val);
}

int get(int v, int st, int ed, int lt, int rt) {
    if (st > rt || lt > ed) return 1e9;
    if (lt <= st && ed <= rt) return (seg[v].size() > 0 ? seg[v].begin()->first : 1e9);
    int mid = (st + ed) / 2;
    return min(get(2 * v, st, mid, lt, rt), get(2 * v + 1, mid + 1, ed, lt, rt));
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        char c; cin >> c;
        arr[i] = (c == 'H' ? -1 : 1);
        sum[i] = sum[i-1] + arr[i];
    }

    dp[0] = 0;
    add(1, 0, 2*n, n+sum[0], dp[0]);
    for (int i = 1; i <= n; i++) {
        dp[i] = 1e9;
        dp[i] = min(dp[i], get(1, 0, 2*n, 0, n+sum[i]) + 1);
        dp[i] = min(dp[i], get(1, 0, 2*n, n+sum[i]+1, 2*n));
        add(1, 0, 2*n, n+sum[i], dp[i]);
        if (i >= k) rem(1, 0, 2*n, n+sum[i-k], dp[i-k]);
    }

    cout << dp[n] << "\n";

    return 0;
}
```

100점 풀이는 $O(N)$에 해결해야 하는데, 덱에 $dp[i], S_i, i$를 넣고 $dp[i]$가 작을수록 앞에 오도록, 만약 같다면, $S_i$가 큰 것이 앞에 오도록 저장하면 된다. 그러면 가장 앞에 있는 원소 $dp[j], S_j, j$에 대해서만 dp 식을 해도 답이 나옴을 보장할 수 있다. 왜냐하면

- $S_j > S_i$라고 하자. $dp[i] \gets dp[j]$이므로 자명히 $j$만 고려해도 된다.
- $S_j \leq S_i$이면 $dp[i] \gets dp[j]+1$이므로 다른 모든 경우에 대해 $dp[k] + [S_i \geq S_k] \geq dp[j]+1$임을 보이면 된다.
    - $dp[j] = dp[k]$인 다른 모든 덱에 포함된 쌍에 대해 $S_k \leq S_i$일 것이므로 $dp[k] + [S_i \geq S_k] =dp[j]+1$
    - $dp[j] < dp[k]$인 다른 모든 덱에 포함된 쌍에 대해 $dp[k] + [S_i \geq S_k] \geq dp[k] \geq dp[j]+1$

만약 제일 앞에 있는 원소가 $i-j > K$를 만족한다면 제거하고 다음 원소를 보면 된다. 따라서 $O(N)$에 문제를 해결할 수 있다.

https://tamref.com/143

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, k;
int arr[10101010];
int sum[10101010];
int dp[10101010];
deque<int> dq;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> k;
    for (int i = 1; i <= n; i++) {
        char c; cin >> c;
        arr[i] = (c == 'H' ? -1 : 1);
        sum[i] = sum[i-1] + arr[i];
    }
 
    dp[0] = 0;
    dq.push_back(0);
    for (int i = 1; i <= n; i++) {
        while (dq.front() < i-k) dq.pop_front();
        dp[i] = dp[dq.front()] + (sum[i] >= sum[dq.front()]);
        while (dp[dq.back()] > dp[i] || (dp[dq.back()] == dp[i] && sum[dq.back()] <= sum[i])) dq.pop_back();
        dq.push_back(i);
    }
 
    cout << dp[n] << "\n";
 
    return 0;
}
```
