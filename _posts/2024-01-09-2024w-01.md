## 1. 세그먼트 트리 테스트 문제 (0:14:36)

말 그대로 테스트 문제

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n, q, arr[252525];
 
struct segtree {
    ll sum[1010101], mn[1010101];
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            sum[v] = arr[st];
            mn[v] = arr[st];
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mn[v] = min(mn[2 * v], mn[2 * v + 1]);
    }
 
    void update(int v, int st, int ed, int idx, ll val) {
        if (st == idx && ed == idx) {
            sum[v] += val - arr[st];
            mn[v] = val;
            return;
        }
        if (idx < st || idx > ed) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, idx, val);
        update(2 * v + 1, mid + 1, ed, idx, val);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mn[v] = min(mn[2 * v], mn[2 * v + 1]);
    }
 
    pll get(int v, int st, int ed, int lt, int rt) {
        if (lt <= st && ed <= rt) return {sum[v], mn[v]};
        if (lt > ed || st > rt) return {0, 1e9};
        int mid = (st + ed) / 2;
        pll lt_val = get(2 * v, st, mid, lt, rt);
        pll rt_val = get(2 * v + 1, mid + 1, ed, lt, rt);
        return {lt_val.first + rt_val.first, min(lt_val.second, rt_val.second)};
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> q;
 
    for (int i = 1; i <= n; i++) cin >> arr[i];
 
    seg.init(1, 1, n);
    for (int i = 1; i <= q; i++) {
        int t;
        cin >> t;
        if (t == 1) {
            int idx, val;
            cin >> idx >> val;
            seg.update(1, 1, n, idx, val);
            arr[idx] = val;
        }
        if (t == 2) {
            int lt, rt;
            cin >> lt >> rt;
            cout << seg.get(1, 1, n, lt, rt).second << "\n";
        }
        if (t == 3) {
            int lt, rt;
            cin >> lt >> rt;
            cout << seg.get(1, 1, n, lt, rt).first << "\n";
        }
    }
 
    return 0;
}
```

## 2. 2차원 세그먼트 트리 테스트 문제 (0:28:32)

2차원 세그트리 구현하기. 동적도 아니므로 포인터를 쓸 필요도 없다

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n, q;
 
struct segtree {
    ll sum[2020][2020];
 
    void update(int x, int y, ll num) {
        int i = x;
        while (i <= n) {
            int j = y;
            while (j <= n) {
                sum[i][j] += num;
                j += (j & -j);
            }
            i += (i & -i);
        }
    }
 
    ll get(int x, int y) {
        ll ret = 0;
        int i = x;
        while (i > 0) {
            int j = y;
            while (j > 0) {
                ret += sum[i][j];
                j -= (j & -j);
            }
            i -= (i & -i);
        }
        return ret;
    }
 
    ll query(int x1, int y1, int x2, int y2) {
        return get(x2, y2) - get(x1-1, y2) - get(x2, y1-1) + get(x1-1, y1-1);
    }
    
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> q;
 
    for (int i = 1; i <= q; i++) {
        int t;
        cin >> t;
        if (t == 1) {
            int x, y, v;
            cin >> x >> y >> v;
            seg.update(x, y, v);
        }
        if (t == 2) {
            int x1, y1, x2, y2;
            cin >> x1 >> y1 >> x2 >> y2;
            cout << seg.query(x1, y1, x2, y2) << "\n";
        }
    }
 
    return 0;
}
```

## 3. 스위치 (0:34:20)

lazy propagation 문제. 구간 반전시 구간 길이에서 현재 개수을 뺀 것이 새로운 개수가 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n, m, arr[101010];
 
struct segtree {
    ll sum[404040], lazy[404040];
 
    void prop(int v, int st, int ed) {
        if (lazy[v] == 1) sum[v] = ed - st + 1 - sum[v];
        if (st != ed) {
            lazy[2 * v] = lazy[2 * v] ^ lazy[v];
            lazy[2 * v + 1] = lazy[2 * v + 1] ^ lazy[v];
        }
        lazy[v] = 0;
    }
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            sum[v] = lazy[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
    }
 
    void update(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) {
            lazy[v] = 1;
            prop(v, st, ed);
            return;
        }
        if (lt > ed || st > rt) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, lt, rt);
        update(2 * v + 1, mid + 1, ed, lt, rt);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
    }
 
    int get(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) return sum[v];
        if (lt > ed || st > rt) return 0;
        int mid = (st + ed) / 2;
        return get(2 * v, st, mid, lt, rt) + get(2 * v + 1, mid + 1, ed, lt, rt);
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> m;
 
    for (int i = 1; i <= m; i++) {
        int t;
        cin >> t;
        if (t == 0) {
            int lt, rt;
            cin >> lt >> rt;
            seg.update(1, 1, n, lt, rt);
        }
        if (t == 1) {
            int lt, rt;
            cin >> lt >> rt;
            cout << seg.get(1, 1, n, lt, rt) << "\n";
        }
    }
 
    return 0;
}
```

## 4. 구간 합 (0:47:03)

$b_i$가 전부 다르므로 두 구간의 합을 비교할 때 구간에 속한 값의 가장 큰 값부터 비교하여 문자열 비교하듯이 비교하면 합을 비교할 수 있다. 여기서 더 나아가면 두 구간이 겹치지 않을 때 최댓값이 분명히 다를 것이므로 최댓값이 큰 쪽이 더 구간 합이 크다는 사실을 파악할 수 있다.

이를 바탕으로 비교 기준을 세워 쿼리를 정렬하자. 두 구간에 대해 겹치는 부분을 제외하고 나머지 부분에 있는 수의 최댓값이 큰 쪽이 더 구간 합이 크다. 같은 경우에는 순서가 유의미하지 않으므로 적당히 처리하면 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n, q, arr[101010];
 
struct segtree {
    ll mx[404040];
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            mx[v] = arr[st];
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        mx[v] = max(mx[2 * v], mx[2 * v + 1]);
    }
 
    int get(int v, int st, int ed, int lt, int rt) {
        if (lt <= st && ed <= rt) return mx[v];
        if (lt > ed || st > rt) return 0;
        int mid = (st + ed) / 2;
        return max(get(2 * v, st, mid, lt, rt), get(2 * v + 1, mid + 1, ed, lt, rt));
    }
} seg;
 
struct qry {
    int l, r, idx;
    bool operator<(const qry& other) const {
        if (r < other.l || other.r < l)
            return seg.get(1, 1, n, l, r) < seg.get(1, 1, n, other.l, other.r);
        if (l <= other.l && other.r <= r) return false;
        if (other.l <= l && r <= other.r) return true;
        if (other.l <= l)
            return seg.get(1, 1, n, other.r+1, r) < seg.get(1, 1, n, other.l, l-1);
        if (l <= other.l) 
            return seg.get(1, 1, n, l, other.l-1) < seg.get(1, 1, n, r+1, other.r);
    }
} qrys[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
 
    for (int i = 1; i <= n; i++) cin >> arr[i];
 
    seg.init(1, 1, n);
 
    cin >> q;
 
    for (int i = 1; i <= q; i++) {
        cin >> qrys[i].l >> qrys[i].r;
        qrys[i].idx = i;
    }
 
    sort(qrys + 1, qrys + q + 1);
 
    for (int i = 1; i <= q; i++) cout << qrys[i].idx << " ";
    cout << "\n";
 
    return 0;
}
```

## 5. 직사각형 (1:28:55)

https://www.acmicpc.net/problem/7626

화성 지도 세그를 구현하면 된다. 일단 스위핑을 통해 직사각형의 왼쪽 끝에 닿으면 높이 범위 $[y_1, y_2]$를 추가하고 오른쪽 끝에 닿으면 그 높이 범위를 세그에서 제거할 것이다. 그리고 세그먼트 트리는 겹치는 선분을 처리하기 위해

- range update를 처리하기 위해 $[st, ed]$를 모두 덮는 직사각형의 개수를 관리하는 세그먼트 트리와
- 실제 답을 구하기 위해 $[st, ed]$에서 덮인 구간의 길이를 관리하는 세그먼트 트리

를 구현하면 된다.

$y$축 값의 범위가 넓으므로 좌표압축을 사용하자. 그리고 $[y_1, y_2]$의 구간이 세그먼트 트리의 한 정점임에 유의하자.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n;
struct rect {
    int x1, x2, y1, y2;
} rects1[202020], rects2[202020];
 
bool comp1(rect a, rect b) {
    return a.x1 < b.x1;
}
bool comp2(rect a, rect b) {
    return a.x2 < b.x2;
}
 
vector<int> xlist, ylist;
 
ll ans = 0;
 
struct segtree {
    ll covered[1212121], length[1212121];
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            covered[v] = length[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
 
        if (covered[v] > 0) length[v] = ylist[ed] - ylist[st];
        if (covered[v] == 0) {
            if (st != ed) length[v] = length[2 * v] + length[2 * v + 1];
            else length[v] = 0;
        }
    }
 
    void update(int v, int st, int ed, int lt, int rt, int diff) {
        if (lt > ed || rt < st) return;
        if (lt <= st && ed <= rt) {
            covered[v] += diff;
        }
        else {
            int mid = (st + ed) / 2;
            update(2 * v, st, mid, lt, rt, diff);
            update(2 * v + 1, mid + 1, ed, lt, rt, diff);
        }
        if (covered[v] > 0) length[v] = ylist[ed] - ylist[st-1];
        if (covered[v] == 0) {
            if (st != ed) length[v] = length[2 * v] + length[2 * v + 1];
            else length[v] = 0;
        }
    }
 
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
 
    for (int i = 1; i <= n; i++) {
        cin >> rects1[i].x1 >> rects1[i].x2 >> rects1[i].y1 >> rects1[i].y2;
        rects2[i] = rects1[i];
    }
 
    sort(rects1 + 1, rects1 + n + 1, comp1);
    sort(rects2 + 1, rects2 + n + 1, comp2);
 
    for (int i = 1; i <= n; i++) {
        xlist.push_back(rects1[i].x1);
        xlist.push_back(rects1[i].x2);
        ylist.push_back(rects1[i].y1);
        ylist.push_back(rects1[i].y2);
    }
 
    sort(xlist.begin(), xlist.end());
    xlist.erase(unique(xlist.begin(), xlist.end()), xlist.end());
    sort(ylist.begin(), ylist.end());
    ylist.erase(unique(ylist.begin(), ylist.end()), ylist.end());
    int m = ylist.size()-1;
 
    for (int i = 1; i <= n; i++) {
        rects1[i].y1 = lower_bound(ylist.begin(), ylist.end(), rects1[i].y1) - ylist.begin();
        rects1[i].y2 = lower_bound(ylist.begin(), ylist.end(), rects1[i].y2) - ylist.begin();
        rects2[i].y1 = lower_bound(ylist.begin(), ylist.end(), rects2[i].y1) - ylist.begin();
        rects2[i].y2 = lower_bound(ylist.begin(), ylist.end(), rects2[i].y2) - ylist.begin();
    }
 
    int p = 1, q = 1;
    for (int i = 0; i < xlist.size(); i++) {
        while (p <= n && rects1[p].x1 < xlist[i]) {
            seg.update(1, 1, m, rects1[p].y1+1, rects1[p].y2, 1);
            p++;
        }
        if (i > 0) ans += (xlist[i] - xlist[i-1]) * seg.length[1];
        while (q <= n && rects2[q].x2 <= xlist[i]) {
            seg.update(1, 1, m, rects2[q].y1+1, rects2[q].y2, -1);
            q++;
        }
    }
 
    cout << ans << "\n";
 
    return 0;
}
```

## 6. 별 헤는 밤 (2:39:39)

이차원 좌표평면에서 각 점의 가중치가 주어질 때 영역에 포함되는 점의 가중치의 합을 온라인으로 구하는 문제이므로 이차원 동적 세그먼트 트리나 퍼시스턴트 세그먼트 트리를 사용하면 쉽게 해결할 수 있다. $x$축 $y$축 모두 좌표 압축이 필요하다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
const ll MOD = 1e9;
 
int n, q;
struct star {
    int x, y;
    ll cnt;
    bool operator<(const star &other) const {
        return x < other.x;
    }
} stars[101010];
vector<int> xlist, ylist;
 
int X, Y;
 
struct segtree {
    struct node {
        node* l;
        node* r;
        ll sum;
        node() { l = r = NULL; sum = 0; }
    };
    node* root[101010];
 
    void init(node* v, int st, int ed) {
        if (st == ed) {
            return;
        }
        int mid = (st + ed) / 2;
        v->l = new node();
        v->r = new node();
        init(v->l, st, mid);
        init(v->r, mid+1, ed);
        v->sum = v->l->sum + v->r->sum;
    }
 
    void update(node* prev, node* now, int st, int ed, int idx, ll diff) {
        if (st == idx && ed == idx) {
            now->sum += prev->sum + diff;
            return;
        }
        int mid = (st + ed) / 2;
        if (idx <= mid) {
            now->l = new node();
            now->r = prev->r;
            update(prev->l, now->l, st, mid, idx, diff);
        }
        else {
            now->l = prev->l;
            now->r = new node();
            update(prev->r, now->r, mid+1, ed, idx, diff);
        }
        now->sum = now->l->sum + now->r->sum;
    }
    
    ll get(node* v, int st, int ed, int lt, int rt) {
        if (lt > ed || st > rt) return 0;
        if (lt <= st && ed <= rt) return v->sum;
        int mid = (st + ed) / 2;
        return get(v->l, st, mid, lt, rt) + get(v->r, mid+1, ed, lt, rt);
    }
} seg;
 
segtree::node* root_ptr[101010];
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> q;
 
    for (int i = 1; i <= n; i++) {
        cin >> stars[i].x >> stars[i].y >> stars[i].cnt;
        xlist.push_back(stars[i].x);
        ylist.push_back(stars[i].y);
    }
 
    sort(stars + 1, stars + n + 1);
 
    sort(xlist.begin(), xlist.end());
    xlist.erase(unique(xlist.begin(), xlist.end()), xlist.end());
    sort(ylist.begin(), ylist.end());
    ylist.erase(unique(ylist.begin(), ylist.end()), ylist.end());
 
    X = xlist.size()-1;
    Y = ylist.size()-1;
 
    for (int i = 1; i <= n; i++) {
        stars[i].x = lower_bound(xlist.begin(), xlist.end(), stars[i].x) - xlist.begin();
        stars[i].y = lower_bound(ylist.begin(), ylist.end(), stars[i].y) - ylist.begin();
    }
 
    int r = 0;
    seg.root[0] = new segtree::node();
    seg.init(seg.root[0], 0, Y);
    root_ptr[0] = seg.root[0];
    for (int i = 1; i <= n; i++) {
        seg.root[r+1] = new segtree::node();
        seg.update(seg.root[r], seg.root[r+1], 0, Y, stars[i].y, stars[i].cnt);
        root_ptr[stars[i].x] = seg.root[r+1];
        r++;
    }
 
    ll ans = 0;
    while (q--) {
        ll a, b, c, d;
        cin >> a >> b >> c >> d;
        a = (a ^ ans) % MOD + 1;
        b = (b ^ ans) % MOD + 1;
        c = (c ^ ans) % MOD + 1;
        d = (d ^ ans) % MOD + 1;
        if (a > b) swap(a, b);
        if (c > d) swap(c, d);
 
        a = lower_bound(xlist.begin(), xlist.end(), a) - xlist.begin();
        b = upper_bound(xlist.begin(), xlist.end(), b) - xlist.begin() - 1;
        c = lower_bound(ylist.begin(), ylist.end(), c) - ylist.begin();
        d = upper_bound(ylist.begin(), ylist.end(), d) - ylist.begin() - 1;
 
        ans = seg.get(root_ptr[b], 0, Y, c, d) - (a > 0 ? seg.get(root_ptr[a-1], 0, Y, c, d) : 0);
        cout << ans << "\n";
    }
 
    return 0;
}
```

## 7. 내비게이션 (2:51:50)

트리에서 두 점 사이의 거리 쿼리를 해결하는 문제이므로 $O(\log N)$에 LCA를 구하는 쿼리와 동치이다. LCA와 두 지점 사이의 거리는 루트로부터의 거리를 전처리하면 $O(1)$에 구할 수 있기 때문이다.

LCA를 구하는 것은 부모 노드로 가는 함수를 그래프로 나타낸 함수형 그래프에서 스파스 테이블을 사용하면 $O(\log N)$에 가능하다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
 
int n, q;
vector< pair<int, ll> > adj[101010];
int par[20][101010], dep[101010], lg2[101010];
ll dis[101010];
 
void DFS(int v, int prev = -1) {
    for (int i = 0; i < adj[v].size(); i++) {
        int u = adj[v][i].first;
        ll w = adj[v][i].second;
        if (u == prev) continue;
        par[0][u] = v;
        dep[u] = dep[v] + 1;
        dis[u] = dis[v] + w;
        DFS(u, v);
    }
}
 
int lca(int u, int v) {
    if (dep[u] > dep[v]) swap(u, v);
    while (dep[u] < dep[v]) {
        int k = lg2[dep[v] - dep[u]];
        v = par[k][v];
    }
 
    if (u == v) return u;
 
    for (int k = 19; k >= 0; k--) {
        if (par[k][v] != par[k][u]) {
            v = par[k][v];
            u = par[k][u];
        }
    }
 
    return par[0][v];
}
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    lg2[1] = 0;
    for (int i = 2; i <= 100000; i++) {
```

## 8. 예약 시스템 (3:46:18)

https://www.acmicpc.net/problem/16670

현재까지의 예약에서 $k$번째 식사가 끝나는 시각을 $a_k$라 두자. 그러면 $a_k = \max(a_{k-1}, t_k) + d_k$이고 이 점화식을 끝까지 전개하게 되면

$$
a_k = \max_{1 \leq i \leq k} {\{t_i + (d_i + d_{i+1} + \cdots + d_k)\}} = \max_{1 \leq i \leq k} {(t_i + S_k - S_{i-1})}
$$

로 전개할 수 있다. 즉, 문제에서 물어보는 질의를 위 값을 구하는 것으로 해결할 수 있다. 여기서 식사가 계속 추가되거나 제거되므로 세그먼트 트리를 통해 $S_k$와 $t_i - S_{i-1}$의 $\max$ 값을 구하면 된다.

문제는 삽입을 세그먼트 트리로 어떻게 구현하냐는 건데 $t$ 시각에 예약한 사람이 모든 때에 최대 한 명이라는 사실을 이용해 삽압을 $t$ 인덱스에 값 추가로, 제거를 $t$ 인덱스에서 값 제거로 생각하면 편하다. 구체적으로 다음과 같이 처리하자.

- $t$ 추가
    - $S_k$를 구하는 세그먼트 트리에서 $t$ 인덱스에 $d$ 추가
    - $\max(t_i - S_{i-1})$을 구하는 세그먼트 트리에서 인덱스 $t$에 $t_i - S_{i-1}$ 추가
    - $\max (t_i - S_{i-1})$을 구하는 세그먼트 트리에서 구간 $[t+1, \infty)$의 값을 $d$ 감소시키기
- $t$ 제거
    - $S_k$를 구하는 세그먼트 트리에서 $t$ 인덱스의 값을 $0$으로 초기화
    - $\max(t_i - S_{i-1})$을 구하는 세그먼트 트리에서 인덱스 $t$를 $-\infty$로 초기화
    - $\max (t_i - S_{i-1})$을 구하는 세그먼트 트리에서 구간 $[t+1, \infty)$의 값을 $d$ 증가시키기
- 대기 시간을 구하는 쿼리
    - $t$번째 예약한 사람보다 우선순위가 뒤이기 때문에 대기 없이 들어갈 수 있는 가장 앞선 시각은
        - $S_k$를 구하는 세그먼트 트리에서 $[1, t]$의 합과
        - $\max(t_i - S_{i-1})$을 구하는 세그먼트 트리에서 $[1, t]$의 최댓값
    - 을 더한 값이다. 따라서 이 값이 $t$보다 작으면 대기 없이 들어갈 수 있고 $t$보다 크면 남은 시간만큼 기다려야 함

시간복잡도는 $t$와 $d$의 범위가 $N$일 때 $O(N \log N)$이다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n = 1000000, q, qrys[303030][2];
 
struct segtree {
    ll sum[4040404], mx[4040404], lazy[4040404];
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            sum[v] = mx[v] = lazy[v] = 0;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mx[v] = max(mx[2 * v], mx[2 * v + 1]);
    }
 
    void prop(int v, int st, int ed) {
        mx[v] += lazy[v];
        if (st != ed) {
            lazy[2 * v] += lazy[v];
            lazy[2 * v + 1] += lazy[v];
        }
        lazy[v] = 0;
    }
 
    void add(int v, int st, int ed, int t, ll d) {
        prop(v, st, ed);
        if (st == t && ed == t) {
            sum[v] = d;
            mx[v] = t - get_sum(1, 1, n, 1, t-1);
            return;
        }
        if (t < st || t > ed) return;
        int mid = (st + ed) / 2;
        add(2 * v, st, mid, t, d);
        add(2 * v + 1, mid + 1, ed, t, d);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mx[v] = max(mx[2 * v], mx[2 * v + 1]);
    }
 
    void rem(int v, int st, int ed, int t) {
        prop(v, st, ed);
        if (st == t && ed == t) {
            sum[v] = 0;
            mx[v] = -1e18;
            return;
        }
        if (t < st || t > ed) return;
        int mid = (st + ed) / 2;
        rem(2 * v, st, mid, t);
        rem(2 * v + 1, mid + 1, ed, t);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mx[v] = max(mx[2 * v], mx[2 * v + 1]);
    }
 
    void update(int v, int st, int ed, int lt, int rt, ll diff) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) {
            lazy[v] = diff;
            prop(v, st, ed);
            return;
        }
        if (lt > ed || st > rt) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, lt, rt, diff);
        update(2 * v + 1, mid + 1, ed, lt, rt, diff);
        sum[v] = sum[2 * v] + sum[2 * v + 1];
        mx[v] = max(mx[2 * v], mx[2 * v + 1]);
    }
 
    ll get_sum(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) return sum[v];
        if (lt > ed || st > rt) return 0;
        int mid = (st + ed) / 2;
        return get_sum(2 * v, st, mid, lt, rt) + get_sum(2 * v + 1, mid + 1, ed, lt, rt);
    }
 
    ll get_mx(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) return mx[v];
        if (lt > ed || st > rt) return 0;
        int mid = (st + ed) / 2;
        return max(get_mx(2 * v, st, mid, lt, rt), get_mx(2 * v + 1, mid + 1, ed, lt, rt));
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> q;
 
    seg.init(1, 1, n);
 
    for (int i = 1; i <= q; i++) {
        char c;
        cin >> c;
        if (c == '+') {
            int t, d;
            cin >> t >> d;
            seg.add(1, 1, n, t, d);
            seg.update(1, 1, n, t+1, n, -d);
            qrys[i][0] = t;
            qrys[i][1] = d;
        }
        if (c == '-') {
            int j, t, d;
            cin >> j;
            t = qrys[j][0];
            d = qrys[j][1];
            seg.rem(1, 1, n, t);
            seg.update(1, 1, n, t+1, n, +d);
        }
        if (c == '?') {
            int t;
            cin >> t;
            cout << max(0LL, seg.get_sum(1, 1, n, 1, t) + seg.get_mx(1, 1, n, 1, t) - t) << "\n";
        }
    }
 
    return 0;
}
```

## 9. 스코어링

https://www.acmicpc.net/problem/16232

주어진 한 스테이지를 거칠 때마다 $(-\infty, k_i)$ 구간의 값은 $a_k$ 증가하고 $[k_i, \infty)$ 구간의 값은 $b_k$ 증가한다. 이때 $a_k \leq b_k$이므로 이 구간은 항상 겹치지 않는다.

우리는 이 스테이지를 하나의 함수로 볼 수 있으며 이렇게 만들어진 함수는 일대일함수가 된다. 따라서, $(-\infty, a_i), (k_i, b_i)$로 함수를 표시할 때 두 함수 사이의 합성함수도 이와 같은 꼴로 표시할 수 있다.

따라서 머지 소트 트리로 구간의 합성함수를 순서쌍의 나열로 나타내자. 그러면 각 쿼리마다 $O(\log N)$개의 구간의 합으로 표현되고, 시작 값을 이용해 구간 합성함수를 지난 값을 찾는데 $O(\log N)$의 시간이 필요하므로 전체 시간복잡도는 $O(N \log^2 N)$이 된다.

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;
 
int n, arr[202020][3];

struct segtree {
    vector<pii> bound[808080];
    void mer(int dst, int lt, int rt) {
        bound[dst].push_back({-2e9, bound[lt][0].second + bound[rt][0].second});
        int p = 0;
        for (int i = 1; i < bound[lt].size(); i++) {
            while (p+1 < bound[rt].size() && bound[lt][i].first + bound[lt][i].second >= bound[rt][p+1].first) p++;
            bound[dst].push_back({bound[lt][i].first, bound[lt][i].second + bound[rt][p].second});
        }
        p = 0;
        for (int i = 1; i < bound[rt].size(); i++) {
            while (p+1 < bound[lt].size() && bound[lt][p+1].first + bound[lt][p+1].second <= bound[rt][i].first) p++;

            if (i+1 < bound[rt].size() && bound[lt][p].first + bound[lt][p].second >= bound[rt][i+1].first) continue;
            if (p+1 < bound[lt].size() && bound[rt][i].first - bound[lt][p].second >= bound[lt][p+1].first) continue;

            bound[dst].push_back({bound[rt][i].first - bound[lt][p].second, bound[lt][p].second + bound[rt][i].second});
        }

        sort(bound[dst].begin(), bound[dst].end());
        bound[dst].erase(unique(bound[dst].begin(), bound[dst].end()), bound[dst].end());

    }

    void init(int v, int st, int ed) {
        if (st == ed) {
            bound[v].push_back({-2e9, arr[st][0]});
            bound[v].push_back({arr[st][2], arr[st][1]});
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        mer(v, 2 * v, 2 * v + 1);
    }

    int query(int v, int val) {
        int idx = upper_bound(bound[v].begin(), bound[v].end(), (pii){val, 2e9}) - bound[v].begin() - 1;
        assert(idx >= 0);
        return val + bound[v][idx].second;
    }

    int get(int v, int st, int ed, int lt, int rt, int val) {
        if (lt <= st && ed <= rt) return query(v, val);
        if (lt > ed || st > rt) return val;
        int mid = (st + ed) / 2;
        return get(2 * v + 1, mid + 1, ed, lt, rt, get(2 * v, st, mid, lt, rt, val));
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> arr[i][0] >> arr[i][1] >> arr[i][2];

    seg.init(1, 1, n);

    for (int i = 1; i <= n; i++) {
        int ans = 0;
        if (i > 1) ans = seg.get(1, 1, n, 1, i-1, ans);
        if (i < n) ans = seg.get(1, 1, n, i+1, n, ans);
        cout << ans << "\n";
    }
 
    return 0;
}
```

## 11. 카드 놀이

다음과 같이 합과 곱을 정의하자.

- 길이가 52인 배열 $a, b, c$에 대해
    - $c = a+ b$라는 것은 $\forall i, c_i = a_i + b_i$과 동치이다.
    - $c = a \times b$라는 것은 $\forall i, c_i = a_{b_i}$과 동치이다.

그러면 주어진 교환 연산은 구간 $[a, b)$에서 항등배열 $I_i = i$의 $I_i = j, I_j = i$로 바꾼 변형 순열 $P$를 곱하는 것이고, 주어진 질의는 $[a, b)$에서 모든 배열의 합의 $51$번 인덱스를 출력하는 것이다.

이를 단순히 세그먼트 트리로 구현하면 $O(52Q \log N)$이 되며 TLE를 받는다.

(시간을 줄이는 방법은 잘 모르겠음)

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll, ll> pll;
 
int n, q;
 
struct cards{
    int card[52];
    int& operator[](int i) {
        return card[i];
    }
    cards operator+(cards &other) {
        cards ret;
        for (int i = 0; i < 52; i++) ret[i] = card[i] + other[i];
        return ret;
    }
    cards operator*(cards &other) {
        cards ret;
        for (int i = 0; i < 52; i++) ret[i] = card[other[i]];
        return ret;
    }
    bool operator==(cards &other) {
        for (int i = 0; i < 52; i++) if (card[i] != other[i]) return false;
        return true;
    }
} E, I;

struct segtree {
    cards per[202020], lazy[202020];
    
    void init(int v, int st, int ed) {
        if (st == ed) {
            per[v] = I;
            lazy[v] = E;
            return;
        }
        int mid = (st + ed) / 2;
        init(2 * v, st, mid);
        init(2 * v + 1, mid + 1, ed);
        per[v] = per[2 * v] + per[2 * v + 1];
        lazy[v] = E;
    }

    void prop(int v, int st, int ed) {
        if (lazy[v] == E) return;
        per[v] = per[v] * lazy[v];
        if (st != ed) {
            lazy[2 * v] = lazy[2 * v] * lazy[v];
            lazy[2 * v + 1] = lazy[2 * v + 1] * lazy[v];
        }
        lazy[v] = E;
    }
 
    void update(int v, int st, int ed, int lt, int rt, cards p) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) {
            lazy[v] = p;
            prop(v, st, ed);
            return;
        }
        if (lt > ed || st > rt) return;
        int mid = (st + ed) / 2;
        update(2 * v, st, mid, lt, rt, p);
        update(2 * v + 1, mid + 1, ed, lt, rt, p);
        per[v] = per[2 * v] + per[2 * v + 1];
    }
 
    int get(int v, int st, int ed, int lt, int rt) {
        prop(v, st, ed);
        if (lt <= st && ed <= rt) return per[v][51];
        if (lt > ed || st > rt) return 0;
        int mid = (st + ed) / 2;
        return get(2 * v, st, mid, lt, rt) + get(2 * v + 1, mid + 1, ed, lt, rt);
    }
} seg;
 
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(0);
 
    cin >> n >> q;
    for (int i = 0; i < 52; i++) E[i] = i;
    for (int i = 0; i < 52; i++) I[i] = i%13 + 1;

    seg.init(1, 0, n-1);

    for (int i = 1; i <= q; i++) {
        int t, a, b, l, r;
        cin >> t;
        if (t == 0) {
            cin >> a >> b >> l >> r;
            cards P = E;
            swap(P[l], P[r]);
            seg.update(1, 0, n-1, a, b-1, P);
        }
        if (t == 1) {
            cin >> a >> b;
            cout << seg.get(1, 0, n-1, a, b-1) << "\n";
        }
    }
 
    return 0;
}
```
